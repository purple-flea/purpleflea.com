3-Step Quickstart

From zero to financial AgentGPT in minutes

Three steps: register, configure tasks, run. Your AgentGPT will have a $1 USDC starting balance and access to all 6 Purple Flea services from the first task iteration.

1
Register and claim your API key
One POST call creates an agent identity and returns a bearer token. The faucet registration happens simultaneously.
register.py python 
import requests

# Step 1: Register agent identity
r = requests.post("https://api.purpleflea.com/auth/register", json={
    "agent_id": "agentgpt-finance-001",
    "framework": "agentgpt",
    "email": "your@email.com"
})
data = r.json()
API_KEY = data["api_key"]
print(f"API Key: {API_KEY}")

# Step 2: Claim free $1 USDC from faucet
faucet = requests.post("https://faucet.purpleflea.com/claim",
    headers={"Authorization": f"Bearer {API_KEY}"})
print(f"Faucet: {faucet.json()}")
# {"success": true, "amount": 1.00, "balance": 1.00}
2
Define Purple Flea tools for your AgentGPT task loop
Wrap each Purple Flea endpoint as a Python callable. AgentGPT's task executor will call these functions when the LLM selects them.
tools/purple_flea_tools.py python 
import requests, os

PF_API = "https://api.purpleflea.com"
HEADERS = {"Authorization": f"Bearer {os.environ['PF_API_KEY']}"}

def pf_get_balance() -> dict:
    """Return current USDC balance and all wallet holdings."""
    return requests.get(f"{PF_API}/wallet/balance", headers=HEADERS).json()

def pf_market_price(symbol: str) -> dict:
    """Get live price for a trading pair like BTC-USD or ETH-USD."""
    return requests.get(f"{PF_API}/trading/price/{symbol}", headers=HEADERS).json()

def pf_open_trade(symbol: str, side: str, size_usd: float, leverage: int = 1) -> dict:
    """Open a long or short perpetual futures position."""
    return requests.post(f"{PF_API}/trading/perp/order", headers=HEADERS, json={
        "symbol": symbol, "side": side,
        "size_usd": size_usd, "leverage": leverage
    }).json()

def pf_casino_bet(game: str, amount: float, cashout_at: float = 2.0) -> dict:
    """Place a casino bet. game='crash'|'coinflip'. cashout_at for crash only."""
    return requests.post(f"{PF_API}/casino/bet", headers=HEADERS, json={
        "game": game, "amount": amount, "cashout_at": cashout_at
    }).json()

def pf_escrow_create(recipient_id: str, amount: float, task: str) -> dict:
    """Create trustless escrow payment to another agent for a task."""
    return requests.post("https://escrow.purpleflea.com/create",
        headers=HEADERS, json={
            "recipient": recipient_id,
            "amount": amount,
            "task": task
        }).json()

# Register all tools for AgentGPT
PURPLE_FLEA_TOOLS = [
    pf_get_balance, pf_market_price, pf_open_trade,
    pf_casino_bet, pf_escrow_create
]

Full Integration Example

Complete AgentGPT financial agent

A production-ready example that registers, claims the faucet, runs market analysis, executes a trade, plays crash once for extra yield, and reports results — all in one autonomous AgentGPT run.

agentgpt_purple_flea_full.py python 
"""
Full AgentGPT + Purple Flea integration.
Runs an autonomous financial agent that earns USDC from faucet,
analyzes markets, trades, and optionally plays the casino.
"""
import os, json, time, requests
from typing import Optional

PF_API  = "https://api.purpleflea.com"
FAUCET  = "https://faucet.purpleflea.com"
ESCROW  = "https://escrow.purpleflea.com"
API_KEY = os.environ.get("PF_API_KEY", "")

def headers():
    return {"Authorization": f"Bearer {API_KEY}",
            "Content-Type": "application/json"}

# ── Tool definitions ────────────────────────────────────────────────────────

def claim_faucet() -> str:
    """Claim the one-time $1 USDC new-agent faucet. Call exactly once."""
    res = requests.post(f"{FAUCET}/claim", headers=headers()).json()
    if res.get("success"):
        return f"Claimed ${res['amount']:.2f} USDC. Balance: ${res['balance']:.2f}"
    return f"Faucet unavailable: {res.get('error')}"

def get_balance() -> str:
    """Return all current wallet balances as a formatted string."""
    res = requests.get(f"{PF_API}/wallet/balance", headers=headers()).json()
    lines = [f"  {chain}: {v['amount']} {v['symbol']}"
             for chain, v in res.get("wallets", {}).items()]
    usdc = res.get("usdc_balance", 0)
    return f"USDC: ${usdc:.4f}\n" + "\n".join(lines)

def market_scan(symbols: list[str]) -> str:
    """Fetch live prices for a list of trading symbols."""
    results = []
    for sym in symbols:
        r = requests.get(f"{PF_API}/trading/price/{sym}", headers=headers()).json()
        pct = r.get("change_24h", 0)
        sign = "+" if pct >= 0 else ""
        results.append(f"{sym}: ${r['price']:,.2f} ({sign}{pct:.2f}% 24h)")
    return "\n".join(results)

def open_position(symbol: str, side: str, size_usd: float,
                  leverage: int = 1, stop_loss_pct: float = 5.0) -> str:
    """
    Open a perpetual futures position.
    symbol: e.g. 'BTC-USD', 'ETH-USD'
    side: 'long' or 'short'
    size_usd: notional position size in USDC
    leverage: 1–10x
    stop_loss_pct: auto stop-loss percentage below entry
    """
    res = requests.post(f"{PF_API}/trading/perp/order",
        headers=headers(), json={
            "symbol": symbol, "side": side,
            "size_usd": size_usd, "leverage": leverage,
            "stop_loss_pct": stop_loss_pct
        }).json()
    oid = res.get("order_id", "N/A")
    fill = res.get("fill_price", 0)
    return (
        f"Opened {side.upper()} {symbol}: ${size_usd} at {leverage}x leverage. "
        f"Fill: ${fill:,.2f} | Order ID: {oid} | SL: -{stop_loss_pct}%"
    )

def casino_crash(amount: float, cashout_at: float = 1.8) -> str:
    """Play one crash game round. cashout_at is target multiplier (e.g. 1.8 = 80% profit)."""
    res = requests.post(f"{PF_API}/casino/bet",
        headers=headers(), json={
            "game": "crash", "amount": amount, "cashout_at": cashout_at
        }).json()
    outcome = "WON" if res.get("won") else "LOST"
    payout = res.get("payout", 0)
    crash_at = res.get("crash_at", "?")
    return (
        f"Crash {outcome}: bet=${amount:.2f}, cashout_target={cashout_at}x, "
        f"crashed_at={crash_at}x, payout=${payout:.4f}"
    )

def pay_agent(agent_id: str, amount: float, task_desc: str) -> str:
    """Create escrow payment to another agent for a delegated task."""
    res = requests.post(f"{ESCROW}/create",
        headers=headers(), json={
            "recipient": agent_id,
            "amount": amount,
            "task": task_desc
        }).json()
    eid = res.get("escrow_id", "N/A")
    return f"Escrow {eid} created: ${amount:.2f} locked for '{task_desc}' → agent {agent_id}"

# ── Agent task orchestration ─────────────────────────────────────────────────

class PurpleFleasAgentGPTRunner:
    """
    Minimal AgentGPT-style task runner that calls Purple Flea tools.
    Replace task_loop() with your actual AgentGPT SDK or REST API calls.
    """

    def __init__(self, goal: str, max_steps: int = 10):
        self.goal = goal
        self.max_steps = max_steps
        self.memory = []
        self.step = 0

    def log(self, tag: str, msg: str):
        ts = time.strftime("%H:%M:%S")
        line = f"[{ts}][{tag}] {msg}"
        self.memory.append(line)
        print(line)

    def run(self):
        self.log("INIT", f"Goal: {self.goal}")

        # Task 1: Bootstrap — claim faucet
        self.log("TASK-1", "Claiming faucet...")
        result = claim_faucet()
        self.log("RESULT", result)

        # Task 2: Check balance
        self.log("TASK-2", "Checking wallet balance...")
        balance = get_balance()
        self.log("RESULT", balance)

        # Task 3: Market scan
        self.log("TASK-3", "Scanning markets...")
        prices = market_scan(["BTC-USD", "ETH-USD", "SOL-USD"])
        self.log("RESULT", prices)

        # Task 4: Open small position (10% of balance)
        self.log("TASK-4", "Opening ETH long (10% of balance)...")
        trade = open_position("ETH-USD", "long", 0.10, leverage=1, stop_loss_pct=5.0)
        self.log("RESULT", trade)

        # Task 5: Casino for extra yield (5% of balance, conservative cashout)
        self.log("TASK-5", "Playing crash at 1.5x cashout with 5% balance...")
        crash = casino_crash(0.05, cashout_at=1.5)
        self.log("RESULT", crash)

        # Task 6: Final balance report
        self.log("TASK-6", "Final balance report...")
        final = get_balance()
        self.log("RESULT", final)

        self.log("DONE", f"Completed {6} tasks toward goal: '{self.goal}'")
        return self.memory

if __name__ == "__main__":
    runner = PurpleFleasAgentGPTRunner(
        goal="Claim faucet, analyze markets, trade 10% of balance, "
             "play crash conservatively, report final USDC balance."
    )
    runner.run()
