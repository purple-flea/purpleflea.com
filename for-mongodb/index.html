Why MongoDB for AI Agent Trade History

Every trading strategy produces differently shaped data. A momentum strategy records RSI signals; a casino agent records game predictions; an escrow agent records multi-party terms. MongoDB's document model stores all of them naturally in one collection, and its aggregation framework turns raw trades into actionable P&L reporting without ETL.

📄

Flexible Schema Per Strategy

A momentum trade document carries different fields than a market-making trade. MongoDB stores both in one collection — no schema migrations when you add a new strategy type.

📈

Group trades by strategy, bucket by time window, compute running P&L, and calculate win rates — all server-side without moving data to a separate analytics system.

🔀

🕑

Python Repository

AgentTradeRepository: Save, Query, and Analyze

The AgentTradeRepository class wraps PyMongo to provide typed methods for every common trade history operation.

trade_repository.py
Python 
from pymongo import MongoClient, ASCENDING, DESCENDING
from pymongo.collection import Collection
from bson.decimal128 import Decimal128
from bson.objectid import ObjectId
from datetime import datetime, timezone
from typing import Optional, List, Dict, Any
import decimal

def d128(value) -> Decimal128:
    """Convert float or string to Decimal128 for monetary precision."""
    return Decimal128(decimal.Decimal(str(value)))

class AgentTradeRepository:
    """
    MongoDB repository for Purple Flea agent trade history.
    All monetary values stored as Decimal128 to avoid float rounding.
    """

    def __init__(self, mongo_uri: str, db_name: str = "purpleflea"):
        client = MongoClient(mongo_uri)
        db = client[db_name]
        self.trades: Collection = db["trades"]
        self.wallets: Collection = db["wallets"]
        self._ensure_indexes()

    def _ensure_indexes(self):
        # Compound index for agent + time queries
        self.trades.create_index([
            ("agent_id", ASCENDING),
            ("opened_at", DESCENDING),
        ])
        # Strategy performance queries
        self.trades.create_index([
            ("agent_id", ASCENDING),
            ("strategy", ASCENDING),
        ])
        # Market-level queries
        self.trades.create_index("market")

    # ── Write ────────────────────────────────────────────────────────────────

    def save_trade(
        self,
        agent_id: str,
        strategy: str,
        market: str,
        side: str,
        size: float,
        entry_price: float,
        exit_price: float,
        pf_tx_id: str,
        fees: float = 0.0,
        signals: dict = None,
        extra: dict = None
    ) -> str:
        """
        Persist a closed trade. Returns the inserted MongoDB document _id.
        Monetary fields use Decimal128 for exact decimal arithmetic.
        """
        pnl = (exit_price - entry_price) * size
        if side == "SHORT":
            pnl = -pnl

        doc = {
            "agent_id": agent_id,
            "strategy": strategy,
            "market": market,
            "side": side,
            "size": d128(size),
            "entry_price": d128(entry_price),
            "exit_price": d128(exit_price),
            "pnl": d128(pnl - fees),
            "fees": d128(fees),
            "pf_tx_id": pf_tx_id,
            "opened_at": datetime.now(timezone.utc),
            "closed_at": datetime.now(timezone.utc),
            "signals": signals or {},
            **(extra or {}),
        }
        result = self.trades.insert_one(doc)
        return str(result.inserted_id)

    # ── Read ─────────────────────────────────────────────────────────────────

    def get_recent_trades(
        self, agent_id: str, limit: int = 50
    ) -> List[Dict]:
        cursor = self.trades.find(
            {"agent_id": agent_id},
            sort=[("opened_at", DESCENDING)],
            limit=limit
        )
        return [{**doc, "_id": str(doc["_id"])} for doc in cursor]

    # ── P&L Aggregation ───────────────────────────────────────────────────────

    def get_pnl_by_strategy(
        self, agent_id: str, days_back: int = 30
    ) -> List[Dict]:
        """
        Aggregation pipeline: group trades by strategy, compute
        total P&L, win rate, trade count, and average trade size.
        """
        since = datetime.now(timezone.utc).replace(
            hour=0, minute=0, second=0
        )
        pipeline = [
            {"$match": {
                "agent_id": agent_id,
                "opened_at": {"$gte": since}
            }},
            {"$group": {
                "_id": "$strategy",
                "total_pnl": {"$sum": {"$toDouble": "$pnl"}},
                "trade_count": {"$sum": 1},
                "wins": {"$sum": {"$cond": [
                    {"$gt": [{"$toDouble": "$pnl"}, 0]},
                    1, 0
                ]}},
                "avg_size": {"$avg": {"$toDouble": "$size"}},
                "total_fees": {"$sum": {"$toDouble": "$fees"}},
            }},
            {"$addFields": {
                "win_rate": {"$divide": ["$wins", "$trade_count"]},
                "strategy": "$_id",
            }},
            {"$sort": {"total_pnl": -1}},
        ]
        return list(self.trades.aggregate(pipeline))

    def get_daily_pnl(self, agent_id: str, days_back: int = 90) -> List[Dict]:
        """Daily P&L time-series via $dateToString bucketing."""
        pipeline = [
            {"$match": {"agent_id": agent_id}},
            {"$group": {
                "_id": {"$dateToString": {
                    "format": "%Y-%m-%d",
                    "date": "$opened_at"
                }},
                "daily_pnl": {"$sum": {"$toDouble": "$pnl"}},
                "trades": {"$sum": 1},
            }},
            {"$sort": {"_id": ASCENDING}},
        ]
        return list(self.trades.aggregate(pipeline))
Aggregation Pipelines

Understanding the P&L Aggregation Pipeline

MongoDB's aggregation framework lets you express multi-stage transformations server-side. Here is how each stage of the get_pnl_by_strategy pipeline works.

$match
Filter: Narrow to the target agent's trades within the time window. Hits the compound index on (agent_id, opened_at) for a fast scan.
$group
Aggregate: Group documents by strategy field. Compute $sum of P&L, count of trades, count of winning trades, and average size. The $toDouble converts Decimal128 to float for arithmetic.
$addFields
Derive: Compute win_rate = wins / trade_count as a derived field. Copy _id to a human-readable strategy key in the output document.
$sort
Rank: Sort strategies descending by total_pnl. Best-performing strategy appears first in the result set.
Change Streams

React to Wallet Balance Updates in Real Time

When Purple Flea webhooks write a new balance document to MongoDB, your agent should react immediately — not on the next polling cycle. MongoDB change streams provide a persistent cursor that fires an event for every insert, update, or delete.

wallet_watcher.py
Python 
from pymongo import MongoClient
import threading, httpx, time

client = MongoClient("mongodb+srv://user:pass@cluster.mongodb.net/")
db = client["purpleflea"]
http = httpx.Client(headers={"Authorization": "Bearer pf_live_yourkey"})

def watch_balance(agent_id: str, min_balance: float = 1.0):
    """
    Watch the wallets collection for balance changes for this agent.
    Fires an alert if USDC balance drops below min_balance.
    Resumes automatically if the connection drops (resume token).
    """
    pipeline = [
        {"$match": {
            "operationType": {"$in": ["insert", "update", "replace"]},
            "fullDocument.agent_id": agent_id,
        }}
    ]

    resume_token = None

    while True:
        try:
            opts = {"full_document": "updateLookup"}
            if resume_token:
                opts["resume_after"] = resume_token

            with db["wallets"].watch(pipeline, **opts) as stream:
                for change in stream:
                    resume_token = stream.resume_token
                    doc = change.get("fullDocument", {})
                    usdc = float(str(doc.get("usdc", 0)))

                    if usdc < min_balance:
                        on_low_balance(agent_id, usdc)

                    if usdc == 0:
                        on_zero_balance(agent_id)

        except Exception as e:
            print(f"Stream error: {e} — resuming in 5s")
            time.sleep(5)   # reconnect with resume token

def on_low_balance(agent_id: str, balance: float):
    """Pause trading and notify when balance is critically low."""
    print(f"ALERT: {agent_id} balance dropped to {balance} USDC")
    # Optionally claim from faucet if eligible
    http.post("https://faucet.purpleflea.com/api/claim",
              json={"agent_id": agent_id})

def on_zero_balance(agent_id: str):
    """Hard stop — agent should not trade with zero balance."""
    print(f"HALT: {agent_id} has zero balance. Stopping strategy.")

# Start watcher in background thread
watcher = threading.Thread(
    target=watch_balance,
    args=("agent-007", 1.0),
    daemon=True
)
watcher.start()

Time-Series Collections

OHLCV Data in MongoDB Time-Series Collections

MongoDB's time-series collection type stores OHLCV candles with automatic time-based sharding, efficient compression, and fast range queries. Feed it from Purple Flea's trading API ticker endpoint for a private market data store.

  • Create the collection once with timeseries option specifying the time field
  • Specify metaField: "market" so MongoDB co-locates candles by market for efficient range queries
  • Use granularity: "seconds" for 1m candles, "minutes" for hourly
  • Query with $gte/$lte on the time field — the index is automatic
  • Use $bucketAuto in aggregations to resample 1m candles into 1h candles
  • Compression ratios of 10-30x over regular collections for OHLCV workloads
  • TTL index auto-expires candles older than your lookback window
ohlcv_collection.py
Python 
from pymongo import MongoClient
from datetime import datetime, timezone
import httpx

client = MongoClient("mongodb+srv://...")
db = client["purpleflea"]

# Create time-series collection (run once)
try:
    db.create_collection("ohlcv", timeseries={
        "timeField": "ts",
        "metaField": "market",
        "granularity": "seconds"
    })
except Exception:
    pass   # already exists

ohlcv = db["ohlcv"]
http = httpx.Client(headers={
    "Authorization": "Bearer pf_live_yourkey"
})

def ingest_candle(market: str):
    """Fetch latest ticker and insert as a candle."""
    resp = http.get(
        f"https://purpleflea.com/api/trading/markets/{market}/ticker"
    )
    tick = resp.json()
    ohlcv.insert_one({
        "ts": datetime.now(timezone.utc),
        "market": market,
        "open": tick["open"],
        "high": tick["high"],
        "low": tick["low"],
        "close": tick["price"],
        "volume": tick["volume_24h"],
    })

def get_candles(market: str,
                start: datetime, end: datetime) -> list:
    """Retrieve candles for a time range."""
    cursor = ohlcv.find({
        "market": market,
        "ts": {"$gte": start, "$lte": end}
    }).sort("ts", 1)
    return list(cursor)
Fleet Analytics

Multi-Agent Analytics Across a Trading Fleet

When you run multiple agents on Purple Flea, you want to compare performance, identify which strategies and which agents are profitable, and detect correlated losses that indicate a shared risk factor. MongoDB aggregation handles the entire analytics layer.

fleet_analytics.py
Python 
def fleet_leaderboard(repo: AgentTradeRepository) -> list:
    """
    Rank all agents by total P&L with win rate and Sharpe proxy.
    Returns sorted list of agent performance summaries.
    """
    pipeline = [
        {"$group": {
            "_id": "$agent_id",
            "total_pnl": {"$sum": {"$toDouble": "$pnl"}},
            "trade_count": {"$sum": 1},
            "wins": {"$sum": {"$cond": [
                {"$gt": [{"$toDouble": "$pnl"}, 0]}, 1, 0
            ]}},
            "pnl_list": {"$push": {"$toDouble": "$pnl"}},
            "strategies": {"$addToSet": "$strategy"},
            "markets": {"$addToSet": "$market"},
            "last_trade": {"$max": "$opened_at"},
        }},
        {"$addFields": {
            "win_rate": {"$divide": ["$wins", "$trade_count"]},
            "agent_id": "$_id",
            # Std deviation proxy using avg absolute deviation
            "avg_pnl_abs": {"$avg": {"$map": {
                "input": "$pnl_list",
                "as": "p",
                "in": {"$abs": "$$p"}
            }}}
        }},
        {"$sort": {"total_pnl": -1}},
        {"$project": {"pnl_list": 0}},   # drop large array from output
    ]
    return list(repo.trades.aggregate(pipeline))

def correlated_losses(
    repo: AgentTradeRepository,
    window_minutes: int = 60
) -> list:
    """
    Find time windows where multiple agents lost simultaneously.
    Indicates shared market risk — useful for position sizing review.
    """
    pipeline = [
        {"$match": {
            "pnl": {"$lt": 0}     # only losing trades
        }},
        {"$group": {
            "_id": {
                "window": {"$dateTrunc": {
                    "date": "$opened_at",
                    "unit": "minute",
                    "binSize": window_minutes
                }}
            },
            "agents_losing": {"$addToSet": "$agent_id"},
            "total_loss": {"$sum": {"$toDouble": "$pnl"}},
            "loss_count": {"$sum": 1},
        }},
        {"$addFields": {
            "agent_count": {"$size": "$agents_losing"}
        }},
        {"$match": {"agent_count": {"$gte": 3}}},   # 3+ agents losing = systemic
        {"$sort": {"total_loss": 1}},
    ]
    return list(repo.trades.aggregate(pipeline))
Atlas Search

Atlas Search: Find Similar Historical Patterns

MongoDB Atlas Search lets your agent query trade history using full-text and vector similarity — useful for finding past market conditions that resemble current ones, or for retrieving strategy notes by semantic meaning.

  • Index the signals sub-document to find trades with similar indicator readings
  • Store vector embeddings of market regime descriptions alongside each trade
  • Use $vectorSearch to find the 5 most similar past market conditions
  • Combine $search with $match to filter by profitability while searching
  • Auto-complete on strategy names, market identifiers, and outcome labels
  • Fuzzy match on agent notes and commentary fields
atlas_search.py
Python 
def find_similar_setups(
    repo: AgentTradeRepository,
    query_embedding: list,    # float[1536] from embedding model
    min_pnl: float = 0,      # only profitable similar trades
    top_k: int = 10
) -> list:
    """
    Find trades with similar market setup using vector search.
    Requires Atlas Vector Search index on 'embedding' field.
    """
    pipeline = [
        {"$vectorSearch": {
            "index": "trade_embeddings",
            "path": "embedding",
            "queryVector": query_embedding,
            "numCandidates": top_k * 10,
            "limit": top_k,
        }},
        {"$match": {
            "pnl": {"$gte": min_pnl}
        }},
        {"$project": {
            "strategy": 1, "market": 1,
            "pnl": 1, "signals": 1,
            "score": {"$meta": "vectorSearchScore"},
            "embedding": 0,   # exclude from output
        }},
    ]
    return list(repo.trades.aggregate(pipeline))

# Example: embed current market state, find similar winning setups
from openai import OpenAI
openai = OpenAI()

market_description = """
BTC-PERP: RSI=68, volume spike +340%,
funding rate 0.02%, price above 20-EMA.
Sentiment: cautiously bullish.
"""

embedding = openai.embeddings.create(
    model="text-embedding-3-small",
    input=market_description
).data[0].embedding

similar = find_similar_setups(repo, embedding, min_pnl=0.05)
for s in similar[:3]:
    print(f"{s['strategy']} | P&L: {s['pnl']} | score: {s['score']:.3f}")
