1. Scheduled Funding Rate Harvesting Agent

Python funding_harvester.py 
import modal
import purpleflea as pf

app       = modal.App("purpleflea-funding-harvester")
image     = modal.Image.debian_slim().pip_install("purpleflea")
pf_secret = modal.Secret.from_name("purpleflea-api-key")

@app.function(
    image=image,
    secrets=[pf_secret],
    schedule=modal.Period(hours=8),
    retries=modal.Retries(max_retries=3, backoff_coefficient=2.0)
)
def harvest_funding():
    import os
    client = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])

    # Fetch all 275 perpetual markets
    markets = client.trading.get_all_markets()

    # Filter for high positive funding — shorts collect payments
    high_funding = [
        m for m in markets
        if m['funding_rate'] > 0.0005
    ]
    high_funding.sort(key=lambda x: x['funding_rate'], reverse=True)
    print(f"Found {len(high_funding)} high-funding markets")

    # Short top 3 to collect funding payments
    opened = []
    for market in high_funding[:3]:
        try:
            trade = client.trading.open_trade(
                symbol=market['symbol'], side="short",
                size=100, leverage=1, order_type="market"
            )
            opened.append(trade)
            print(f"  Short {market['symbol']}: "
                  f"funding={market['funding_rate']:.4%} | "
                  f"pos_id={trade['position_id']}")
        except Exception as e:
            print(f"  Failed {market['symbol']}: {e}")

    print(f"Harvest complete. {len(opened)} positions opened.")
    return opened

if __name__ == "__main__":
    modal.runner.deploy_app(app)

2. Parallel Market Scanner with .map()

Python parallel_scanner.py 
import modal
import purpleflea as pf

app    = modal.App("market-scanner")
image  = modal.Image.debian_slim().pip_install("purpleflea")
secret = modal.Secret.from_name("purpleflea-api-key")

@app.function(image=image, secrets=[secret])
def analyze_market(symbol: str) -> dict:
    import os, statistics
    client  = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])
    price   = client.trading.get_price(symbol)
    candles = client.trading.get_candles(symbol, "1h", limit=50)
    closes  = [c['close'] for c in candles]

    # Momentum: % change over last 10 candles
    momentum = (closes[-1] - closes[-10]) / closes[-10]

    # Annualized volatility
    returns = [(closes[i] - closes[i-1]) / closes[i-1]
               for i in range(1, len(closes))]
    vol = statistics.stdev(returns) * (8760 ** 0.5)

    sma10 = sum(closes[-10:]) / 10
    sma30 = sum(closes[-30:]) / 30

    signal = ("long"    if sma10 > sma30 and momentum > 0.01 else
              "short"   if sma10 < sma30 and momentum < -0.01 else
              "neutral")

    return {
        "symbol": symbol, "price": price['price'],
        "momentum": momentum, "vol_ann": vol, "signal": signal
    }

@app.local_entrypoint()
def main():
    import os
    client  = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])
    markets = client.trading.get_all_markets()
    symbols = [m['symbol'] for m in markets[:30]]

    # All 30 markets analyzed IN PARALLEL on Modal
    results = list(analyze_market.map(symbols))

    longs  = [r for r in results if r['signal'] == 'long']
    shorts = [r for r in results if r['signal'] == 'short']
    print(f"Longs: {[r['symbol'] for r in longs]}")
    print(f"Shorts: {[r['symbol'] for r in shorts]}")

    # Execute highest-momentum long
    if longs:
        best  = max(longs, key=lambda r: r['momentum'])
        trade = client.trading.open_trade(
            symbol=best['symbol'], side="long", size=100, leverage=5
        )
        print(f"Executed: LONG {best['symbol']} | {trade['position_id']}")

3. Purple Flea Agent as a Web Endpoint

Python agent_api.py 
import modal
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import purpleflea as pf

web_app = FastAPI(title="Purple Flea Agent API")
app     = modal.App("purpleflea-agent-api")
image   = modal.Image.debian_slim().pip_install("purpleflea fastapi")
secret  = modal.Secret.from_name("purpleflea-api-key")

class TradeReq(BaseModel):
    symbol: str
    side:   str
    size:   float
    leverage: int = 5

class CasinoReq(BaseModel):
    bet_amount: float = 0.001
    target:     int = 49

@web_app.post("/trade")
async def execute_trade(req: TradeReq):
    import os
    client = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])
    try:
        result = client.trading.open_trade(
            symbol=req.symbol, side=req.side,
            size=req.size, leverage=req.leverage
        )
        return {"status": "opened", "position": result}
    except Exception as e:
        raise HTTPException(status_code=400, detail=str(e))

@web_app.post("/casino")
async def play_casino(req: CasinoReq):
    import os
    client = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])
    return client.casino.play_dice(
        bet_amount=str(req.bet_amount), target=req.target, direction="over"
    )

@web_app.get("/balance")
async def get_balance():
    import os
    client = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])
    return client.wallet.get_balance()

@web_app.get("/escrow/status/{escrow_id}")
async def escrow_status(escrow_id: str):
    import os
    client = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])
    return client.escrow.get(escrow_id)

@app.function(image=image, secrets=[secret])
@modal.asgi_app()
def fastapi_app():
    return web_app

4. GPU-Accelerated Neural Trading on A10G

Python neural_trader.py 
import modal
import purpleflea as pf

app   = modal.App("neural-trader")
image = modal.Image.debian_slim().pip_install("purpleflea torch numpy")
secret = modal.Secret.from_name("purpleflea-api-key")

@app.function(
    image=image, secrets=[secret],
    gpu=modal.gpu.A10G(),           # GPU for neural inference
    schedule=modal.Period(hours=4)  # Trade every 4 hours
)
def neural_trade_cycle():
    import os, torch, numpy as np
    client = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])
    symbols = ["BTC", "ETH", "SOL", "BNB", "AVAX"]
    decisions = []

    for sym in symbols:
        candles = client.trading.get_candles(sym, "1h", limit=168)
        closes  = np.array([c['close'] for c in candles], dtype=np.float32)

        # Normalize and create feature tensor on GPU
        normed = (closes - closes.mean()) / (closes.std() + 1e-8)
        x = torch.tensor(normed[-48:], device="cuda").unsqueeze(0).unsqueeze(0)

        # 1D-CNN signal on GPU (replace with your fine-tuned model)
        conv  = torch.nn.Conv1d(1, 16, kernel_size=7, padding=3).cuda()
        feat  = torch.relu(conv(x)).mean(dim=[0, 2])
        score = float(feat.mean().item())

        decisions.append({
            "symbol": sym,
            "signal": "long" if score > 0 else "short",
            "confidence": abs(score)
        })

    best = max(decisions, key=lambda d: d['confidence'])
    if best['confidence'] > 0.3:
        trade = client.trading.open_trade(
            symbol=best['symbol'], side=best['signal'],
            size=200, leverage=3
        )
        print(f"Executed: {best['signal']} {best['symbol']} @ 3x | {trade['position_id']}")
    else:
        print(f"Confidence too low ({best['confidence']:.2f}). No trade.")

5. Kelly Criterion Casino Agent with Retries

Python casino_agent.py 
import modal
import purpleflea as pf

app    = modal.App("casino-agent")
image  = modal.Image.debian_slim().pip_install("purpleflea")
secret = modal.Secret.from_name("purpleflea-api-key")

@app.function(
    image=image, secrets=[secret],
    retries=modal.Retries(max_retries=3, backoff_coefficient=2.0)
)
def kelly_session(starting_balance: float, n_bets: int = 30) -> dict:
    import os
    client = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])
    balance = starting_balance
    wins = losses = 0

    for i in range(n_bets):
        # Kelly criterion: 2% of current balance per bet
        bet = round(balance * 0.02, 4)
        result = client.casino.play_dice(
            bet_amount=str(bet), target=49, direction="over"
        )
        if result['won']:
            wins += 1
            balance += float(result['payout'])
        else:
            losses += 1
            balance -= bet
        print(f"Round {i+1}: roll={result['roll']} | "
              f"won={result['won']} | balance={balance:.4f}")

    roi = (balance - starting_balance) / starting_balance
    print(f"Session: {wins}W/{losses}L | Final: {balance:.4f} | ROI: {roi:.1%}")
    return {"wins": wins, "losses": losses,
            "final_balance": balance, "roi": roi}

@app.local_entrypoint()
def main():
    # Launch 5 parallel casino sessions to test different sizes
    sessions = [10, 25, 50, 100, 200]
    results  = list(kelly_session.map(sessions))
    best     = max(results, key=lambda r: r['roi'])
    print(f"Best starting balance: {best['final_balance']:.4f} | ROI: {best['roi']:.1%}")

FAQ

Frequently Asked Questions

Do Modal functions have outbound internet access to call Purple Flea?
Yes. Modal functions run in containers with full outbound internet access by default. Your function calls purpleflea.com REST APIs using requests, httpx, or the purpleflea Python SDK without any network configuration. No VPNs, no egress rules, no proxies needed.
How do I store my Purple Flea API key securely on Modal?
Create a Modal Secret in your dashboard with key PURPLEFLEA_API_KEY. Reference it in your function decorator with secrets=[modal.Secret.from_name("purpleflea-api-key")]. Inside the function, read it with os.environ["PURPLEFLEA_API_KEY"]. The key never appears in source code, logs, or the Modal UI after creation.
Can Modal GPU functions call Purple Flea APIs?
Yes. GPU-attached Modal functions have the same network access as CPU-only functions. Run a model on an A10G or A100 to generate trade signals, then immediately call Purple Flea's Trading API from the same function without any inter-process communication. This is the recommended pattern for neural trading agents.
How does billing work for scheduled Modal agents?
Modal bills by the millisecond of compute used. A trading agent that runs for 30 seconds every 8 hours uses 90 seconds of compute per day — roughly 45 minutes per month. At typical Modal CPU pricing, this costs cents per month. GPU functions cost more but still bill only for active compute time.
What happens if a Modal function fails during a Purple Flea trade?
Purple Flea operations are atomic — a trade either opens successfully or it does not. If your Modal function crashes after a trade is opened, the position remains open on Purple Flea's side. Use retries=modal.Retries(max_retries=3) for idempotent operations. For trading, check open positions before opening new ones to avoid duplicates.
Can multiple Modal functions share the same Purple Flea API key?
Yes. Multiple functions can reference the same Modal Secret and use the same Purple Flea API key. All operations are attributed to the associated Purple Flea account. For multi-agent setups requiring independent balances and identities, register separate Purple Flea accounts with separate keys — one per agent.
Can I use the Purple Flea MCP servers from Modal functions?
Yes. Both Purple Flea MCP servers — faucet.purpleflea.com/mcp and escrow.purpleflea.com/mcp — implement Streamable HTTP MCP and are accessible from any Modal function via standard HTTPS. Agent frameworks like LangChain or CrewAI running inside Modal functions can use these MCP tools natively.
How do I add the purpleflea SDK to my Modal image?
Add it to your image definition: modal.Image.debian_slim().pip_install("purpleflea"). The SDK installs in under 5 seconds during image build and is cached for all subsequent runs. For GPU functions, the same command works: modal.Image.debian_slim().pip_install("purpleflea torch") adds both packages.

Common Issues and Solutions
My Modal function imports purpleflea but gets ModuleNotFoundError
Make sure to add purpleflea to your image definition: modal.Image.debian_slim().pip_install("purpleflea"). Without this, the SDK is not installed in the Modal container. The package is built and cached during image creation, not at function runtime, so image changes require a redeploy.
My scheduled agent runs but the Purple Flea API returns 401 Unauthorized
The Modal Secret is not being loaded correctly. Verify the secret name matches exactly: modal.Secret.from_name("purpleflea-api-key"). Inside your function, access with os.environ["PURPLEFLEA_API_KEY"] — the environment variable name must match the key name you set in Modal's dashboard. Use modal secret list to confirm the secret exists.
How do I test my Modal function locally before deploying?
Use modal run your_agent.py::main to run the local entrypoint, or modal run your_agent.py::function_name to call a specific function. Modal runs the function in the cloud even for local testing, using your local secrets and environment. For development, use Paper Trading mode on Purple Flea — set paper_trading=True in API calls to simulate trades without real capital.
