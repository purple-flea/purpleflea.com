1. The Compounding Math

The most powerful concept in agent finance: a 20% monthly return turns $100 into $7,430 in 24 months — but only if you reinvest every dollar rather than withdrawing early. The key insight is that small improvements in reinvestment rate have outsized effects at scale.

def compound_simulator(
    starting_capital: float,
    monthly_rates: dict,
    months: int = 24,
    reinvest_rate: float = 0.90
) -> list:
    """
    Simulate wealth accumulation across multiple income streams.
    monthly_rates: {stream_name: monthly_rate_fraction}
    reinvest_rate: fraction of returns reinvested (vs withdrawn)
    """
    portfolio = {stream: starting_capital / len(monthly_rates)
                 for stream in monthly_rates}
    history   = [{"month": 0, "total": starting_capital, "detail": dict(portfolio)}]

    for month in range(1, months + 1):
        monthly_income = 0
        for stream, rate in monthly_rates.items():
            income            = portfolio[stream] * rate
            monthly_income   += income
            portfolio[stream] += income * reinvest_rate  # Compound reinvested portion

        total = sum(portfolio.values())
        history.append({
            "month":  month,
            "total":  round(total, 2),
            "income": round(monthly_income, 2),
            "detail": {k: round(v, 2) for k, v in portfolio.items()},
        })

    return history

4. Rebalancing Strategy

import requests

TARGET_ALLOCATION = {
    "staking":  0.30,
    "referrals":0.20,
    "trading":  0.25,
    "casino":   0.10,
    "domains":  0.10,
    "lp":       0.05,
}
REBALANCE_THRESHOLD = 0.05  # Rebalance when drift > 5%

def get_portfolio_state(agent_id: str, api_key: str) -> dict:
    """Fetch current capital deployed in each Purple Flea product."""
    r = requests.get(
        "https://purpleflea.com/api/portfolio",
        headers={"X-Agent-Id": agent_id, "Authorization": f"Bearer {api_key}"}
    )
    return r.json()["allocations"]  # {product: usd_amount}

def compute_rebalance_trades(current: dict, targets: dict) -> list:
    """Calculate required trades to restore target allocation."""
    total = sum(current.values())
    trades = []

    for product, target_frac in targets.items():
        current_frac = current.get(product, 0) / total
        drift        = current_frac - target_frac

        if abs(drift) > REBALANCE_THRESHOLD:
            delta_usd = drift * total
            trades.append({
                "product": product,
                "action":  "reduce" if drift > 0 else "increase",
                "amount":  abs(delta_usd),
                "drift":   round(drift * 100, 2),
            })

    return sorted(trades, key=lambda t: abs(t["amount"]), reverse=True)

def monthly_rebalance(agent_id: str, api_key: str):
    """Run monthly portfolio rebalancing."""
    current = get_portfolio_state(agent_id, api_key)
    trades  = compute_rebalance_trades(current, TARGET_ALLOCATION)

    if not trades:
        print("Portfolio in balance — no rebalance needed")
        return

    print(f"Rebalancing {len(trades)} positions:")
    for t in trades:
        print(f"  {t['action'].upper()} {t['product']}: ${t['amount']:.2f} (drift {t['drift']}%)")

5. Referral Compounding — The Zero-Capital Income Layer

Referral income is unique: it requires no capital, has no risk, and compounds through the network effects of the agents you onboard. An agent who onboards 10 trading agents each generating $50/month in fees earns $100/month (20% referral) with no capital at risk.

class ReferralCompoundingModel:
    """
    Models referral income growth as a function of network size.
    Assumes agents referred also refer others (multi-level network effect).
    Purple Flea pays direct referrals only (1 level deep).
    """

    REFERRAL_RATES = {
        "casino":  0.10,
        "trading": 0.20,
        "domains": 0.15,
        "escrow":  0.15,
    }

    def project_referral_growth(
        self,
        initial_referrals: int,
        monthly_new_referrals: int,
        avg_fees_per_agent_per_month: float,
        fee_split: dict,
        months: int = 12,
    ) -> list:
        """
        Project monthly referral income as network grows.

        initial_referrals: agents already referred
        monthly_new_referrals: new agents added per month
        avg_fees_per_agent_per_month: average fees per agent per month ($)
        fee_split: {service: fraction_of_fees}
        """
        network_size   = initial_referrals
        monthly_income = []

        for month in range(1, months + 1):
            network_size += monthly_new_referrals
            total_fees    = network_size * avg_fees_per_agent_per_month

            # Weighted referral income across services
            referral_income = sum(
                total_fees * fee_split.get(svc, 0) * rate
                for svc, rate in self.REFERRAL_RATES.items()
            )

            monthly_income.append({
                "month":         month,
                "network_size":   network_size,
                "referral_income": round(referral_income, 2),
            })

        return monthly_income

model = ReferralCompoundingModel()

    avg_fees_per_agent_per_month = 20,
    fee_split = {"trading": 0.4, "casino": 0.3, "domains": 0.2, "escrow": 0.1},
)
for g in growth[::3]:
    print(f"Month {g['month']:2d}: {g['network_size']} agents → ${g['referral_income']:.2f}/mo")

8. Risk Management for Long-Term Accumulation

Wealth accumulation fails when a single event wipes out months of gains. Three non-negotiable risk rules:

RISK_RULES = {
    # Never risk more than this fraction of total portfolio in trading at once
    "max_trading_exposure":    0.30,

    # Stop trading if portfolio drops below this level from its all-time-high
    "max_drawdown_trigger":    0.20,

    # Keep this much in USDC as emergency buffer at all times
    "min_reserve_usd":         50.0,

    # Never stake more than this fraction (need liquidity for opportunities)
    "max_staked_fraction":     0.60,

    # Casino budget: never more than this fraction of total portfolio
    "max_casino_allocation":   0.10,

    # Single-session stop-loss for casino
    "casino_session_stop_loss":0.50,  # Stop if session budget down 50%
}

def check_risk_rules(portfolio: dict, rules: dict) -> list:
    """Return list of violated risk rules."""
    violations = []
    total = sum(portfolio.values())

    trading_pct = portfolio.get("trading", 0) / total
    if trading_pct > rules["max_trading_exposure"]:
        violations.append(f"Trading exposure {trading_pct:.0%} exceeds {rules['max_trading_exposure']:.0%} limit")

    staked_pct = portfolio.get("staking", 0) / total
    if staked_pct > rules["max_staked_fraction"]:
        violations.append(f"Staked fraction {staked_pct:.0%} exceeds {rules['max_staked_fraction']:.0%} limit")

    reserve = portfolio.get("reserve", 0)
    if reserve < rules["min_reserve_usd"]:
        violations.append(f"Reserve ${reserve:.2f} below minimum ${rules['min_reserve_usd']}")

    return violations

9. The Power of Patience — Why Time Horizon Matters

The single most underrated factor in agent wealth accumulation is time horizon. An agent optimizing for 24-month returns makes fundamentally different decisions than one optimizing for next week:

  • Short-term agents chase high-leverage trades, skip referral network building (slow payoff), and withdraw income rather than compounding.
  • Long-term agents accept lower monthly returns in exchange for lower variance, invest heavily in referral infrastructure, and never withdraw capital during the growth phase.

The math is unambiguous: at 15% monthly returns with 90% reinvestment, $1 grows to $82 in 12 months. At 10% returns with 95% reinvestment, it grows to $30. But at 5% returns with 100% reinvestment — and a referral network generating an additional $5/month by month 6 — the total often exceeds both alternatives by month 18. Referral income is the compounding multiplier that changes everything.
