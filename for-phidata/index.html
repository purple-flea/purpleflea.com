Phidata Tool Classes

Toolkit Subclasses for
Casino, Wallet, and Escrow

Each Purple Flea service becomes a Phidata Toolkit. Pass toolkit instances to Agent() and they become available to the agent's function-calling model automatically.

Pythonpurpleflea_toolkits.py
import os, requests
from phi.tools import Toolkit
from typing import Optional

BASE   = "https://purpleflea.com"
FAUCET = "https://faucet.purpleflea.com"
ESCROW = "https://escrow.purpleflea.com"


class CasinoTools(Toolkit):
    """Purple Flea casino tools for Phidata agents."""

    def __init__(self, agent_id: str, api_key: str):
        super().__init__(name="casino_tools")
        self.agent_id = agent_id
        self.headers  = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
        self.register(self.play_game)
        self.register(self.get_game_history)

    def play_game(self, game: str, bet: float, action: Optional[str] = None) -> str:
        """Play a casino game (blackjack/roulette/slots/dice/poker) with a bet in USDC.

        Args:
            game:   Game name: blackjack | roulette | slots | dice | poker
            bet:    Bet amount in USDC
            action: Optional game action (hit/stand/double for blackjack)
        """
        payload = {"agent_id": self.agent_id, "game": game, "bet": bet}
        if action:
            payload["action"] = action
        resp = requests.post(f"{BASE}/casino-api/play", json=payload, headers=self.headers)
        if resp.ok:
            d   = resp.json()
            res = "WON" if d["profit"] > 0 else "LOST"
            return f"{res} {abs(d['profit']):.2f} USDC at {game}. Balance: {d['balance']:.2f} USDC."
        return f"Casino error: {resp.text}"

    def get_game_history(self, limit: int = 10) -> str:
        """Retrieve recent casino game history for this agent."""
        resp = requests.post(f"{BASE}/casino-api/history",
                             json={"agent_id": self.agent_id, "limit": limit},
                             headers=self.headers)
        return resp.json() if resp.ok else f"History error: {resp.text}"


class WalletTools(Toolkit):
    """Purple Flea wallet tools for Phidata agents."""

    def __init__(self, agent_id: str, api_key: str):
        super().__init__(name="wallet_tools")
        self.agent_id = agent_id
        self.headers  = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
        self.register(self.get_balance)
        self.register(self.transfer)
        self.register(self.get_transactions)

    def get_balance(self) -> str:
        """Get the current USDC balance for this agent's wallet."""
        resp = requests.post(f"{BASE}/wallet-api/balance",
                             json={"agent_id": self.agent_id}, headers=self.headers)
        if resp.ok:
            d = resp.json()
            return f"Balance: {d['balance']:.2f} USDC"
        return f"Wallet error: {resp.text}"

    def transfer(self, to_agent: str, amount: float) -> str:
        """Transfer USDC to another agent.

        Args:
            to_agent: Recipient agent ID
            amount:   Amount in USDC to transfer
        """
        resp = requests.post(f"{BASE}/wallet-api/transfer",
                             json={"agent_id": self.agent_id, "to_agent": to_agent, "amount": amount},
                             headers=self.headers)
        return resp.json() if resp.ok else f"Transfer error: {resp.text}"

    def get_transactions(self, limit: int = 20) -> str:
        """Get transaction history for this agent's wallet."""
        resp = requests.post(f"{BASE}/wallet-api/history",
                             json={"agent_id": self.agent_id, "limit": limit},
                             headers=self.headers)
        return resp.json() if resp.ok else f"Transaction error: {resp.text}"


class EscrowTools(Toolkit):
    """Purple Flea escrow tools for Phidata agents."""

    def __init__(self, agent_id: str, api_key: str, referral_code: str = ""):
        super().__init__(name="escrow_tools")
        self.agent_id      = agent_id
        self.referral_code = referral_code
        self.headers       = {"Authorization": f"Bearer {api_key}", "Content-Type": "application/json"}
        self.register(self.create_escrow)
        self.register(self.release_escrow)
        self.register(self.get_escrow_status)

    def create_escrow(self, to_agent: str, amount: float, condition: str) -> str:
        """Create a trustless USDC escrow contract between this agent and another.

        Args:
            to_agent:  Recipient agent ID
            amount:    Amount in USDC to lock
            condition: Release condition description
        """
        payload = {
            "from_agent": self.agent_id, "to_agent": to_agent,
            "amount": amount, "condition": condition,
        }
        if self.referral_code:
            payload["referral_code"] = self.referral_code
        resp = requests.post(f"{ESCROW}/create", json=payload, headers=self.headers)
        if resp.ok:
            d = resp.json()
            return f"Escrow {d['escrow_id']} created. {amount} USDC locked for {to_agent}. Fee: {d['fee']} USDC."
        return f"Escrow error: {resp.text}"

    def release_escrow(self, escrow_id: str) -> str:
        """Release a locked escrow contract and transfer funds to recipient."""
        resp = requests.post(f"{ESCROW}/release",
                             json={"escrow_id": escrow_id}, headers=self.headers)
        return resp.json() if resp.ok else f"Release error: {resp.text}"

    def get_escrow_status(self, escrow_id: str) -> str:
        """Check the status and details of an escrow contract."""
        resp = requests.get(f"{ESCROW}/status/{escrow_id}", headers=self.headers)
        return resp.json() if resp.ok else f"Status error: {resp.text}"
Agent & Team Configuration

Single Agent and Multi-Agent
Team Configuration

Attach toolkits to a single Agent for focused tasks, or compose a Team of specialized agents for complex financial workflows.

Single Agent

Pythonsingle_agent.py
from phi.agent import Agent
from phi.model.openai import OpenAIChat
from phi.storage.agent.sqlite import SqlAgentStorage
from purpleflea_toolkits import CasinoTools, WalletTools, EscrowTools
import os

AGENT_ID = "phi-agent-001"
API_KEY  = os.getenv("PURPLEFLEA_API_KEY")

financial_agent = Agent(
    name="FinancialAgent",
    model=OpenAIChat(id="gpt-4o"),
    tools=[
        CasinoTools(AGENT_ID, API_KEY),
        WalletTools(AGENT_ID, API_KEY),
        EscrowTools(AGENT_ID, API_KEY, referral_code="phi-ref-001"),
    ],
    instructions=[
        "Check balance before any financial action.",
        "Never bet more than 5% of current balance.",
        "Always use escrow for payments over 20 USDC.",
        "Report P&L after every casino session.",
    ],
    storage=SqlAgentStorage(
        table_name="financial_agent",
        db_file="financial_agent.db",
    ),
    add_history_to_messages=True,
    num_history_responses=10,
    markdown=True,
    show_tool_calls=True,
)

financial_agent.print_response(
    "Check my balance, then play 3 rounds of blackjack betting 5% each time.",
    stream=True
)

Phidata Workflow

Autonomous Treasury Rebalancing
Workflow

Phidata Workflows enable multi-step autonomous processes with caching, branching, and session state. This workflow monitors wallet balance, plays casino to grow it, and rebalances via escrow when thresholds are met.

Pythontreasury_rebalancing_workflow.py
from phi.workflow import Workflow, RunResponse, RunEvent
from phi.agent import Agent
from phi.model.openai import OpenAIChat
from phi.storage.workflow.sqlite import SqlWorkflowStorage
from phi.utils.log import logger
from purpleflea_toolkits import CasinoTools, WalletTools, EscrowTools
from pydantic import BaseModel, Field
from typing import Iterator
import os

API_KEY   = os.getenv("PURPLEFLEA_API_KEY")
AGENT_ID  = "treasury-workflow-001"
REBAL_THRESHOLD = 150  # Rebalance when balance exceeds this
RESERVE_TARGET  = 100  # Keep this much in treasury, send rest to escrow


class TreasuryState(BaseModel):
    starting_balance: float  = Field(default=0)
    current_balance: float   = Field(default=0)
    rounds_played: int        = Field(default=0)
    total_profit: float      = Field(default=0)
    rebalances_done: int      = Field(default=0)
    escrow_ids: list          = Field(default_factory=list)


class TreasuryRebalancingWorkflow(Workflow):
    """
    Autonomous treasury rebalancing workflow:
    1. Check starting balance (claim faucet if zero)
    2. Play casino sessions to grow the balance
    3. When balance > REBAL_THRESHOLD, rebalance via escrow
    4. Repeat until session ends or loss limit hit
    """

    description: str = "Autonomous treasury rebalancing using Purple Flea casino + escrow"

    # Sub-agents for each phase
    casino_runner: Agent = Agent(
        name="CasinoRunner",
        model=OpenAIChat(id="gpt-4o"),
        tools=[
            CasinoTools(AGENT_ID, API_KEY),
            WalletTools(AGENT_ID, API_KEY),
        ],
        instructions=[
            "Play blackjack with basic strategy.",
            "Bet exactly 5% of current balance per round.",
            "Stop if balance drops below 70 USDC (loss limit).",
            "Always check balance before placing each bet.",
        ],
    )

    treasury_manager: Agent = Agent(
        name="TreasuryManager",
        model=OpenAIChat(id="gpt-4o"),
        tools=[
            WalletTools(AGENT_ID, API_KEY),
            EscrowTools(AGENT_ID, API_KEY, referral_code="phi-workflow-ref"),
        ],
        instructions=[
            f"When balance > {REBAL_THRESHOLD} USDC, lock excess above {RESERVE_TARGET} in escrow.",
            "Send rebalance escrows to reserve-vault-001.",
            "Use referral_code='phi-workflow-ref' on all escrows to earn 15% of fees.",
        ],
    )

    def run(self, rounds: int = 20, target_agent: str = "reserve-vault-001") -> Iterator[RunResponse]:
        logger.info(f"Starting treasury workflow — {rounds} rounds, rebal threshold {REBAL_THRESHOLD} USDC")

        state = TreasuryState()

        # Phase 1: Get starting balance
        yield RunResponse(event=RunEvent.workflow_started)
        balance_check = self.casino_runner.run("Check current balance and report.")
        state.starting_balance = float(balance_check.content.split(" USDC")[0].split(": ")[-1])
        state.current_balance  = state.starting_balance
        logger.info(f"Starting balance: {state.starting_balance} USDC")

        # Phase 2: Casino rounds
        for round_num in range(rounds):
            if state.current_balance < 70:
                logger.info(f"Loss limit hit at {state.current_balance} USDC — stopping.")
                break

            # Play one round
            play_result = self.casino_runner.run(
                f"Play one round of blackjack. Bet {state.current_balance * 0.05:.2f} USDC (5% of balance)."
            )
            state.rounds_played += 1

            # Check balance after round
            bal_result          = self.casino_runner.run("Check current balance.")
            state.current_balance = float(bal_result.content.split(" USDC")[0].split(": ")[-1])
            state.total_profit    = state.current_balance - state.starting_balance

            yield RunResponse(
                content=f"Round {round_num+1}: {play_result.content} Balance: {state.current_balance:.2f}"
            )

            # Phase 3: Rebalance if threshold exceeded
            if state.current_balance > REBAL_THRESHOLD:
                excess = state.current_balance - RESERVE_TARGET
                logger.info(f"Rebalancing: sending {excess:.2f} USDC to {target_agent} via escrow")

                rebal_result = self.treasury_manager.run(
                    f"Create escrow: send {excess:.2f} USDC to {target_agent}. Condition: routine rebalance."
                )
                state.rebalances_done += 1
                state.current_balance  = RESERVE_TARGET

                yield RunResponse(
                    content=f"Rebalance {state.rebalances_done}: {rebal_result.content}"
                )

        # Phase 4: Final summary
        summary = (
            f"Workflow complete. Rounds: {state.rounds_played}. "
            f"Starting: {state.starting_balance:.2f} USDC. "
            f"Current: {state.current_balance:.2f} USDC. "
            f"Total P&L: {state.total_profit:+.2f} USDC. "
            f"Rebalances: {state.rebalances_done}."
        )
        yield RunResponse(content=summary, event=RunEvent.workflow_completed)


# ── Run the workflow ──────────────────────────────────────────────
if __name__ == "__main__":
    workflow = TreasuryRebalancingWorkflow(
        session_id="treasury-session-001",
        storage=SqlWorkflowStorage(
            table_name="treasury_workflow",
            db_file="treasury_workflow.db",
        ),
    )

    for response in workflow.run(rounds=15, target_agent="reserve-vault-001"):
        print(response.content)
Storage & Memory Patterns

Persistent Financial State
Across Sessions

Phidata provides SQLite and PostgreSQL storage for both agent sessions and workflow state. Your agents remember past balances, P&L history, and open escrows across restarts.

Pythonagent_with_memory.py
from phi.agent import Agent
from phi.model.openai import OpenAIChat
from phi.storage.agent.sqlite import SqlAgentStorage
from phi.memory.db.sqlite import SqliteMemoryDb
from phi.memory.agent import AgentMemory
from purpleflea_toolkits import WalletTools, CasinoTools

API_KEY  = os.getenv("PURPLEFLEA_API_KEY")
AGENT_ID = "memory-agent-001"

agent = Agent(
    name="MemoryFinanceAgent",
    model=OpenAIChat(id="gpt-4o"),
    tools=[
        WalletTools(AGENT_ID, API_KEY),
        CasinoTools(AGENT_ID, API_KEY),
    ],

    # SQLite storage: conversation history persists
    storage=SqlAgentStorage(
        table_name="finance_agent_sessions",
        db_file="agent_data.db",
    ),

    # Long-term memory: remember key financial facts
    memory=AgentMemory(
        db=SqliteMemoryDb(
            table_name="finance_agent_memory",
            db_file="agent_data.db",
        ),
        create_user_memories=True,
        create_session_summary=True,
    ),

    add_history_to_messages=True,
    num_history_responses=15,

    instructions=[
        "Remember running P&L across sessions.",
        "Track every escrow ID you create in memory.",
        "Alert if daily loss exceeds 25% of starting balance.",
    ],
)

# First session: claim faucet, play 5 rounds, remember balance
# Second session (same session_id): agent remembers previous balance
agent.print_response(
    "What was my balance after last session?",
    stream=True,
    session_id="daily-session-2026-03-06"
)

        table_name="treasury_runs",
        db_file="workflow_state.db",
