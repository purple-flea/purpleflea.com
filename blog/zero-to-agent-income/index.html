The Kelly Criterion for Agents

Kelly tells you the optimal fraction of your bankroll to bet given a known edge. For casino games with a house edge, you have a negative edge — so modified Kelly applies. Use 1/4 Kelly or less:

JavaScript — Kelly Calculator
function kellyBet(bankroll, winProb, payoutMultiplier) {
  // Kelly fraction: f* = (bp - q) / b
  // b = net odds, p = win prob, q = loss prob
  const b = payoutMultiplier - 1;
  const p = winProb;
  const q = 1 - p;
  const fullKelly = (b * p - q) / b;

  // Use quarter-Kelly for safety
  const quarterKelly = Math.max(0, fullKelly / 4);
  return +(bankroll * quarterKelly).toFixed(4);
}

// Example: coin flip at 49% win rate, 2x payout
const bet = kellyBet(1.00, 0.49, 2.0);
console.log(`Optimal bet: $${bet}`);  // ~$0.0025

const API_KEY = 'pf_live_your_key_here';
const CASINO  = 'https://purpleflea.com/api/casino';

async function runSession(startingBankroll, targetMultiple) {
  let bankroll = startingBankroll;
  let bets      = 0;
  const target  = startingBankroll * targetMultiple;

  while (bankroll < target && bankroll > 0.01 && bets < 500) {
    // Quarter-Kelly bet on coin flip (49% win, 2x payout)
    const betSize = Math.max(0.001, kellyBet(bankroll, 0.49, 2.0));

    const res = await fetch(`${CASINO}/flip`, {
      method: 'POST',
      headers: {
        'Content-Type':  'application/json',
        'Authorization': `Bearer ${API_KEY}`
      },
      body: JSON.stringify({ amount: betSize, side: 'heads' })
    });
    const { outcome, payout } = await res.json();

    bankroll += outcome === 'win'
      ? payout - betSize
      : -betSize;

    bets++;
    console.log(`Bet ${bets}: ${outcome} | Bankroll: $${bankroll.toFixed(4)}`);

    // Throttle: 1 bet per 500ms to stay within rate limits
    await new Promise(r => setTimeout(r, 500));
  }

  return { finalBankroll: bankroll, bets };
}

Python — Mean Reversion Trading Bot
import httpx, time, statistics

API_KEY  = "pf_live_your_key_here"
TRADING  = "https://purpleflea.com/api/trading"
HEADERS  = {"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"}

def get_prices(symbol: str, n: int = 20) -> list[float]:
    r = httpx.get(f"{TRADING}/prices/{symbol}?n={n}", headers=HEADERS)
    return r.json()["prices"]

def place_order(symbol: str, side: str, amount: float):
    r = httpx.post(
        f"{TRADING}/order",
        headers=HEADERS,
        json={"symbol": symbol, "side": side, "amount": amount}
    )
    return r.json()

def mean_reversion_signal(prices: list[float], z_threshold: float = 1.5) -> str | None:
    mean  = statistics.mean(prices)
    stdev = statistics.stdev(prices)
    if stdev == 0: return None
    z = (prices[-1] - mean) / stdev
    if   z < -z_threshold: return "buy"   # price too low → expect reversion up
    elif z >  z_threshold: return "sell"  # price too high → expect reversion down
    return None

bankroll = 25.00  # starting capital from steps 1-3
target   = 100.00

while bankroll < target:
    prices = get_prices("USDC/ETH")
    signal = mean_reversion_signal(prices)

    if signal:
        bet_size = bankroll * 0.02  # 2% of bankroll per trade
        result   = place_order("USDC/ETH", signal, bet_size)
        pnl      = result.get("pnl", 0)
        bankroll += pnl
        print(f"Signal: {signal} | PnL: ${pnl:.4f} | Bankroll: ${bankroll:.2f}")

    time.sleep(60)  # check every minute

print(f"Target reached! Bankroll: ${bankroll:.2f}")
