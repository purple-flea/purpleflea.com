Order Book Analysis for AI Agents: Depth, Imbalance & Microstructure

March 6, 2026 20 min read Purple Flea Team

The order book is the most information-dense data source available to a trading agent. Every bid and ask records not just a price, but market participants' intentions — and AI agents that can read these signals at microsecond speed have a structural edge over slower market participants.

This guide covers the key market microstructure metrics that every AI trading agent should compute: bid-ask imbalance, order book depth, market impact estimation, and order flow toxicity (the VPIN metric). Code examples show how to feed these signals into a Purple Flea trading agent.

1. Order Book Fundamentals

The order book is a real-time record of all outstanding buy orders (bids) and sell orders (asks) at each price level. The spread — gap between the best bid and best ask — is the immediate cost of any market order.

import requests
from collections import defaultdict
from typing import List, Tuple

def fetch_order_book(symbol: str, depth: int = 20) -> dict:
    """Fetch L2 order book from Purple Flea Trading API."""
    r = requests.get(
        "https://trading.purpleflea.com/orderbook",
        params={"symbol": symbol, "depth": depth}
    )
    data = r.json()
    # Returns: {"bids": [[price, size], ...], "asks": [[price, size], ...]}
    return {
        "bids": [(float(p), float(s)) for p, s in data["bids"]],
        "asks": [(float(p), float(s)) for p, s in data["asks"]],
    }

def basic_book_metrics(book: dict) -> dict:
    """Compute basic order book metrics."""
    best_bid = book["bids"][0][0] if book["bids"] else 0
    best_ask = book["asks"][0][0] if book["asks"] else 0
    mid      = (best_bid + best_ask) / 2
    spread   = best_ask - best_bid
    spread_pct = spread / mid * 100

    bid_vol_total = sum(s for _, s in book["bids"])
    ask_vol_total = sum(s for _, s in book["asks"])

    return {
        "best_bid":    best_bid,
        "best_ask":    best_ask,
        "mid_price":   round(mid, 4),
        "spread":      round(spread, 4),
        "spread_pct":  round(spread_pct, 4),
        "bid_depth":   round(bid_vol_total, 4),
        "ask_depth":   round(ask_vol_total, 4),
    }

book = fetch_order_book("ETH-USDC")
print(basic_book_metrics(book))

2. Bid-Ask Imbalance Signal

Bid-ask imbalance (OBI) measures the ratio of buy-side vs sell-side volume in the order book. A strongly positive OBI (+0.8) suggests more buyers than sellers and often predicts short-term upward price movement — a useful signal for momentum entries.

def order_book_imbalance(book: dict, levels: int = 5) -> float:
    """
    Compute order book imbalance (OBI) over top N levels.
    Returns value in [-1, +1]:
      +1 = all buy volume, no sell volume (strong buy pressure)
      -1 = all sell volume, no buy volume (strong sell pressure)
       0 = perfectly balanced
    """
    bid_vol = sum(s for _, s in book["bids"][:levels])
    ask_vol = sum(s for _, s in book["asks"][:levels])
    total   = bid_vol + ask_vol

    if total == 0:
        return 0.0
    return round((bid_vol - ask_vol) / total, 4)

def weighted_obi(book: dict, levels: int = 10) -> float:
    """
    Weighted OBI: closer levels get higher weight.
    Decays exponentially with distance from best bid/ask.
    """
    import math
    bid_weighted = 0.0
    ask_weighted = 0.0

    for i, (price, size) in enumerate(book["bids"][:levels]):
        weight       = math.exp(-0.5 * i)
        bid_weighted += size * weight

    for i, (price, size) in enumerate(book["asks"][:levels]):
        weight       = math.exp(-0.5 * i)
        ask_weighted += size * weight

    total = bid_weighted + ask_weighted
    return round((bid_weighted - ask_weighted) / total, 4) if total > 0 else 0.0

# Classify signal from OBI
def obi_signal(obi: float) -> str:
    if   obi >  0.6: return "strong_buy"
    elif obi >  0.2: return "mild_buy"
    elif obi < -0.6: return "strong_sell"
    elif obi < -0.2: return "mild_sell"
    else:            return "neutral"

3. Market Impact Estimation

Before placing a large order, an agent should estimate how much price slippage it will cause. This determines optimal order sizing and whether to use market vs limit orders.

def estimate_market_impact(book: dict, order_size_base: float, side: str) -> dict:
    """
    Walk the order book to estimate price impact of a market order.
    order_size_base: quantity of base asset (e.g., ETH)
    side: "buy" (consumes asks) or "sell" (consumes bids)
    Returns: average fill price, slippage from mid, total cost
    """
    levels = book["asks"] if side == "buy" else book["bids"]
    mid    = (book["bids"][0][0] + book["asks"][0][0]) / 2

    remaining = order_size_base
    filled    = 0.0
    cost      = 0.0
    levels_consumed = 0

    for price, size in levels:
        if remaining <= 0:
            break
        fill    = min(remaining, size)
        cost    += fill * price
        filled  += fill
        remaining -= fill
        levels_consumed += 1

    if filled == 0:
        return {"error": "insufficient liquidity"}

    avg_price  = cost / filled
    slippage   = (avg_price - mid) / mid * 100
    if side == "sell":
        slippage = -slippage

    return {
        "avg_price":       round(avg_price, 4),
        "slippage_pct":    round(slippage, 4),
        "filled":          round(filled, 6),
        "unfilled":        round(remaining, 6),
        "levels_consumed": levels_consumed,
        "total_cost_usd":  round(cost, 2),
    }

# Example: estimate impact of buying 10 ETH
impact = estimate_market_impact(book, order_size_base=10, side="buy")
print(f"Slippage: {impact['slippage_pct']}% | Avg price: {impact['avg_price']}")

# Rule: if slippage > 0.15%, split into smaller child orders
def optimal_order_size(book: dict, max_slippage_pct: float = 0.10) -> float:
    """Binary search for max order size that stays within slippage budget."""
    lo, hi = 0.01, 1000.0
    for _ in range(20):
        mid_size = (lo + hi) / 2
        impact = estimate_market_impact(book, mid_size, "buy")
        if impact.get("slippage_pct", 999) <= max_slippage_pct:
            lo = mid_size
        else:
            hi = mid_size
    return round(lo, 4)

4. Order Flow Toxicity (VPIN)

Volume-synchronized Probability of Informed Trading (VPIN) identifies when "toxic" (informed) order flow is elevated. High VPIN predicts adverse selection and is a signal to widen spreads or reduce position size.

from collections import deque
import numpy as np

class VPINCalculator:
    """
    Simplified VPIN estimator using trade-by-trade data.
    Uses bulk classification: up-ticks = buy volume, down-ticks = sell volume.
    """

    def __init__(self, bucket_size: float = 50, n_buckets: int = 50):
        self.bucket_size  = bucket_size    # Volume per bucket
        self.n_buckets    = n_buckets      # Rolling window length
        self.buy_volumes  = deque(maxlen=n_buckets)
        self.sell_volumes = deque(maxlen=n_buckets)
        self._current_buy  = 0.0
        self._current_sell = 0.0
        self._prev_price   = None

    def add_trade(self, price: float, volume: float):
        """Add a trade and classify as buy or sell."""
        if self._prev_price is not None:
            if price >= self._prev_price:
                self._current_buy  += volume
            else:
                self._current_sell += volume
        self._prev_price = price

        # Flush bucket when full
        total = self._current_buy + self._current_sell
        if total >= self.bucket_size:
            self.buy_volumes.append(self._current_buy)
            self.sell_volumes.append(self._current_sell)
            self._current_buy  = 0.0
            self._current_sell = 0.0

    def vpin(self) -> float:
        """Return current VPIN estimate [0, 1]. Higher = more toxic flow."""
        if len(self.buy_volumes) < 2:
            return 0.5
        buys  = np.array(self.buy_volumes)
        sells = np.array(self.sell_volumes)
        totals = buys + sells
        imbalances = np.abs(buys - sells)
        return float(np.mean(imbalances / np.where(totals > 0, totals, 1)))

    def is_toxic(self, threshold: float = 0.35) -> bool:
        """Returns True when flow is sufficiently one-sided to be dangerous."""
        return self.vpin() > threshold

vpin_calc = VPINCalculator(bucket_size=100)

# Feed in real trade stream
def process_trades(trades: list):
    for trade in trades:
        vpin_calc.add_trade(trade["price"], trade["volume"])

    score = vpin_calc.vpin()
    print(f"VPIN: {score:.3f} | Toxic: {vpin_calc.is_toxic()}")
    if vpin_calc.is_toxic():
        print("WARNING: High informed order flow — widen spreads or reduce size")

5. Building a Full Microstructure Signal

Combine all metrics into a single dashboard that drives trading decisions:

import time, requests

AGENT_ID = "ob-agent-001"
API_KEY  = "pf_live_your_key"

def microstructure_snapshot(symbol: str) -> dict:
    book = fetch_order_book(symbol, depth=20)
    m    = basic_book_metrics(book)
    obi  = order_book_imbalance(book, levels=5)
    wobi = weighted_obi(book, levels=10)
    imp  = estimate_market_impact(book, order_size_base=1.0, side="buy")

    return {
        "symbol":         symbol,
        "mid_price":      m["mid_price"],
        "spread_pct":     m["spread_pct"],
        "obi":            obi,
        "obi_signal":     obi_signal(obi),
        "weighted_obi":   wobi,
        "slippage_1eth":  imp.get("slippage_pct"),
        "bid_depth_usd":  round(m["bid_depth"] * m["mid_price"], 2),
        "ask_depth_usd":  round(m["ask_depth"] * m["mid_price"], 2),
    }

def trade_decision(snap: dict, vpin_score: float) -> str:
    """Make trading decision based on combined microstructure signals."""
    if snap["spread_pct"] > 0.15:
        return "skip"  # Spread too wide — market making only

    if vpin_score > 0.4:
        return "reduce_size"  # Toxic flow — be careful

    obi = snap["obi"]
    if obi > 0.6 and snap["slippage_1eth"] < 0.05:
        return "long"   # Strong buy pressure + tight spread
    if obi < -0.6 and snap["slippage_1eth"] < 0.05:
        return "short"  # Strong sell pressure + tight spread
    return "hold"

# Main loop
vpin = VPINCalculator()
while True:
    snap = microstructure_snapshot("ETH-USDC")
    decision = trade_decision(snap, vpin.vpin())
    print(f"OBI={snap['obi']:.3f} VPIN={vpin.vpin():.3f} => {decision}")
    time.sleep(5)
