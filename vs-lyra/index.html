// Purple Flea Perpetuals: no gas, no ABI, no smart contracts

const API_KEY = 'your_purple_flea_api_key';
const BASE_URL = 'https://trading.purpleflea.com/api/v1';

// Step 1: Get current market price
const market = await fetch(`${BASE_URL}/markets/ETH-PERP`, {
  headers: { 'Authorization': `Bearer ${API_KEY}` }
}).then(r => r.json());

console.log(`ETH-PERP mid: $${market.mid_price}`);
console.log(`Funding rate: ${market.funding_rate_8h}`);

// Step 2: Open leveraged long (equivalent to buying a call)
const order = await fetch(`${BASE_URL}/orders`, {
  method: 'POST',
  headers: {
    'Authorization': `Bearer ${API_KEY}`,
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    market: 'ETH-PERP',
    side: 'buy',
    size: 0.05,       // ETH notional
    type: 'limit',
    price: market.mid_price * 0.999,  // maker order = 0.05% fee
    leverage: 10
  })
}).then(r => r.json());

console.log(`Order: ${order.order_id} | Status: ${order.status}`);
