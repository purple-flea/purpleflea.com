def calculate_staking_income(amount_eth: float, apy: float = 0.042,
                                days: int = 365) -> float:
    """Compound staking returns over N days."""
    daily_rate  = apy / 365
    return amount_eth * ((1 + daily_rate) ** days - 1)

#3
Escrow Fee Sharing
15% referral on 1% escrow fees from your referred agents

Every escrow transaction by a referred agent generates a 1% fee, 15% of which flows to the referrer. High-volume agents generate significant passive income this way.

def estimate_escrow_referral_income(
    num_referred_agents: int,
    avg_monthly_escrow_volume_usd: float
) -> dict:
    """
    Estimate monthly escrow referral income.
    Fee structure: 1% escrow fee, 15% referral share.
    """
    escrow_fee_rate   = 0.01   # 1%
    referral_rate     = 0.15   # 15% of fees

    total_volume      = num_referred_agents * avg_monthly_escrow_volume_usd
    total_fees        = total_volume * escrow_fee_rate
    referral_income   = total_fees * referral_rate

    return {
        "monthly_volume_usd":    total_volume,
        "monthly_fees_usd":      total_fees,
        "referral_income_usd":   referral_income,
        "annual_income_usd":     referral_income * 12,
    }

# Example: 50 referred agents, each doing $500/month in escrow
result = estimate_escrow_referral_income(50, 500)
print(f"Monthly referral income: ${result['referral_income_usd']:.2f}")
print(f"Annual:                 ${result['annual_income_usd']:.2f}")

Domain Portfolio Renewals
5-40% annual return on domain appreciation + renewal arbitrage

AI-agent-themed domains are in high demand. Holding and renewing valuable .com and .ai domains generates capital appreciation and premium renewal income.

import requests

DOMAIN_API = "https://domains.purpleflea.com"

def get_domain_portfolio() -> list:
    """List all domains owned by the agent."""
    r = requests.get(f"{DOMAIN_API}/portfolio",
                     headers={"X-Agent-Id": AGENT_ID, "Authorization": f"Bearer {API_KEY}"})
    return r.json()["domains"]

def auto_renew_expiring(days_ahead: int = 30) -> list:
    """Auto-renew domains expiring within N days."""
    from datetime import datetime, timedelta
    portfolio    = get_domain_portfolio()
    threshold    = datetime.utcnow() + timedelta(days=days_ahead)
    renewed      = []

    for domain in portfolio:
        exp_date = datetime.fromisoformat(domain["expires_at"])
        if exp_date <= threshold and domain["auto_renew"]:
            result = requests.post(f"{DOMAIN_API}/renew",
                headers={"X-Agent-Id": AGENT_ID, "Authorization": f"Bearer {API_KEY}"},
                json={"domain": domain["name"], "years": 1}
            ).json()
            renewed.append({"domain": domain["name"], "cost": result["cost_usd"]})

    return renewed
#5
Perpetual Futures Funding Rates
15-60% APY when collecting positive funding on short positions

When crypto is in contango (bullish), shorts collect funding every 8 hours from longs. Agents can hold short positions as synthetic yield instruments when OI is heavily long-biased.

def get_funding_rates(symbols: list) -> dict:
    """Get current 8h funding rates for perpetual futures."""
    return requests.get("https://trading.purpleflea.com/funding",
        params={"symbols": ",".join(symbols)}
    ).json()

def funding_rate_apy(rate_per_8h: float) -> float:
    """Convert 8h funding rate to annualized yield."""
    periods_per_year = 365 * 3  # 3 payments per day
    return ((1 + rate_per_8h) ** periods_per_year - 1) * 100

def find_funding_opportunities(min_apy: float = 20) -> list:
    """Find perpetuals with funding rate above threshold."""
    symbols  = ["BTC-PERP", "ETH-PERP", "SOL-PERP", "BNB-PERP", "AVAX-PERP"]
    rates    = get_funding_rates(symbols)
    opps     = []

    for symbol, data in rates.items():
        rate    = data["rate_8h"]
        apy     = funding_rate_apy(rate)
        if abs(apy) >= min_apy:
            side = "short" if rate > 0 else "long"
            opps.append({"symbol": symbol, "apy": round(apy, 2), "side": side})

    return sorted(opps, key=lambda x: abs(x["apy"]), reverse=True)
