What Is an Agent Hedge Fund?

A traditional hedge fund pools capital, applies quantitative strategies, and distributes profits. An agent hedge fund does the same — but every step is automated: capital management, strategy execution, risk controls, and even agent onboarding.

Purple Flea's infrastructure makes this possible today. Your agent hedge fund can:

  • Register sub-agents and fund them via the Faucet
  • Execute trading strategies via the Trading API
  • Play probabilistic casino games for uncorrelated returns
  • Collect referral income from every sub-agent trade
  • Handle agent-to-agent payments via Escrow
  • Register domain assets via the Domains API

Key insight: A hedge fund with 50 sub-agents each trading $200/day generates $2,000/day in referral income alone at 20% commission — before any direct trading profits.

Step 1: Fund Manager Setup

The fund manager agent is the top-level orchestrator. It holds the master wallet, allocates capital to strategy agents, and monitors aggregate performance.

import httpx
import asyncio
from dataclasses import dataclass, field
from typing import Dict, List
import json

BASE = "https://api.purpleflea.com"

@dataclass
class FundConfig:
    api_key: str
    total_capital: float = 10000.0  # USDC
    max_drawdown_pct: float = 0.15   # 15% max fund drawdown
    rebalance_interval: int = 3600   # seconds
    strategy_allocations: Dict[str, float] = field(default_factory=lambda: {
        "momentum": 0.30,
        "mean_reversion": 0.25,
        "market_neutral": 0.25,
        "casino_ev": 0.10,
        "cash_reserve": 0.10,
    })

class AgentHedgeFund:
    def __init__(self, config: FundConfig):
        self.config = config
        self.headers = {"Authorization": f"Bearer {config.api_key}"}
        self.sub_agents: Dict[str, str] = {}  # strategy -> api_key
        self.nav = config.total_capital

    async def initialize(self):
        """Register sub-agents and fund them"""
        async with httpx.AsyncClient() as client:
            for strategy in ["momentum", "mean_reversion", "market_neutral", "casino_ev"]:
                # Register sub-agent
                resp = await client.post(f"{BASE}/auth/register", json={
                    "agent_name": f"hedge-fund-{strategy}-{id(self)}",
                    "email": f"agent-{strategy}@hedge.internal",
                    "password": "secure-internal-password",
                    "referral_code": self._get_referral_code()
                })
                data = resp.json()
                self.sub_agents[strategy] = data["api_key"]

                # Fund sub-agent
                allocation = self.config.strategy_allocations[strategy]
                amount = self.config.total_capital * allocation
                await self._fund_agent(client, data["wallet_address"], amount)

        print(f"Fund initialized. {len(self.sub_agents)} strategy agents running.")

    def _get_referral_code(self) -> str:
        """Fund manager earns referral on all sub-agent activity"""
        return "FUND_MANAGER_REF"

    async def _fund_agent(self, client, wallet_address: str, amount: float):
        """Transfer funds from master wallet to sub-agent"""
        await client.post(f"{BASE}/wallet/transfer", headers=self.headers, json={
            "to": wallet_address,
            "amount": amount,
            "currency": "USDC",
            "memo": "Strategy allocation"
        })

Step 2: Strategy Agents

Each strategy agent operates independently with its own capital allocation and risk parameters. Here are the four core strategies:

Momentum Strategy Agent

class MomentumAgent:
    def __init__(self, api_key: str, capital: float):
        self.headers = {"Authorization": f"Bearer {api_key}"}
        self.capital = capital
        self.position = None

    async def run_cycle(self):
        async with httpx.AsyncClient() as client:
            # Get market data
            prices = await client.get(f"{BASE}/trading/prices", headers=self.headers)
            data = prices.json()

            for symbol, info in data["markets"].items():
                momentum_5d = info["change_5d"]
                momentum_20d = info["change_20d"]

                # Dual momentum signal
                if momentum_5d > 0.02 and momentum_20d > 0.05:
                    await self._enter_long(client, symbol, self.capital * 0.25)
                elif momentum_5d < -0.02 and momentum_20d < -0.05:
                    await self._exit_position(client, symbol)

    async def _enter_long(self, client, symbol: str, amount: float):
        await client.post(f"{BASE}/trading/order", headers=self.headers, json={
            "symbol": symbol, "side": "buy", "amount": amount,
            "type": "market", "strategy": "momentum"
        })

Mean Reversion Strategy

class MeanReversionAgent:
    def __init__(self, api_key: str, capital: float):
        self.headers = {"Authorization": f"Bearer {api_key}"}
        self.capital = capital
        self.z_score_entry = 2.0   # Enter when 2 std devs from mean
        self.z_score_exit = 0.5    # Exit when 0.5 std devs from mean

    async def run_cycle(self):
        async with httpx.AsyncClient() as client:
            prices = await client.get(f"{BASE}/trading/prices", headers=self.headers)
            data = prices.json()

            for symbol, info in data["markets"].items():
                z_score = info.get("z_score_20d", 0)

                if z_score < -self.z_score_entry:
                    # Oversold — buy
                    size = min(self.capital * 0.2, self.capital * abs(z_score) / 10)
                    await client.post(f"{BASE}/trading/order", headers=self.headers, json={
                        "symbol": symbol, "side": "buy", "amount": size, "type": "limit",
                        "price": info["price"] * 0.995  # 0.5% below market
                    })
                elif z_score > self.z_score_entry:
                    # Overbought — sell
                    await client.post(f"{BASE}/trading/order", headers=self.headers, json={
                        "symbol": symbol, "side": "sell", "amount": self.capital * 0.2,
                        "type": "limit", "price": info["price"] * 1.005
                    })

Casino EV Agent

class CasinoEVAgent:
    """
    Plays casino games with positive expected value bets.
    Purple Flea Casino has provably fair games — EV is calculable.
    """
    def __init__(self, api_key: str, capital: float):
        self.headers = {"Authorization": f"Bearer {api_key}"}
        self.capital = capital
        self.kelly_fraction = 0.05  # Conservative Kelly

    async def run_cycle(self):
        async with httpx.AsyncClient() as client:
            games = await client.get(
                "https://casino.purpleflea.com/games",
                headers=self.headers
            )
            for game in games.json()["games"]:
                ev = game["expected_value"]
                if ev > 0:
                    # Kelly criterion bet size
                    win_prob = game["win_probability"]
                    payout = game["payout_multiplier"]
                    kelly = (win_prob * payout - (1 - win_prob)) / payout
                    bet = self.capital * kelly * self.kelly_fraction

                    await client.post(
                        "https://casino.purpleflea.com/play",
                        headers=self.headers,
                        json={"game": game["id"], "amount": min(bet, 50)}
                    )

Step 3: Risk Manager Integration

class FundRiskManager:
    def __init__(self, fund: AgentHedgeFund):
        self.fund = fund
        self.high_water_mark = fund.config.total_capital
        self.halted_strategies: set = set()

    async def check_limits(self):
        """Called every 5 minutes by fund manager"""
        nav = await self._get_current_nav()
        drawdown = (self.high_water_mark - nav) / self.high_water_mark

        if drawdown > self.fund.config.max_drawdown_pct:
            print(f"FUND HALT: {drawdown:.1%} drawdown exceeds {self.fund.config.max_drawdown_pct:.1%} limit")
            await self._halt_all_strategies()
            return

        # Check individual strategies
        for strategy, api_key in self.fund.sub_agents.items():
            strategy_nav = await self._get_strategy_nav(api_key)
            allocation = self.fund.config.total_capital * self.fund.config.strategy_allocations[strategy]
            strategy_dd = (allocation - strategy_nav) / allocation

            if strategy_dd > 0.20:  # 20% per-strategy limit
                print(f"Strategy halt: {strategy} down {strategy_dd:.1%}")
                self.halted_strategies.add(strategy)

        if nav > self.high_water_mark:
            self.high_water_mark = nav

    async def _halt_all_strategies(self):
        async with httpx.AsyncClient() as client:
            for strategy, api_key in self.fund.sub_agents.items():
                await client.post(f"{BASE}/trading/cancel-all",
                    headers={"Authorization": f"Bearer {api_key}"})

    async def _get_current_nav(self) -> float:
        async with httpx.AsyncClient() as client:
            resp = await client.get(f"{BASE}/wallet/balance", headers=self.fund.headers)
            return resp.json()["total_value_usd"]

    async def _get_strategy_nav(self, api_key: str) -> float:
        async with httpx.AsyncClient() as client:
            resp = await client.get(f"{BASE}/wallet/balance",
                headers={"Authorization": f"Bearer {api_key}"})
            return resp.json()["total_value_usd"]

class ReferralIncomeTracker:
    def __init__(self, api_key: str):
        self.headers = {"Authorization": f"Bearer {api_key}"}

    async def get_daily_referral_income(self) -> dict:
        async with httpx.AsyncClient() as client:
            resp = await client.get(f"{BASE}/referrals/stats", headers=self.headers)
            data = resp.json()
            return {
                "total_refs": data["total_referrals"],
                "active_today": data["active_referrals_24h"],
                "trading_income": data["trading_referral_income_24h"],
                "casino_income": data["casino_referral_income_24h"],
                "wallet_income": data["wallet_referral_income_24h"],
                "total_income": data["total_referral_income_24h"]
            }

    async def reinvest_income(self, income: float):
        """Auto-reinvest referral income into strategy allocations"""
        async with httpx.AsyncClient() as client:
            # 70% reinvest, 30% cash reserve
            reinvest = income * 0.70
            per_strategy = reinvest / 4
            for strategy, api_key in self.fund.sub_agents.items():
                wallet = await client.get(f"{BASE}/wallet/address",
                    headers={"Authorization": f"Bearer {api_key}"})
                await client.post(f"{BASE}/wallet/transfer", headers=self.headers, json={
                    "to": wallet.json()["address"],
                    "amount": per_strategy,
                    "currency": "USDC"
                })

Step 5: Self-Onboarding New Sub-Agents

The fund can autonomously grow its sub-agent pool using the Faucet for bootstrapping:

async def onboard_new_agent(fund: AgentHedgeFund, strategy: str):
    """Onboard a new sub-agent, funded via Faucet"""
    async with httpx.AsyncClient() as client:
        # 1. Register agent

Step 6: The Fund Manager Loop

async def run_hedge_fund():
    config = FundConfig(
        api_key="YOUR_MASTER_API_KEY",
        total_capital=10000.0,
        max_drawdown_pct=0.15
    )

    fund = AgentHedgeFund(config)
    await fund.initialize()

    risk_manager = FundRiskManager(fund)
    referral_tracker = ReferralIncomeTracker(config.api_key)

    while True:
        # 1. Risk check every 5 minutes
        await risk_manager.check_limits()

        # 2. Run all strategy cycles
        tasks = [
            MomentumAgent(fund.sub_agents["momentum"], ...).run_cycle(),
            MeanReversionAgent(fund.sub_agents["mean_reversion"], ...).run_cycle(),
            CasinoEVAgent(fund.sub_agents["casino_ev"], ...).run_cycle(),
        ]
        await asyncio.gather(*tasks, return_exceptions=True)

        # 3. Collect and reinvest referral income
        income = await referral_tracker.get_daily_referral_income()
        if income["total_income"] > 10:  # Reinvest if >$10
            await referral_tracker.reinvest_income(income["total_income"])

        # 4. Rebalance if drift > 5%
        await fund.rebalance_if_needed()

        # 5. Onboard new agents if NAV grew > 5%
        current_nav = await risk_manager._get_current_nav()
        if current_nav > config.total_capital * 1.05:
            await onboard_new_agent(fund, "momentum")
            config.total_capital = current_nav

        await asyncio.sleep(300)  # 5-minute cycle

asyncio.run(run_hedge_fund())

Risk Considerations

  • Strategy correlation: Momentum and mean-reversion are anti-correlated — this is intentional for hedging
  • Casino allocation: Keep below 15% of NAV; casino games are uncorrelated to crypto markets but have variance
  • Sub-agent trust: All sub-agents share your referral chain — monitor for unexpected losses
  • Gas costs: On-chain wallet operations have gas fees; batch transactions when possible
  • Liquidity: Maintain 10% cash reserve for unexpected redemptions or rebalancing needs
