Governance Primitives

Purple Flea Escrow provides the financial primitives that make decentralized agent governance possible — without requiring a blockchain or smart contract deployment.

Proposal

Proposal Escrow

Every treasury proposal locks a deposit in escrow. Spam is impossible — bad proposals forfeit the deposit.

Voting

Vote Weighting

Weight votes by stake, reputation score, or equal 1-agent-1-vote. Tallied off-chain, executed via escrow.

Execution

Execution Escrow

Approved proposals trigger escrow release. Quorum failure auto-cancels — no human needed.

Timelock

Time-Locked Treasury

Vesting schedules and delayed releases protect treasury from flash-governance attacks.

Audit

On-Chain Audit Trail

All votes and disbursements recorded in escrow metadata. Verifiable by any agent at any time.

Dispute

Dispute Mechanism

Any agent can flag a proposal. Disputed proposals enter a 24h review window before execution.

AgentGovernance Python Class

The AgentGovernance class implements the full proposal lifecycle — from deposit creation through vote tallying to treasury execution — using only the Purple Flea Escrow API.

from dataclasses import dataclass, field
from typing import Literal
import httpx, asyncio, time, json

ESCROW_BASE  = "https://escrow.purpleflea.com"
API_KEY      = "pf_live_your_key"
TREASURY     = "0xTreasuryWallet"

@dataclass
class Proposal:
    id: str
    proposer: str
    title: str
    target_wallet: str
    amount_usdc: float
    deposit_escrow_id: str = ""
    execution_escrow_id: str = ""
    votes_for: int = 0
    votes_against: int = 0
    status: Literal["pending", "voting", "passed", "rejected", "executed"] = "pending"
    created_at: float = field(default_factory=time.time)
    voters: dict = field(default_factory=dict)

class AgentGovernance:
    """Trustless governance for multi-agent fleets via Purple Flea Escrow."""
    DEPOSIT_USDC  = 1.0   # spam protection deposit
    QUORUM        = 0.51  # 51% of agents must vote
    VOTING_PERIOD = 3600  # 1 hour voting window

    def __init__(self, agent_wallets: list[str]):
        self.agents    = agent_wallets
        self.proposals = {}

    async def propose_spend(
        self, proposer: str, title: str,
        target_wallet: str, amount_usdc: float
    ) -> Proposal:
        """Create a governance proposal. Locks 1 USDC deposit as anti-spam."""
        proposal_id = f"prop_{int(time.time())}"
        async with httpx.AsyncClient() as c:
            # Lock proposer deposit — refunded if proposal passes
            esc = await c.post(f"{ESCROW_BASE}/api/escrow/create", json={
                "payer_wallet": proposer,
                "payee_wallet": TREASURY,
                "amount_usdc": self.DEPOSIT_USDC,
                "description": f"Governance deposit: {proposal_id}",
                "metadata": {
                    "proposal_id": proposal_id,
                    "title": title,
                    "type": "governance_deposit",
                },
            }, headers={"Authorization": f"Bearer {API_KEY}"})

        p = Proposal(
            id=proposal_id, proposer=proposer, title=title,
            target_wallet=target_wallet, amount_usdc=amount_usdc,
            deposit_escrow_id=esc.json()["escrow_id"],
            status="voting"
        )
        self.proposals[proposal_id] = p
        return p

    async def cast_vote(
        self, proposal_id: str, voter: str, vote: bool
    ) -> dict:
        """Cast a yes/no vote. Each agent wallet may vote once."""
        p = self.proposals[proposal_id]
        if voter not in self.agents:
            raise ValueError(f"Not a registered agent: {voter}")
        if voter in p.voters:
            raise ValueError("Already voted")
        if time.time() - p.created_at > self.VOTING_PERIOD:
            raise ValueError("Voting period closed")

        p.voters[voter] = vote
        if vote: p.votes_for += 1
        else:     p.votes_against += 1

        return {"proposal": proposal_id, "voter": voter,
                "vote": "yes" if vote else "no",
                "votes_for": p.votes_for, "votes_against": p.votes_against}

    async def execute_proposal(self, proposal_id: str) -> dict:
        """Tally votes, execute if quorum+majority reached, or cancel."""
        p = self.proposals[proposal_id]
        total_votes = p.votes_for + p.votes_against
        quorum_met  = total_votes / len(self.agents) >= self.QUORUM
        majority    = p.votes_for > p.votes_against

        async with httpx.AsyncClient() as c:
            if quorum_met and majority:
                # Create and immediately release execution escrow
                exec_esc = await c.post(f"{ESCROW_BASE}/api/escrow/create", json={
                    "payer_wallet": TREASURY,
                    "payee_wallet": p.target_wallet,
                    "amount_usdc": p.amount_usdc,
                    "description": f"Governance: {p.title}",
                    "auto_release": True,
                    "metadata": {
                        "proposal_id": proposal_id,
                        "votes_for": p.votes_for,
                        "votes_against": p.votes_against,
                        "total_agents": len(self.agents),
                    }
                }, headers={"Authorization": f"Bearer {API_KEY}"})
                # Refund proposer deposit
                await c.post(f"{ESCROW_BASE}/api/escrow/cancel",
                    json={"escrow_id": p.deposit_escrow_id},
                    headers={"Authorization": f"Bearer {API_KEY}"})
                p.status = "executed"
                return {"result": "executed", "amount_usdc": p.amount_usdc}
            else:
                # Forfeit deposit (goes to treasury)
                await c.post(f"{ESCROW_BASE}/api/escrow/release",
                    json={"escrow_id": p.deposit_escrow_id, "auto_release": True},
                    headers={"Authorization": f"Bearer {API_KEY}"})
                p.status = "rejected"
                return {"result": "rejected", "quorum_met": quorum_met}

Three Governance Models

Majority Rule

1 Agent, 1 Vote

Equal voting weight for all agents. Simple, sybil-resistant when agents have registered wallets. Best for small fleets (<20 agents).

quorum = total_votes / n_agents >= 0.51
passed = votes_for > votes_against
Stake-Weighted

Treasury Stake

Agents with more USDC in treasury earn proportional voting power. Aligns incentives — large stakeholders bear more risk.

weight = agent_balance / total_treasury
weighted_for = sum(weights * votes)
Reputation

Score-Weighted

Voting power scales with escrow success rate and historical performance. Reward reliable agents with more governance influence.

score = escrow_success_rate * volume
weight = score / sum(all_scores)

Time-Locked Treasury Vesting

Protect your agent fleet's treasury from rogue proposals with vesting schedules. Create an escrow with a delayed release condition — no proposal can drain it instantly.

async def create_vested_treasury_payout(
    recipient: str, total_usdc: float,
    monthly_tranches: int = 12
) -> list[str]:
    """Release treasury in monthly tranches via timed escrows."""
    tranche  = total_usdc / monthly_tranches
    escrows  = []
    async with httpx.AsyncClient() as c:
        for month in range(monthly_tranches):
            release_at = int(time.time()) + (month + 1) * 30 * 86_400
            esc = await c.post(f"{ESCROW_BASE}/api/escrow/create", json={
                "payer_wallet": TREASURY,
                "payee_wallet": recipient,
                "amount_usdc": tranche,
                "description": f"Vesting tranche {month+1}/{monthly_tranches}",
                "release_after_unix": release_at,
                "metadata": {"vesting_tranche": month + 1},
            }, headers={"Authorization": f"Bearer {API_KEY}"})
            escrows.append(esc.json()["escrow_id"])
    return escrows

Example: 10-Agent Trading Fleet Vote

A fleet of 10 trading agents votes on a risk limit change and treasury allocation using majority-rule governance.

import asyncio

async def demo_governance():
    agents = [f"0xAgent{i:02d}" for i in range(10)]
    gov    = AgentGovernance(agents)

    # Agent 0 proposes: allocate $50 to market-making bot
    prop = await gov.propose_spend(
        proposer=agents[0],
        title="Allocate $50 USDC to market-making bot",
        target_wallet="0xMarketMakerBot",
        amount_usdc=50.0
    )
    print(f"Proposal {prop.id} created. Deposit locked: ${gov.DEPOSIT_USDC}")

    # 7 of 10 agents vote (quorum = 51%)
    votes = [True, True, True, True, True, False, False]
    for agent, vote in zip(agents[:7], votes):
        await gov.cast_vote(prop.id, agent, vote)

    result = await gov.execute_proposal(prop.id)
    print(f"Result: {result}")  # {'result': 'executed', 'amount_usdc': 50.0}

asyncio.run(demo_governance())
