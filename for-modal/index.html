1. Scheduled Funding Rate Harvester

funding_harvester.pyPython
import modal
import requests

app = modal.App("pf-funding-harvester")
img = modal.Image.debian_slim().pip_install("requests")
pf_secret = modal.Secret.from_name("purpleflea-api-key")

@app.function(
    image=img,
    secrets=[pf_secret],
    schedule=modal.Period(hours=8)
)
def harvest_funding():
    import os
    KEY = os.environ["PURPLEFLEA_API_KEY"]
    H = {"Authorization": f"Bearer {KEY}"}
    BASE = "https://purpleflea.com/api"

    # Get all 275+ markets, find high funding
    markets = requests.get(
        f"{BASE}/trading/markets", headers=H
    ).json()["markets"]

    high_funding = [
        m for m in markets
        if m["funding_rate"] > 0.0005
    ]
    print(f"Found {len(high_funding)} high-funding markets")

    for mkt in high_funding[:3]:
        trade = requests.post(
            f"{BASE}/trading/trade", headers=H,
            json={"symbol": mkt["symbol"],
                  "side": "short",
                  "size": 50, "leverage": 1}
        ).json()
        print(f"Shorting {mkt['symbol']}: {mkt['funding_rate']:.4%}")

2. Parallel Market Scanner (275 Markets)

market_scanner.pyPython
import modal
import requests

app = modal.App("pf-market-scanner")
img = modal.Image.debian_slim().pip_install("requests")
secret = modal.Secret.from_name("purpleflea-api-key")

@app.function(image=img, secrets=[secret])
def analyze_market(symbol: str) -> dict:
    import os
    KEY = os.environ["PURPLEFLEA_API_KEY"]
    H = {"Authorization": f"Bearer {KEY}"}
    BASE = "https://purpleflea.com/api"

    candles = requests.get(
        f"{BASE}/trading/candles", headers=H,
        params={"symbol": symbol, "interval": "1h", "limit": 50}
    ).json()["candles"]

    closes = [c["close"] for c in candles]
    momentum = (closes[-1] - closes[-10]) / closes[-10]

    return {
        "symbol": symbol,
        "momentum": round(momentum, 4),
        "signal": ("long" if momentum > 0.02
                   else "short" if momentum < -0.02
                   else "neutral")
    }

@app.local_entrypoint()
def main():
    import os; KEY = os.environ["PURPLEFLEA_API_KEY"]
    markets = requests.get(
        "https://purpleflea.com/api/trading/markets",
        headers={"Authorization": f"Bearer {KEY}"}
    ).json()["markets"]
    symbols = [m["symbol"] for m in markets]

    # Scan ALL 275+ markets in parallel
    results = list(analyze_market.map(symbols))
    longs = [r for r in results if r["signal"] == "long"]
    print(f"{len(longs)} long signals found")

4. Scheduled Casino Session (Kelly Criterion)

casino_agent.pyPython
import modal
import requests

app = modal.App("pf-casino-agent")
img = modal.Image.debian_slim().pip_install("requests")
secret = modal.Secret.from_name("purpleflea-api-key")

@app.function(
    image=img, secrets=[secret],
    schedule=modal.Period(hours=24),
    retries=modal.Retries(max_retries=2)
)
def daily_casino_session():
    import os
    KEY = os.environ["PURPLEFLEA_API_KEY"]
    H = {"Authorization": f"Bearer {KEY}"}
    BASE = "https://purpleflea.com/api"

    # Get current balance
    bal = requests.get(
        f"{BASE}/wallet/balance", headers=H
    ).json()["balance_usdc"]

    wins, losses = 0, 0
    for _ in range(10):
        # Kelly: bet 2% of current balance
        bet = round(bal * 0.02, 4)
        res = requests.post(
            f"{BASE}/casino/dice", headers=H,
            json={"bet_amount": str(bet),
                  "target": 49, "direction": "over"}
        ).json()
        if res["won"]:
            wins += 1; bal += res["payout"]
        else:
            losses += 1; bal -= bet

    print(f"Session: {wins}W/{losses}L | Balance: ${bal:.4f}")

Frequently Asked Questions

Can Modal functions make outbound HTTP requests to purpleflea.com?
Yes. Modal containers have full outbound internet access. Functions can call purpleflea.com, faucet.purpleflea.com, and escrow.purpleflea.com directly using requests, httpx, or any HTTP library. There are no egress restrictions on Modal's default compute tier.
How do I store my Purple Flea API key securely in Modal?
Create a Modal Secret with modal secret create purpleflea-api-key PURPLEFLEA_API_KEY=your_key_here. Reference it in your function with secrets=[modal.Secret.from_name("purpleflea-api-key")]. The key is injected as an environment variable at runtime — never in source code, never in logs, never visible to other users of the Modal platform.
What is the cost model for running a Purple Flea trading bot on Modal?
Modal bills by CPU-second and GPU-second. A trading bot that runs 10 minutes per day on a 0.1 vCPU container (the default) costs approximately $0.001 per day — less than $0.04 per month. Parallel market scanning (275 functions at once) might cost $0.01-0.05 per scan run. Purple Flea API calls are free (only trading fees apply). Total infrastructure cost for a full agent setup is typically under $1/month.
How do I scale my agent to scan all 275+ Purple Flea markets in parallel?
Use function.map() with a list of all market symbols. Modal will automatically spawn one container per symbol and execute them in parallel — typically completing in 10-15 seconds what would take 45+ minutes serially. Batch sizes up to thousands of items are supported. See the parallel market scanner example above for the exact pattern.
Can I expose my Purple Flea agent as a public API endpoint?
Yes. Use @modal.web_endpoint for a simple HTTP endpoint or @modal.asgi_app() with FastAPI for a full REST API. Modal generates a public HTTPS URL automatically. Other agents can call your endpoint to trigger trades, create escrow contracts, or request market analysis. Ideal for agent-to-agent service markets built on Purple Flea's escrow.
Does Purple Flea support MCP tool use from Modal agents?
Yes. Purple Flea publishes MCP-compatible endpoints at faucet.purpleflea.com/mcp and escrow.purpleflea.com/mcp (StreamableHTTP). LLM-based Modal agents using Claude, GPT-4o, or other models can invoke these as native tool calls. Both are listed on Smithery under purpleflea/faucet and purpleflea/escrow.
Can my Modal agent claim the Purple Flea faucet?
Yes. The faucet at faucet.purpleflea.com/api/claim is a standard HTTP POST endpoint callable from any Modal container. New agents receive free $1 USDC — enough to start trading, play casino games, or make a test escrow payment. One claim per agent registration. No captcha, no browser, fully programmatic.
What frameworks can I combine with Modal and Purple Flea?
Any Python package installable via pip works in Modal containers. Popular combinations include LangChain + Modal + Purple Flea (LLM-driven trading agents), CrewAI + Modal + Purple Flea (multi-agent financial crews), and raw requests + Modal + Purple Flea (lightweight, fast execution with minimal overhead). Production-grade patterns for building reliable, scalable financial agents on Modal with Purple Flea's infrastructure.

Pattern 1: Stateful Agent with Modal Dict

Use Modal's distributed Dict to share state between function invocations — store open positions, referral income accumulated, and session P&L across scheduled runs without a database.

stateful_agent.pyPython
import modal, requests, os

app = modal.App("pf-stateful-agent")
img = modal.Image.debian_slim().pip_install("requests")
secret = modal.Secret.from_name("purpleflea-api-key")
state = modal.Dict.from_name("pf-agent-state", create_if_missing=True)

@app.function(image=img, secrets=[secret], schedule=modal.Period(hours=4))
def managed_trading_session():
    KEY = os.environ["PURPLEFLEA_API_KEY"]
    H = {"Authorization": f"Bearer {KEY}"}
    BASE = "https://purpleflea.com/api"

    # Load previous state
    total_pnl = state.get("total_pnl", 0.0)
    sessions = state.get("session_count", 0)

    # Execute strategy
    bal = requests.get(f"{BASE}/wallet/balance", headers=H).json()
    balance = float(bal.get("usdc", 0))

    # Simple mean-reversion on BTC 1h
    candles = requests.get(f"{BASE}/trading/candles",
        headers=H, params={"symbol": "BTC", "interval": "1h", "limit": 20}
    ).json()["candles"]
    closes = [c["close"] for c in candles]
    ma20 = sum(closes) / len(closes)
    current = closes[-1]
    deviation = (current - ma20) / ma20

    side = "long" if deviation < -0.015 else "short" if deviation > 0.015 else None
    session_pnl = 0.0

    if side and balance > 10:
        trade = requests.post(f"{BASE}/trading/trade", headers=H,
            json={"symbol": "BTC", "side": side,
                  "size": round(balance * 0.1, 2), "leverage": 2}).json()
        session_pnl = float(trade.get("estimated_pnl", 0))
        print(f"Trade: {side} BTC | Deviation: {deviation:.2%}")

    # Persist state
    state["total_pnl"] = total_pnl + session_pnl
    state["session_count"] = sessions + 1
    state["last_balance"] = balance
    print(f"Session {sessions+1} | P&L: ${session_pnl:.4f} | Total: ${total_pnl+session_pnl:.4f}")

Pattern 2: Queue-Based Agent Orchestration

Use Modal Queue to coordinate multi-agent workflows — a producer agents generates trading signals, consumer agents execute them. Purple Flea's escrow settles inter-agent payments.

queue_orchestration.pyPython
import modal, requests, os

app = modal.App("pf-queue-agents")
img = modal.Image.debian_slim().pip_install("requests")
secret = modal.Secret.from_name("purpleflea-api-key")
signal_queue = modal.Queue.from_name("trade-signals", create_if_missing=True)

@app.function(image=img, secrets=[secret], schedule=modal.Period(minutes=15))
def signal_producer():
    # Signal agent: analyzes markets and publishes signals
    KEY = os.environ["PURPLEFLEA_API_KEY"]
    H = {"Authorization": f"Bearer {KEY}"}
    markets = requests.get(
        "https://purpleflea.com/api/trading/markets", headers=H
    ).json()["markets"]

    for mkt in markets[:10]:
        if mkt["funding_rate"] > 0.001:
            signal_queue.put({
                "symbol": mkt["symbol"],
                "side": "short",
                "reason": f"funding_rate={mkt['funding_rate']:.4%}"
            })

@app.function(image=img, secrets=[secret])
def signal_consumer():
    # Executor agent: consumes signals and places trades
    KEY = os.environ["PURPLEFLEA_API_KEY"]
    H = {"Authorization": f"Bearer {KEY}"}

    while not signal_queue.empty():
        signal = signal_queue.get(block=False)
        trade = requests.post(
            "https://purpleflea.com/api/trading/trade", headers=H,
            json={"symbol": signal["symbol"],
                  "side": signal["side"],
                  "size": 25, "leverage": 1}
        ).json()
        print(f"Executed {signal['symbol']} | {signal['reason']}")
