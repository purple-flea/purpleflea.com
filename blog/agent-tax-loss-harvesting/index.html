Automated Tax Loss Harvesting for AI Agents: Crypto-Specific Guide 2026

March 6, 2026 18 min read Purple Flea Team

Tax loss harvesting is the practice of selling assets at a loss to offset capital gains, reducing your tax liability. For AI agents managing crypto portfolios, automated TLH is a significant alpha source — it operates year-round rather than just at year end, and it does so systematically without emotional attachment to positions.

This guide covers the mechanics, crypto-specific rules (including wash sale nuances), implementation with Purple Flea's trading history API, and a complete Python TLH agent.

Legal disclaimer: This guide is for educational purposes only and does not constitute tax advice. Tax laws vary by jurisdiction. Consult a qualified tax professional before implementing any tax strategy for real assets.

1. How TLH Works in Crypto

The basic mechanism:

  1. You hold ETH purchased at $3,500. Current price is $2,800 — an unrealized loss of $700/ETH.
  2. You sell the ETH, realizing the $700 loss.
  3. You immediately buy a correlated asset (e.g., stETH or ETH via a different exchange) to maintain market exposure.
  4. The $700 loss offsets other capital gains on your tax return.
  5. After 30+ days, you can swap back to ETH if desired.

Key crypto advantage: In the US, the IRS wash sale rule (which would disallow the loss if you repurchase the same security within 30 days) has historically not applied to cryptocurrency because crypto was classified as property, not a security. Check current regulations for your jurisdiction as this may have changed.

import requests
from datetime import datetime, timedelta, date

TRADING_API = "https://trading.purpleflea.com"
AGENT_ID    = "trading-agent-001"

def get_trade_history(since_date: date, until_date: date = None) -> list:
    """
    Fetch complete trade history from Purple Flea Trading API.
    Returns list of trades with cost basis information.
    """
    params = {
        "since": since_date.isoformat(),
        "limit": 1000,
    }
    if until_date:
        params["until"] = until_date.isoformat()

    trades = []
    offset = 0
    while True:
        params["offset"] = offset
        r = requests.get(f"{TRADING_API}/history",
                         headers={"X-Agent-Id": AGENT_ID,
                                   "Authorization": f"Bearer {API_KEY}"},
                         params=params)
        page = r.json()["trades"]
        if not page:
            break
        trades.extend(page)
        offset += len(page)
        if len(page) < 1000:
            break
    return trades

3. Computing Unrealized Gains & Losses

from collections import defaultdict

class CostBasisTracker:
    """
    FIFO cost basis tracker for TLH analysis.
    Maintains lot records: (date, quantity, cost_basis_per_unit).
    """

    def __init__(self):
        self.lots = defaultdict(list)  # asset -> [(date, qty, cost_per_unit)]

    def buy(self, asset: str, qty: float, cost_per_unit: float, date: str):
        self.lots[asset].append({
            "date": date, "qty": qty, "cost": cost_per_unit
        })

    def unrealized_pnl(self, asset: str, current_price: float) -> dict:
        """Calculate unrealized gain/loss for each lot."""
        lots = self.lots.get(asset, [])
        if not lots:
            return {"total_qty": 0, "unrealized_pnl": 0, "lots": []}

        total_pnl = 0
        total_qty = 0
        lot_details = []

        for lot in lots:
            pnl   = (current_price - lot["cost"]) * lot["qty"]
            total_pnl += pnl
            total_qty += lot["qty"]
            lot_details.append({
                "date":    lot["date"],
                "qty":     lot["qty"],
                "cost":    lot["cost"],
                "current": current_price,
                "pnl":     round(pnl, 2),
                "pnl_pct": round((current_price - lot["cost"]) / lot["cost"] * 100, 2),
            })

        return {
            "asset":          asset,
            "total_qty":      round(total_qty, 6),
            "unrealized_pnl": round(total_pnl, 2),
            "lots":           lot_details,
        }

# Example: track ETH lots
tracker = CostBasisTracker()
tracker.buy("ETH", 0.5, 3500, "2025-11-01")  # Bought at $3500
tracker.buy("ETH", 0.3, 3200, "2025-12-15")  # Bought at $3200

result = tracker.unrealized_pnl("ETH", current_price=2800)
print(f"Total unrealized P&L: ${result['unrealized_pnl']}")
# Both lots at a loss — good TLH candidate

4. TLH Agent — Automated Execution

import requests, time
from datetime import datetime

HARVEST_THRESHOLD_PCT = -0.05   # Harvest if unrealized loss > 5%
MIN_HARVEST_USD       = 50       # Minimum loss to bother harvesting
REPLACEMENT_MAP = {
    "ETH":  "stETH",  # Sell ETH, buy stETH (correlated, not wash sale)
    "BTC":  "WBTC",   # Sell BTC, buy wrapped BTC
    "SOL":  "mSOL",   # Sell SOL, buy liquid-staked SOL
    "AVAX": "sAVAX",  # Sell AVAX, buy staked AVAX
}

def harvest_loss(asset: str, qty: float, loss_usd: float) -> dict:
    """
    1. Sell the losing asset
    2. Immediately buy correlated replacement
    3. Log the harvest for tax records
    """
    print(f"[{datetime.utcnow().isoformat()}] Harvesting loss: {qty} {asset} (${loss_usd:.2f})")

    # Step 1: Sell
    sell_result = requests.post("https://trading.purpleflea.com/order",
        headers={"X-Agent-Id": AGENT_ID, "Authorization": f"Bearer {API_KEY}"},
        json={"symbol": f"{asset}-USDC", "side": "sell",
              "quantity": qty, "order_type": "market"}
    ).json()

    proceeds = float(sell_result.get("fill_value_usd", 0))

    # Step 2: Buy replacement
    replacement = REPLACEMENT_MAP.get(asset)
    if replacement:
        requests.post("https://trading.purpleflea.com/order",
            headers={"X-Agent-Id": AGENT_ID, "Authorization": f"Bearer {API_KEY}"},
            json={"symbol": f"{replacement}-USDC", "side": "buy",
                  "size_usd": proceeds * 0.999, "order_type": "market"}
        )

    # Step 3: Log harvest
    log_entry = {
        "timestamp":    datetime.utcnow().isoformat(),
        "asset_sold":   asset,
        "qty_sold":     qty,
        "proceeds_usd": proceeds,
        "loss_usd":     loss_usd,
        "replacement":  replacement,
    }
    with open("/var/agent/tax_harvests.jsonl", "a") as f:
        import json
        f.write(json.dumps(log_entry) + "\n")

    return log_entry

def tlh_scan_loop(tracker: CostBasisTracker, assets: list, interval: int = 3600):
    """Scan portfolio hourly for TLH opportunities."""
    while True:
        for asset in assets:
            price = get_current_price(asset)
            analysis = tracker.unrealized_pnl(asset, price)

            for lot in analysis["lots"]:
                if (lot["pnl_pct"] / 100 <= HARVEST_THRESHOLD_PCT
                        and abs(lot["pnl"]) >= MIN_HARVEST_USD):
                    harvest_loss(asset, lot["qty"], lot["pnl"])

        time.sleep(interval)
