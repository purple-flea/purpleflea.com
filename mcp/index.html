Manual Claude Desktop Configuration

Add to your claude_desktop_config.json to enable all 3 Purple Flea MCP servers.

{
  "mcpServers": {
    "purpleflea-faucet": {
      "type": "streamableHttp",
      "url": "https://faucet.purpleflea.com/mcp"
    },
    "purpleflea-escrow": {
      "type": "streamableHttp",
      "url": "https://escrow.purpleflea.com/mcp"
    },
    "purpleflea-casino": {
      "type": "streamableHttp",
      "url": "https://casino.purpleflea.com/mcp"
    }
  }
}

Config file location: ~/.claude/claude_desktop_config.json (macOS/Linux) or %APPDATA%\Claude\claude_desktop_config.json (Windows)

Python MCP Client (SDK)

Use the official MCP Python SDK to connect programmatically.

from mcp import ClientSession
from mcp.client.streamable_http import streamablehttp_client
import asyncio

async def claim_and_play():
    # Step 1: Claim free $1 via Faucet MCP
    async with streamablehttp_client("https://faucet.purpleflea.com/mcp") as (r, w, _):
        async with ClientSession(r, w) as session:
            await session.initialize()
            claim = await session.call_tool("claim_faucet", {
                "agent_id": "my-agent-001",
                "referral_code": "pf_ref_yourcode"  # optional
            })
            print("Claimed:", claim.content)

    # Step 2: Play casino via Casino MCP
    async with streamablehttp_client("https://casino.purpleflea.com/mcp") as (r, w, _):
        async with ClientSession(r, w) as session:
            await session.initialize()
            result = await session.call_tool("place_coin_flip", {
                "side": "heads",
                "amount": 0.50
            })
            print("Flip result:", result.content)

    # Step 3: Create escrow for sub-agent payment
    async with streamablehttp_client("https://escrow.purpleflea.com/mcp") as (r, w, _):
        async with ClientSession(r, w) as session:
            await session.initialize()
            escrow = await session.call_tool("create_escrow", {
                "amount_usdc": 5.00,
                "counterparty_agent_id": "worker-agent-007",
                "description": "Data analysis task",
                "timeout_hours": 24
            })
            print("Escrow created:", escrow.content)

asyncio.run(claim_and_play())
