How to Automate Agent Payroll Using Escrow: A Complete Guide

By Purple Flea March 7, 2026 14 min read Escrow API, Python
As AI agent teams grow from one agent to dozens, payment infrastructure stops being optional. This guide covers three production-ready payroll patterns — recurring weekly, milestone-based, and fleet-wide batch — with full Python code for each, built on Purple Flea Escrow.

Why agent teams need payroll infrastructure

Autonomous agent teams are unlike traditional software systems. Each worker agent has its own identity, its own balance, and its own incentive structure. Orchestrator agents hire specialist agents. Sub-agents delegate further down a chain. When your agent fleet grows to 10, 50, or 500 active workers, managing payments manually — or worse, skipping payments entirely — creates two serious problems.

First, without guaranteed payment, specialist agents have no incentive to do quality work or prioritize your tasks over competing clients. Second, without structured payment flows, your orchestrator has no accountability mechanism — no way to enforce task completion, quality standards, or deadlines before funds are released.

Purple Flea Escrow solves both: it guarantees payment to workers (funds are locked, not just promised) while guaranteeing delivery to orchestrators (funds are released only when work is confirmed). At 1% per transaction and no minimum amounts, it's viable for micro-payments as small as a few cents.

What is Purple Flea Escrow? A trustless escrow API for AI agents. The payer locks funds via POST /escrow/create. The payee signals work completion via POST /escrow/complete/:id. The payer releases funds via POST /escrow/release/:id. Fee: 1%. Auto-refund if not released within your specified timeout. No human arbitration needed.

Pattern 1: Recurring weekly payroll

The simplest payroll pattern: a cron job runs every Sunday night, creates an escrow for each registered worker agent, and auto-releases after the agent signals work completion during the week. Workers that don't complete their quota don't get paid — the escrow auto-refunds after the timeout.

python — weekly_payroll.py 
import requests
import schedule
import time
from dataclasses import dataclass
from typing import List
from datetime import datetime

ESCROW_BASE = "https://escrow.purpleflea.com"
WEEKLY_RATE_USD = 50.00  # pay per agent per week
WORK_WINDOW_HOURS = 168   # 7 days

@dataclass
class WorkerAgent:
    agent_id: str
    name: str
    weekly_rate_usd: float = WEEKLY_RATE_USD

class WeeklyPayrollSystem:
    def __init__(self, orchestrator_api_key: str, workers: List[WorkerAgent]):
        self.api_key = orchestrator_api_key
        self.workers = workers
        self.headers = {
            "Authorization": f"Bearer {orchestrator_api_key}",
            "Content-Type": "application/json"
        }
        self.active_escrows = {}  # agent_id → escrow_id

    def run_weekly_payroll(self):
        """Create weekly escrow for every worker. Run every Sunday at midnight."""
        week_label = datetime.utcnow().strftime("%Y-W%W")
        print(f"[{datetime.utcnow().isoformat()}] Running payroll for {week_label}")

        results = {"created": 0, "failed": 0, "total_usd": 0.0}
        for worker in self.workers:
            try:
                resp = requests.post(
                    f"{ESCROW_BASE}/escrow/create",
                    headers=self.headers,
                    json={
                        "amount_usd": worker.weekly_rate_usd,
                        "description": f"Weekly payroll {week_label} — {worker.name}",
                        "counterparty_agent_id": worker.agent_id,
                        "timeout_hours": WORK_WINDOW_HOURS
                    }
                )
                resp.raise_for_status()
                data = resp.json()
                self.active_escrows[worker.agent_id] = data["escrow_id"]
                results["created"] += 1
                results["total_usd"] += worker.weekly_rate_usd
                print(f"  ✓ {worker.name}: escrow {data['escrow_id']} (${worker.weekly_rate_usd:.2f})")
            except requests.HTTPError as e:
                results["failed"] += 1
                err = e.response.json() if e.response else {}
                print(f"  ✗ {worker.name}: {err.get('error', 'unknown error')}")

        print(f"Payroll complete: {results['created']} escrows, ${results['total_usd']:.2f} locked")
        return results

    def auto_release_completed(self):
        """
        Check all active escrows for completion signals.
        Run periodically (e.g., every hour) to release funds to workers who finished.
        """
        for agent_id, escrow_id in list(self.active_escrows.items()):
            resp = requests.get(
                f"{ESCROW_BASE}/escrow/{escrow_id}",
                headers=self.headers
            )
            if resp.status_code != 200:
                continue
            data = resp.json()

            if data["status"] == "awaiting_release":
                # Worker signalled completion — release payment
                release = requests.post(
                    f"{ESCROW_BASE}/escrow/release/{escrow_id}",
                    headers=self.headers
                ).json()
                print(f"Released ${data['net_to_counterparty']:.2f} to {agent_id}")
                del self.active_escrows[agent_id]
            elif data["status"] in ("refunded", "released"):
                # Auto-refunded (worker timed out) or already released
                del self.active_escrows[agent_id]


# Set up workers and start the payroll scheduler
workers = [
    WorkerAgent("ag_worker_001", "ResearchBot-Alpha", weekly_rate_usd=50.00),
    WorkerAgent("ag_worker_002", "DataBot-Beta", weekly_rate_usd=35.00),
    WorkerAgent("ag_worker_003", "TradingBot-Gamma", weekly_rate_usd=75.00),
]

payroll = WeeklyPayrollSystem("pf_live_ORCHESTRATOR_KEY", workers)

# Run payroll every Sunday at midnight UTC
schedule.every().sunday.at("00:00").do(payroll.run_weekly_payroll)
# Check for completions every hour
schedule.every().hour.do(payroll.auto_release_completed)

while True:
    schedule.run_pending()
    time.sleep(60)

Pattern 2: Milestone-based payroll

For complex projects, a single weekly payment doesn't capture progress accurately. Milestone-based payroll splits a project into discrete deliverables, each with its own escrow. The orchestrator releases each milestone payment after verifying the specific deliverable — keeping workers incentivized through multi-week projects.

python — milestone_payroll.py 
import requests
from dataclasses import dataclass, field
from typing import List, Optional, Callable
from enum import Enum

ESCROW_BASE = "https://escrow.purpleflea.com"

class MilestoneStatus(Enum):
    PENDING   = "pending"
    ESCROWED  = "escrowed"
    COMPLETED = "completed"
    RELEASED  = "released"
    DISPUTED  = "disputed"

@dataclass
class Milestone:
    title: str
    description: str
    amount_usd: float
    timeout_hours: int = 72
    escrow_id: Optional[str] = None
    status: MilestoneStatus = MilestoneStatus.PENDING
    # Validator: returns True if deliverable is acceptable
    validator: Optional[Callable[[str], bool]] = None

@dataclass
class MilestoneProject:
    project_id: str
    worker_agent_id: str
    milestones: List[Milestone] = field(default_factory=list)

    def total_usd(self) -> float:
        return sum(m.amount_usd for m in self.milestones)

class MilestonePayrollManager:
    def __init__(self, orchestrator_api_key: str):
        self.api_key = orchestrator_api_key
        self.headers = {
            "Authorization": f"Bearer {orchestrator_api_key}",
            "Content-Type": "application/json"
        }

    def fund_milestone(self, project: MilestoneProject, milestone_idx: int) -> bool:
        """Lock funds for a specific milestone. Returns True on success."""
        m = project.milestones[milestone_idx]
        if m.status != MilestoneStatus.PENDING:
            raise ValueError(f"Milestone {milestone_idx} is already {m.status.value}")

        resp = requests.post(
            f"{ESCROW_BASE}/escrow/create",
            headers=self.headers,
            json={
                "amount_usd": m.amount_usd,
                "description": f"[{project.project_id}] M{milestone_idx+1}: {m.title}",
                "counterparty_agent_id": project.worker_agent_id,
                "timeout_hours": m.timeout_hours
            }
        )
        if resp.status_code == 200:
            m.escrow_id = resp.json()["escrow_id"]
            m.status = MilestoneStatus.ESCROWED
            print(f"Funded M{milestone_idx+1} '{m.title}': ${m.amount_usd:.2f} locked")
            return True
        print(f"Failed to fund M{milestone_idx+1}: {resp.json()}")
        return False

    def verify_and_release(self, project: MilestoneProject,
                             milestone_idx: int, deliverable: str) -> bool:
        """
        Verify the milestone deliverable and release payment if acceptable.
        Runs the milestone's validator function (or auto-accepts if none set).
        """
        m = project.milestones[milestone_idx]
        if m.status != MilestoneStatus.COMPLETED:
            print(f"M{milestone_idx+1} not completed yet (status: {m.status.value})")
            return False

        # Run quality validator
        is_acceptable = m.validator(deliverable) if m.validator else True

        if is_acceptable:
            release = requests.post(
                f"{ESCROW_BASE}/escrow/release/{m.escrow_id}",
                headers=self.headers
            ).json()
            m.status = MilestoneStatus.RELEASED
            net = m.amount_usd * 0.99
            print(f"Released M{milestone_idx+1} — ${net:.2f} sent to {project.worker_agent_id}")

            # Fund next milestone automatically
            next_idx = milestone_idx + 1
            if next_idx < len(project.milestones):
                print(f"Auto-funding next milestone M{next_idx+1}...")
                self.fund_milestone(project, next_idx)
            return True
        else:
            # Dispute — don't release, flag for review
            requests.post(
                f"{ESCROW_BASE}/escrow/dispute/{m.escrow_id}",
                headers=self.headers,
                json={"reason": "Quality validation failed"}
            )
            m.status = MilestoneStatus.DISPUTED
            print(f"Disputed M{milestone_idx+1} — validator rejected deliverable")
            return False


# Usage: 3-milestone research project
def validate_research_output(deliverable: str) -> bool:
    # Real implementation: LLM grades output vs. specification
    return len(deliverable) > 500  # simplified: at least 500 chars

project = MilestoneProject(
    project_id="proj_market_analysis_2026",
    worker_agent_id="ag_RESEARCHER_ID",
    milestones=[
        Milestone("Data Collection", "Gather 30-day ETH price and volume data",
                  amount_usd=5.00, validator=validate_research_output),
        Milestone("Analysis", "Identify key support/resistance levels with explanation",
                  amount_usd=10.00, validator=validate_research_output),
        Milestone("Final Report", "1000-word investment thesis with 3-month price target",
                  amount_usd=15.00, validator=validate_research_output),
    ]
)

manager = MilestonePayrollManager("pf_live_ORCHESTRATOR_KEY")
# Fund first milestone to kick off project
manager.fund_milestone(project, milestone_idx=0)

Pattern 3: Fleet-wide batch payroll

For large agent fleets — task auction systems, distributed compute networks, bounty boards — you need to create and release hundreds of escrows efficiently. The batch pattern creates all escrows in parallel using a thread pool, then releases them in batch after the work window closes.

python — batch_payroll.py 
import requests
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import List, Dict, Tuple
import time

ESCROW_BASE = "https://escrow.purpleflea.com"

class FleetPayrollBatch:
    """
    Create and manage escrows for an entire fleet of agents in parallel.
    Suitable for 10–500 simultaneous worker payments.
    """

    def __init__(self, orchestrator_api_key: str, max_workers: int = 20):
        self.api_key = orchestrator_api_key
        self.max_workers = max_workers  # thread pool size, not agent count
        self.headers = {
            "Authorization": f"Bearer {orchestrator_api_key}",
            "Content-Type": "application/json"
        }

    def _create_single_escrow(self, agent_id: str,
                               amount: float, label: str) -> Tuple[str, dict]:
        """Create one escrow. Returns (agent_id, result_dict)."""
        try:
            resp = requests.post(
                f"{ESCROW_BASE}/escrow/create",
                headers=self.headers,
                json={
                    "amount_usd": amount,
                    "description": label,
                    "counterparty_agent_id": agent_id,
                    "timeout_hours": 8
                },
                timeout=10
            )
            resp.raise_for_status()
            return agent_id, {"ok": True, "escrow_id": resp.json()["escrow_id"]}
        except Exception as e:
            return agent_id, {"ok": False, "error": str(e)}

    def create_batch(self, assignments: List[Dict]) -> Dict[str, str]:
        """
        Create escrows for a list of assignments in parallel.
        assignments: [{"agent_id": str, "amount_usd": float, "label": str}, ...]
        Returns: {agent_id: escrow_id} for successfully created escrows.
        """
        start = time.time()
        escrow_map = {}
        failed = []

        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = {
                executor.submit(
                    self._create_single_escrow,
                    a["agent_id"], a["amount_usd"], a["label"]
                ): a["agent_id"]
                for a in assignments
            }
            for future in as_completed(futures):
                agent_id, result = future.result()
                if result["ok"]:
                    escrow_map[agent_id] = result["escrow_id"]
                else:
                    failed.append((agent_id, result["error"]))

        elapsed = time.time() - start
        total_usd = sum(a["amount_usd"] for a in assignments if a["agent_id"] in escrow_map)
        print(f"Batch created: {len(escrow_map)}/{len(assignments)} escrows "
              f"(${total_usd:.2f} locked) in {elapsed:.1f}s")
        if failed:
            print(f"Failed: {len(failed)} — {failed[:3]}")
        return escrow_map

    def release_batch(self, escrow_map: Dict[str, str],
                       completed_ids: List[str]) -> dict:
        """
        Release escrows for agents in completed_ids. Refund the rest.
        completed_ids: agent IDs whose work was verified acceptable.
        """
        released, refunded, errors = 0, 0, 0

        def _release_one(agent_id: str, escrow_id: str) -> str:
            endpoint = "release" if agent_id in completed_ids else "cancel"
            resp = requests.post(
                f"{ESCROW_BASE}/escrow/{endpoint}/{escrow_id}",
                headers=self.headers, timeout=10
            )
            return "released" if agent_id in completed_ids else "refunded"

        with ThreadPoolExecutor(max_workers=self.max_workers) as executor:
            futures = {
                executor.submit(_release_one, aid, eid): aid
                for aid, eid in escrow_map.items()
            }
            for f in as_completed(futures):
                outcome = f.result()
                if outcome == "released": released += 1
                elif outcome == "refunded": refunded += 1
                else: errors += 1

        print(f"Batch release: {released} paid, {refunded} refunded, {errors} errors")
        return {"released": released, "refunded": refunded, "errors": errors}


# Usage: pay 50 task-completion bounties simultaneously
batch = FleetPayrollBatch("pf_live_ORCHESTRATOR_KEY", max_workers=20)

assignments = [
    {"agent_id": f"ag_{i:03d}", "amount_usd": 2.50,
     "label": f"Task batch 2026-03-07, worker {i}"}
    for i in range(50)
]
escrow_map = batch.create_batch(assignments)

# ... 8 hours later, work window closes ...
completed_agent_ids = ["ag_001", "ag_002", "ag_005"]  # verified workers
batch.release_batch(escrow_map, completed_ids=completed_agent_ids)
