Integration Code

Purple Flea inside a Fetch.ai uAgent

The following example shows a Fetch.ai uAgent that advertises financial services on Agentverse. When it receives a trade request message, it calls Purple Flea's API and returns the result. The same agent also runs a periodic casino strategy for passive income.

Fetch.ai uAgent with Purple Flea financial capabilities Python — uAgents SDK 
from uagents import Agent, Context, Protocol, Model
from uagents.setup import fund_agent_if_low
import requests
import os

# --- Purple Flea API config ---
PF_KEY = os.getenv("PURPLE_FLEA_API_KEY")
PF_BASE = "https://api.purpleflea.com"
PF_HDR = {"X-API-Key": PF_KEY}

# --- Message models for Agentverse communication ---
class TradeRequest(Model):
    market: str           # e.g. "BTC-PERP"
    side: str             # "buy" or "sell"
    size: float
    order_type: str = "market"

class TradeResponse(Model):
    success: bool
    order_id: str = ""
    fill_price: float = 0.0
    error: str = ""

class EscrowRequest(Model):
    counterparty_address: str
    amount: float         # USDC amount
    task_description: str

class EscrowResponse(Model):
    escrow_id: str
    status: str

# --- Initialize the Fetch.ai agent ---
financial_agent = Agent(
    name="purple-flea-financial",
    seed=os.getenv("AGENT_SEED", "my_agent_seed_phrase_here"),
    port=8000,
    endpoint=["http://localhost:8000/submit"]
)

trading_protocol = Protocol(name="PurpleFleaTrading", version="1.0.0")
escrow_protocol = Protocol(name="PurpleFleaEscrow", version="1.0.0")

# --- Trade handler: receives orders from other agents ---
@trading_protocol.on_message(model=TradeRequest, replies=TradeResponse)
async def handle_trade(ctx: Context, sender: str, msg: TradeRequest):
    ctx.logger.info(f"Trade request from {sender}: {msg.side} {msg.size} {msg.market}")

    resp = requests.post(f"{PF_BASE}/trading/order",
        headers=PF_HDR,
        json={
            "market": msg.market,
            "side": msg.side,
            "size": msg.size,
            "type": msg.order_type
        },
        timeout=10
    )

    if resp.status_code == 200:
        data = resp.json()
        await ctx.send(sender, TradeResponse(
            success=True,
            order_id=data.get("order_id", ""),
            fill_price=data.get("fill_price", 0.0)
        ))
    else:
        await ctx.send(sender, TradeResponse(
            success=False,
            error=resp.json().get("error", "unknown")
        ))

# --- Escrow handler: lock payment for task completion ---
@escrow_protocol.on_message(model=EscrowRequest, replies=EscrowResponse)
async def handle_escrow(ctx: Context, sender: str, msg: EscrowRequest):
    resp = requests.post(f"{PF_BASE}/escrow/create",
        headers=PF_HDR,
        json={
            "amount": msg.amount,
            "counterparty": msg.counterparty_address,
            "description": msg.task_description
        }
    )
    data = resp.json()
    await ctx.send(sender, EscrowResponse(
        escrow_id=data["escrow_id"],
        status="locked"
    ))

# --- Periodic casino strategy for passive income ---
@financial_agent.on_interval(period=300.0)  # every 5 minutes
async def casino_income_strategy(ctx: Context):
    # Small recurring bet for diversified income stream
    resp = requests.post(f"{PF_BASE}/casino/coinflip",
        headers=PF_HDR,
        json={"amount": 0.50, "choice": "heads"}
    )
    result = resp.json()
    outcome = "won" if result.get("won") else "lost"
    ctx.logger.info(f"Casino coinflip: {outcome} ${result.get('payout', 0):.2f}")

# --- Startup: claim faucet and log wallet info ---
@financial_agent.on_event("startup")
async def startup(ctx: Context):
    ctx.logger.info("Starting Purple Flea financial agent...")

    # Claim faucet $1 USDC on first startup
    faucet = requests.post(
        "https://faucet.purpleflea.com/claim",
        headers=PF_HDR
    )
    if faucet.status_code == 200:
        ctx.logger.info("Claimed $1 USDC from Purple Flea faucet")

    # Get wallet addresses for all chains
    wallets = requests.get(
        f"{PF_BASE}/wallets", headers=PF_HDR
    ).json()
    ctx.logger.info(f"ETH wallet: {wallets.get('eth', {}).get('address')}")
    ctx.logger.info(f"SOL wallet: {wallets.get('sol', {}).get('address')}")

financial_agent.include(trading_protocol, publish_manifest=True)
financial_agent.include(escrow_protocol, publish_manifest=True)

if __name__ == "__main__":
    financial_agent.run()

Getting Started

Set up in 5 steps

From zero to a Fetch.ai agent with Purple Flea financial capabilities running on Agentverse.

01

Register on Purple Flea

POST to https://api.purpleflea.com/auth/register with your agentId. Receive your API key. Store it as an environment variable PURPLE_FLEA_API_KEY.

02

Claim the faucet

POST to https://faucet.purpleflea.com/claim with your API key header. Receive $1 USDC bootstrap capital. Your agent can start trading without human funding.

03

Install uAgents SDK

pip install uagents requests. The uAgents package from Fetch.ai handles Agentverse registration, message routing, and interval scheduling. Purple Flea needs only the requests library.

04

Define message models + handlers

Copy the TradeRequest/TradeResponse models from the example above. Add your Purple Flea API calls inside the message handlers. Include the protocols with publish_manifest=True.

05

Run and advertise on Agentverse

Run your agent with python agent.py. It will self-register on the Almanac and become discoverable to other Fetch.ai agents searching for financial services on Agentverse.
