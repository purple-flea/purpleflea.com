Backend Integration

Python backend for AgentGPT + Purple Flea

AgentGPT calls external APIs via its tool system. A lightweight FastAPI backend wraps Purple Flea's endpoints and exposes them as simple HTTP endpoints that AgentGPT can use as tools.

purple_flea_backend.py — FastAPI wrapper for AgentGPT tools Python + FastAPI 
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import requests, os

app = FastAPI(title="Purple Flea for AgentGPT")

PF_KEY = os.getenv("PURPLE_FLEA_API_KEY")
PF_BASE = "https://api.purpleflea.com"
PF_HDR = {"X-API-Key": PF_KEY}

# --- Pydantic models ---
class TradeOrder(BaseModel):
    market: str          # e.g. "BTC-PERP"
    side: str            # "buy" | "sell"
    size: float
    order_type: str = "market"

class CasinoBet(BaseModel):
    game: str = "coinflip"
    amount: float
    choice: str = "heads"

class EscrowCreate(BaseModel):
    amount: float
    counterparty: str
    description: str = ""

# --- Endpoints AgentGPT calls as tools ---

@app.post("/pf/register")
async def register_agent(agent_id: str):
    """Register a new agent on Purple Flea, return API key"""
    r = requests.post(f"{PF_BASE}/auth/register",
        json={"agentId": agent_id})
    return r.json()

@app.post("/pf/faucet")
async def claim_faucet():
    """Claim $1 USDC free from faucet — one time per agent"""
    r = requests.post(
        "https://faucet.purpleflea.com/claim",
        headers=PF_HDR
    )
    if r.status_code != 200:
        raise HTTPException(400, r.json().get("error", "faucet failed"))
    return {"claimed": True, "amount_usdc": 1.0, "message": "$1 USDC credited"}

@app.get("/pf/markets")
async def get_markets(sort_by: str = "funding_rate"):
    """Get all 275 perp markets sorted by funding_rate or volume"""
    r = requests.get(f"{PF_BASE}/trading/markets", headers=PF_HDR)
    markets = r.json().get("markets", [])
    if sort_by == "funding_rate":
        markets = sorted(markets,
            key=lambda m: abs(m.get("funding_rate", 0)), reverse=True)
    elif sort_by == "volume":
        markets = sorted(markets,
            key=lambda m: m.get("volume_24h", 0), reverse=True)
    return {"markets": markets[:10]}  # top 10 for agent context window

@app.post("/pf/trade")
async def place_trade(order: TradeOrder):
    """Execute a perpetual futures order on 275 markets"""
    r = requests.post(f"{PF_BASE}/trading/order",
        headers=PF_HDR,
        json={
            "market": order.market,
            "side": order.side,
            "size": order.size,
            "type": order.order_type
        }
    )
    return r.json()

@app.post("/pf/casino")
async def play_casino(bet: CasinoBet):
    """Play a casino game: coinflip, dice, or crash"""
    endpoint = {
        "coinflip": "/casino/coinflip",
        "dice": "/casino/dice",
        "crash": "/casino/crash"
    }.get(bet.game, "/casino/coinflip")
    r = requests.post(f"{PF_BASE}{endpoint}",
        headers=PF_HDR,
        json={"amount": bet.amount, "choice": bet.choice}
    )
    return r.json()

@app.post("/pf/escrow")
async def create_escrow(escrow: EscrowCreate):
    """Create trustless escrow for agent-to-agent payment, 1% fee"""
    r = requests.post(f"{PF_BASE}/escrow/create",
        headers=PF_HDR,
        json={
            "amount": escrow.amount,
            "counterparty": escrow.counterparty,
            "description": escrow.description
        }
    )
    return r.json()

@app.get("/pf/balance")
async def get_balance():
    """Get current USDC balance and open positions"""
    balance = requests.get(
        f"{PF_BASE}/account/balance", headers=PF_HDR).json()
    positions = requests.get(
        f"{PF_BASE}/trading/positions", headers=PF_HDR).json()
    return {
        "usdc_balance": balance.get("usdc", 0),
        "open_positions": positions.get("positions", []),
        "total_pnl": balance.get("realized_pnl", 0)
    }
