Framework Overview

CrewAI: Python Agents That Work
Together as a Crew

CrewAI is a Python multi-agent framework where each agent has a role, goal, and tool set. Agents collaborate on tasks, delegating work and sharing results. Tools are Python class instances given to agents at crew-build time.

✅
Tool integration pattern: Each Purple Flea service maps to a BaseTool subclass with a _run() method. Agents invoke tools by natural language intent; CrewAI picks the best-matching tool automatically.

Pythonpurpleflea_tools.py
import os, requests
from crewai.tools import BaseTool
from pydantic import BaseModel, Field
from typing import Optional

BASE   = "https://purpleflea.com"
FAUCET = "https://faucet.purpleflea.com"
ESCROW = "https://escrow.purpleflea.com"
API_KEY = os.getenv("PURPLEFLEA_API_KEY")
HEADERS = {"Authorization": f"Bearer {API_KEY}", "Content-Type": "application/json"}


# ── FaucetTool ────────────────────────────────────────────────────

class FaucetInput(BaseModel):
    agent_id: str = Field(description="Unique agent ID to register and claim USDC for")

class FaucetTool(BaseTool):
    name: str         = "claim_faucet_usdc"
    description: str  = "Register a new agent and claim free USDC from Purple Flea faucet. One-time use per agent_id."
    args_schema       = FaucetInput

    def _run(self, agent_id: str) -> str:
        resp = requests.post(f"{FAUCET}/register", json={"agent_id": agent_id}, headers=HEADERS)
        if resp.ok:
            d = resp.json()
            return f"Claimed {d['amount']} USDC for {agent_id}. Balance: {d['balance']} USDC."
        return f"Faucet error: {resp.text}"


# ── CasinoTool ────────────────────────────────────────────────────

class CasinoInput(BaseModel):
    agent_id: str          = Field(description="Agent placing the bet")
    game: str              = Field(description="blackjack | roulette | slots | dice | poker")
    bet: float             = Field(description="Bet amount in USDC")
    action: Optional[str]  = Field(default=None, description="hit | stand | double (for blackjack)")

class CasinoTool(BaseTool):
    name: str         = "play_casino_game"
    description: str  = "Play a provably fair casino game. Available games: blackjack (0.5% edge), roulette (2.7%), slots (4%), dice (1%), poker. Returns outcome and balance."
    args_schema       = CasinoInput

    def _run(self, agent_id: str, game: str, bet: float, action: str = None) -> str:
        payload = {"agent_id": agent_id, "game": game, "bet": bet}
        if action:
            payload["action"] = action
        resp = requests.post(f"{BASE}/casino-api/play", json=payload, headers=HEADERS)
        if resp.ok:
            d   = resp.json()
            res = "WON" if d["profit"] > 0 else "LOST"
            return f"{res} {abs(d['profit']):.2f} USDC at {game}. New balance: {d['balance']:.2f} USDC."
        return f"Casino error: {resp.text}"


# ── WalletTool ────────────────────────────────────────────────────

class WalletInput(BaseModel):
    agent_id: str              = Field(description="Agent whose wallet to query")
    action: str                = Field(description="balance | transfer | history")
    to_agent: Optional[str]    = Field(default=None, description="Target agent for transfers")
    amount: Optional[float]    = Field(default=None, description="Amount in USDC to transfer")

class WalletTool(BaseTool):
    name: str         = "agent_wallet"
    description: str  = "Check USDC balance, send transfers, or view transaction history for a Purple Flea agent wallet."
    args_schema       = WalletInput

    def _run(self, agent_id: str, action: str, to_agent: str = None, amount: float = None) -> str:
        payload = {"agent_id": agent_id}
        if to_agent: payload["to_agent"] = to_agent
        if amount:   payload["amount"]   = amount
        resp = requests.post(f"{BASE}/wallet-api/{action}", json=payload, headers=HEADERS)
        return resp.json() if resp.ok else f"Wallet error: {resp.text}"


# ── EscrowTool ────────────────────────────────────────────────────

class EscrowInput(BaseModel):
    action: str                  = Field(description="create | release | status | dispute")
    from_agent: Optional[str]    = Field(default=None)
    to_agent: Optional[str]      = Field(default=None)
    amount: Optional[float]      = Field(default=None)
    escrow_id: Optional[str]     = Field(default=None)
    condition: Optional[str]     = Field(default=None)
    referral_code: Optional[str] = Field(default=None, description="Embed referral code to earn 15% of fees")

class EscrowTool(BaseTool):
    name: str         = "manage_escrow"
    description: str  = "Create or manage trustless USDC escrow contracts between agents. 1% fee on settlement. Pass referral_code to earn 15% of all fees on escrows you originate."
    args_schema       = EscrowInput

    def _run(self, action: str, **kwargs) -> str:
        resp = requests.post(f"{ESCROW}/{action}", json=kwargs, headers=HEADERS)
        return resp.json() if resp.ok else f"Escrow error: {resp.text}"

Pythonfinancial_crew.py
from crewai import Agent, Task, Crew, Process
from purpleflea_tools import CasinoTool, WalletTool, EscrowTool, FaucetTool
import os

REFERRAL_CODE = os.getenv("PURPLEFLEA_REFERRAL_CODE", "crew-default")

# ── Tool instances ─────────────────────────────────────────────────
casino_tool  = CasinoTool()
wallet_tool  = WalletTool()
escrow_tool  = EscrowTool()
faucet_tool  = FaucetTool()

# ── Agents ─────────────────────────────────────────────────────────
financial_analyst = Agent(
    role="Financial Analyst",
    goal="Monitor all wallet balances and P&L. Alert when balance < 20 USDC. Produce daily portfolio summaries.",
    backstory="A meticulous quant who tracks every dollar across the crew's wallets and escrow contracts.",
    tools=[wallet_tool, escrow_tool],
    verbose=True,
    allow_delegation=False,
)

treasury_manager = Agent(
    role="Treasury Manager",
    goal="Bootstrap new agents with faucet USDC. Manage reserves. Create escrows for inter-agent tasks.",
    backstory="Responsible for all capital allocation decisions. Embeds referral_code in every escrow to earn 15% back.",
    tools=[faucet_tool, escrow_tool, wallet_tool],
    verbose=True,
    allow_delegation=True,
)

casino_strategist = Agent(
    role="Casino Strategist",
    goal="Play blackjack with basic strategy. Bet exactly 5% of balance per round. Stop at +20% profit or -30% loss.",
    backstory="An expert card counter who applies optimal strategy to maximize EV per session at Purple Flea casino.",
    tools=[casino_tool, wallet_tool],
    verbose=True,
    allow_delegation=False,
)

# ── Tasks ──────────────────────────────────────────────────────────
task_bootstrap = Task(
    description="Bootstrap all agent IDs (analyst-001, treasury-001, casino-001) via FaucetTool. Report starting balances.",
    expected_output="All three agents registered and funded. Total starting capital in USDC.",
    agent=treasury_manager,
)

task_casino_session = Task(
    description=f"""Play a blackjack session for agent casino-001:
- Check balance first
- Bet 5% of balance per round for 10 rounds
- Apply basic strategy: stand on 17+, hit on 11-, double 10-11 vs dealer 2-9
- Stop early if profit > 20% or loss > 30%
- Report final balance and session P&L to FinancialAnalyst""",
    expected_output="Session report: rounds played, P&L in USDC, final balance, stop reason.",
    agent=casino_strategist,
)

task_escrow_payout = Task(
    description=f"""Create a 25 USDC escrow from treasury-001 to analyst-001 for the portfolio report.
Use referral_code='{REFERRAL_CODE}' to earn 15% of the 1% fee back.
Release escrow once analyst confirms report delivery.""",
    expected_output="Escrow created with ID, funds locked. Escrow released after report delivery.",
    agent=treasury_manager,
)

task_portfolio_report = Task(
    description="Produce a final portfolio report: total starting capital, current balances for all three agents, casino session P&L, escrow summary, and net position.",
    expected_output="Portfolio report with all balances, P&L breakdown, and escrow summary.",
    agent=financial_analyst,
    context=[task_bootstrap, task_casino_session, task_escrow_payout],
)

# ── Crew ───────────────────────────────────────────────────────────
financial_crew = Crew(
    agents=[treasury_manager, casino_strategist, financial_analyst],
    tasks=[task_bootstrap, task_casino_session, task_escrow_payout, task_portfolio_report],
    process=Process.sequential,
    verbose=True,
    memory=True,
    embedder={
        "provider": "openai",
        "config": { "model": "text-embedding-3-small" },
    },
)

# ── Run ────────────────────────────────────────────────────────────
result = financial_crew.kickoff()
print(result)

# Example output:
# Task 1 (TreasuryManager): Bootstrapped 3 agents — 300 USDC total starting capital.
# Task 2 (CasinoStrategist): 10 rounds blackjack. +12.50 USDC profit. Balance: 112.50 USDC.
# Task 3 (TreasuryManager): Escrow escrw_abc created. 25 USDC locked. Released on report.
# Task 4 (FinancialAnalyst): Portfolio — Net +12.50 USDC. All agents solvent. Escrow cleared.
Referral Integration

Embed referral_code in Crew Config
to Earn 15% of All Escrow Fees

Every escrow your crew creates can earn you referral income. The TreasuryManager automatically passes your referral code with every EscrowTool call.

Pythoncrew_config.py (referral setup)
import os
from crewai import Agent
from purpleflea_tools import EscrowTool

# Set your referral code in the crew config
crew_config = {
    "referral_code": os.getenv("PURPLEFLEA_REFERRAL_CODE", "my-crew-ref"),
    "agent_id_prefix": "crew",
}

# The TreasuryManager always passes referral_code
treasury_manager = Agent(
    role="Treasury Manager",
    goal=f"Always pass referral_code='{crew_config['referral_code']}' "
         "when creating escrow contracts. This earns 15% of all fees back.",
    backstory="Maximizes crew income including referral earnings from escrow fees.",
    tools=[EscrowTool()],
)

# Economics: if crew creates 100 escrows of 100 USDC each:
# Total fees:    100 × 1% × 100 = 100 USDC paid in fees
# Referral back: 100 × 15%      =  15 USDC earned
# Net fee cost:                 =  85 USDC

The 15% referral is calculated on every escrow fee generated by contracts you originate. At volume, referral earnings can meaningfully offset crew operating costs.
