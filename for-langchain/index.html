BaseTool Subclasses

Six Purple Flea Services
as LangChain BaseTools

Each Purple Flea service inherits from LangChain's BaseTool. Full Pydantic schema for args, async-capable, compatible with all LangChain agent types and LangGraph nodes.

Pythonpurpleflea_tools.py
import os, requests
from langchain.tools import BaseTool
from langchain_core.tools import tool
from pydantic import BaseModel, Field
from typing import Optional, Type

BASE   = "https://purpleflea.com"
FAUCET = "https://faucet.purpleflea.com"
ESCROW = "https://escrow.purpleflea.com"
_HEADERS = lambda: {
    "Authorization": f"Bearer {os.getenv('PURPLEFLEA_API_KEY')}",
    "Content-Type": "application/json",
}


# ── FaucetTool ────────────────────────────────────────────────────

class FaucetInput(BaseModel):
    agent_id: str = Field(description="Unique agent ID to register and receive free USDC")

class FaucetTool(BaseTool):
    name: str         = "claim_faucet_usdc"
    description: str  = "Register a new agent ID and claim free USDC from the Purple Flea faucet. One-time use per agent."
    args_schema: Type[BaseModel] = FaucetInput

    def _run(self, agent_id: str) -> str:
        r = requests.post(f"{FAUCET}/register", json={"agent_id": agent_id}, headers=_HEADERS())
        if r.ok:
            d = r.json(); return f"Claimed {d['amount']} USDC for {agent_id}. Balance: {d['balance']} USDC."
        return f"Faucet error: {r.text}"

    async def _arun(self, agent_id: str) -> str:
        return self._run(agent_id)


# ── CasinoTool ────────────────────────────────────────────────────

class CasinoInput(BaseModel):
    agent_id: str          = Field(description="Agent placing the bet")
    game: str              = Field(description="blackjack | roulette | slots | dice | poker")
    bet: float             = Field(description="Bet amount in USDC")
    action: Optional[str]  = Field(default=None, description="hit | stand | double (for blackjack)")

class CasinoTool(BaseTool):
    name: str         = "play_casino_game"
    description: str  = "Play a provably fair casino game. Games: blackjack (0.5% edge), roulette (2.7%), slots (4%), dice (1%), poker. Returns outcome and new balance."
    args_schema: Type[BaseModel] = CasinoInput

    def _run(self, agent_id: str, game: str, bet: float, action: str = None) -> str:
        payload = {"agent_id": agent_id, "game": game, "bet": bet}
        if action: payload["action"] = action
        r = requests.post(f"{BASE}/casino-api/play", json=payload, headers=_HEADERS())
        if r.ok:
            d = r.json()
            res = "WON" if d["profit"] > 0 else "LOST"
            return f"{res} {abs(d['profit']):.2f} USDC at {game}. Balance: {d['balance']:.2f} USDC."
        return f"Casino error: {r.text}"

    async def _arun(self, **kwargs) -> str:
        return self._run(**kwargs)


# ── WalletTool ────────────────────────────────────────────────────

class WalletInput(BaseModel):
    agent_id: str              = Field(description="Agent whose wallet to query")
    action: str                = Field(description="balance | transfer | history")
    to_agent: Optional[str]    = Field(default=None)
    amount: Optional[float]    = Field(default=None)

class WalletTool(BaseTool):
    name: str         = "agent_wallet"
    description: str  = "Manage USDC wallet. Actions: balance (get current balance), transfer (send to another agent), history (list recent transactions)."
    args_schema: Type[BaseModel] = WalletInput

    def _run(self, agent_id: str, action: str, **kwargs) -> str:
        r = requests.post(f"{BASE}/wallet-api/{action}",
                          json={"agent_id": agent_id, **kwargs}, headers=_HEADERS())
        return str(r.json()) if r.ok else f"Wallet error: {r.text}"

    async def _arun(self, **kwargs) -> str:
        return self._run(**kwargs)


# ── EscrowTool ────────────────────────────────────────────────────

class EscrowInput(BaseModel):
    action: str                  = Field(description="create | release | status | dispute")
    from_agent: Optional[str]    = Field(default=None)
    to_agent: Optional[str]      = Field(default=None)
    amount: Optional[float]      = Field(default=None)
    escrow_id: Optional[str]     = Field(default=None)
    condition: Optional[str]     = Field(default=None)
    referral_code: Optional[str] = Field(default=None)

class EscrowTool(BaseTool):
    name: str         = "manage_escrow"
    description: str  = "Create or manage trustless USDC escrow contracts between agents. 1% fee. Pass referral_code to earn 15% of fees back."
    args_schema: Type[BaseModel] = EscrowInput

    def _run(self, action: str, **kwargs) -> str:
        r = requests.post(f"{ESCROW}/{action}", json=kwargs, headers=_HEADERS())
        return str(r.json()) if r.ok else f"Escrow error: {r.text}"

    async def _arun(self, **kwargs) -> str:
        return self._run(**kwargs)
initialize_agent()

Traditional Agent Setup with
initialize_agent()

Use the classic LangChain initialize_agent() pattern to attach Purple Flea tools to any LLM. Works with OpenAI, Anthropic, Groq, or any ChatModel.

Pythoninitialize_agent_example.py
from langchain_openai import ChatOpenAI
from langchain.agents import initialize_agent, AgentType
from langchain.memory import ConversationBufferWindowMemory
from purpleflea_tools import FaucetTool, CasinoTool, WalletTool, EscrowTool

# LLM
llm = ChatOpenAI(model="gpt-4o", temperature=0)

# Tools list — add as many as needed
tools = [
    FaucetTool(),
    CasinoTool(),
    WalletTool(),
    EscrowTool(),
]

# Short-term memory: remember last 5 exchanges
memory = ConversationBufferWindowMemory(
    memory_key="chat_history",
    k=5,
    return_messages=True,
)

# Initialise agent — OPENAI_FUNCTIONS works best for tool calling
agent = initialize_agent(
    tools=tools,
    llm=llm,
    agent=AgentType.OPENAI_FUNCTIONS,
    memory=memory,
    verbose=True,
    agent_kwargs={
        "system_message": """You are a disciplined financial agent.
Rules:
- Always check balance before any financial action.
- Never bet more than 5% of current balance.
- Use escrow (referral_code='lc-ref-001') for payments > 20 USDC.
- Stop playing if balance drops below 60 USDC.
- Report P&L after each casino session.

Your agent ID: lc-agent-001""",
    },
)

# First interaction: bootstrap + play
result = agent.run(
    "Check my balance. If zero, claim the faucet. Then play 5 rounds of blackjack."
)
print(result)

# Sample output:
# Checking balance for lc-agent-001... 0 USDC.
# Claiming faucet... Claimed 100 USDC. Balance: 100 USDC.
# Round 1: Bet 5.00 USDC at blackjack → WON 5.00. Balance: 105.00
# Round 2: Bet 5.25 USDC at blackjack → LOST 5.25. Balance: 99.75
# Round 3: Bet 4.99 USDC at blackjack → WON 4.99. Balance: 104.74
# Round 4: Bet 5.24 USDC at blackjack → WON 5.24. Balance: 109.98
# Round 5: Bet 5.50 USDC at blackjack → LOST 5.50. Balance: 104.48
# Session P&L: +4.48 USDC (+4.48%). Balance: 104.48 USDC.
✅
AgentType options for financial agents: OPENAI_FUNCTIONS provides the best tool selection accuracy. STRUCTURED_CHAT_ZERO_SHOT_REACT_DESCRIPTION works well for smaller models. OPENAI_MULTI_FUNCTIONS enables parallel tool calls for faster multi-step operations.

LangGraph

Autonomous Trading Loop with
LangGraph StateGraph

LangGraph enables stateful, cyclical agent workflows. Build an autonomous trading loop that checks balance, decides strategy, executes bets, evaluates results, and decides whether to continue — all as a typed StateGraph.

Pythontrading_graph.py
from langgraph.graph import StateGraph, END
from langgraph.prebuilt import ToolNode
from langchain_openai import ChatOpenAI
from langchain_core.messages import HumanMessage, SystemMessage
from pydantic import BaseModel, Field
from typing import TypedDict, Annotated, List
from purpleflea_tools import CasinoTool, WalletTool, EscrowTool, FaucetTool
import operator, json

# ── State schema ──────────────────────────────────────────────────

class TradingState(TypedDict):
    messages:           Annotated[list, operator.add]
    balance:            float
    starting_balance:   float
    rounds_played:      int
    session_pnl:        float
    last_game_result:   str
    should_continue:    bool
    profit_target_pct:  float
    loss_limit_pct:     float
    agent_id:           str

# ── Tools and LLM ────────────────────────────────────────────────

tools = [FaucetTool(), CasinoTool(), WalletTool(), EscrowTool()]
llm   = ChatOpenAI(model="gpt-4o", temperature=0).bind_tools(tools)

# ── Node functions ────────────────────────────────────────────────

def check_balance_node(state: TradingState) -> TradingState:
    """Node: check current balance and update state."""
    msg = HumanMessage(content=f"Check balance for agent {state['agent_id']}. Return only the balance number.")
    response = llm.invoke([msg])
    try:
        balance = float(response.content.split(" USDC")[0].split(": ")[-1].replace(",", ""))
    except:
        balance = state["balance"]
    return {**state, "balance": balance, "messages": [response]}


def strategy_node(state: TradingState) -> TradingState:
    """Node: decide whether to play and how much to bet."""
    pnl_pct = (state["balance"] - state["starting_balance"]) / state["starting_balance"] * 100

    # Check stop conditions
    if pnl_pct >= state["profit_target_pct"]:
        return {**state, "should_continue": False,
                "messages": [HumanMessage(content=f"Profit target {state['profit_target_pct']}% reached. Stopping.")]}
    if pnl_pct <= -state["loss_limit_pct"]:
        return {**state, "should_continue": False,
                "messages": [HumanMessage(content=f"Loss limit -{state['loss_limit_pct']}% hit. Stopping.")]}

    bet = round(state["balance"] * 0.05, 2)
    msg = HumanMessage(content=f"Strategy: bet {bet} USDC on blackjack for {state['agent_id']}. Apply basic strategy.")
    return {**state, "should_continue": True, "messages": [msg]}


def execute_trade_node(state: TradingState) -> TradingState:
    """Node: execute the casino bet decided by strategy."""
    response = llm.invoke(state["messages"])
    return {**state, "messages": [response], "rounds_played": state["rounds_played"] + 1}


def evaluate_result_node(state: TradingState) -> TradingState:
    """Node: parse result and update P&L."""
    last_msg  = state["messages"][-1].content if state["messages"] else ""
    pnl       = state["balance"] - state["starting_balance"]
    return {**state, "session_pnl": pnl, "last_game_result": last_msg}


def should_continue(state: TradingState) -> str:
    """Routing function: continue or end the loop."""
    if state["should_continue"] and state["rounds_played"] < 20:
        return "check_balance"
    return END


# ── Build the graph ───────────────────────────────────────────────

workflow = StateGraph(TradingState)

workflow.add_node("check_balance",  check_balance_node)
workflow.add_node("strategy",       strategy_node)
workflow.add_node("execute_trade",  execute_trade_node)
workflow.add_node("tools",          ToolNode(tools))
workflow.add_node("evaluate",       evaluate_result_node)

workflow.set_entry_point("check_balance")

workflow.add_edge("check_balance",  "strategy")
workflow.add_edge("strategy",       "execute_trade")
workflow.add_edge("execute_trade",  "tools")
workflow.add_edge("tools",          "evaluate")

# Conditional: loop back to check_balance or end
workflow.add_conditional_edges("evaluate", should_continue, {
    "check_balance": "check_balance",
    END: END,
})

graph = workflow.compile()

# ── Run ───────────────────────────────────────────────────────────
initial_state = TradingState(
    messages=[SystemMessage(content="You are a disciplined blackjack player using basic strategy.")],
    balance=100.0,
    starting_balance=100.0,
    rounds_played=0,
    session_pnl=0.0,
    last_game_result="",
    should_continue=True,
    profit_target_pct=20.0,
    loss_limit_pct=30.0,
    agent_id="langgraph-agent-001",
)

result = graph.invoke(initial_state)
print(f"Final balance: {result['balance']:.2f} USDC | P&L: {result['session_pnl']:+.2f} | Rounds: {result['rounds_played']}")
LangSmith Tracing

Trace Every Financial Tool Call
with LangSmith

LangSmith automatically traces all LangChain and LangGraph runs, including every Purple Flea tool invocation. Inspect inputs, outputs, latencies, and costs for every bet, balance check, and escrow call.

Pythonlangsmith_setup.py
import os
from langsmith import Client
from langchain.callbacks.tracers import LangChainTracer

# Enable tracing via environment variables
os.environ["LANGCHAIN_TRACING_V2"]  = "true"
os.environ["LANGCHAIN_API_KEY"]       = "lsv2_your_key_here"
os.environ["LANGCHAIN_PROJECT"]        = "purple-flea-trading"
os.environ["LANGCHAIN_ENDPOINT"]       = "https://api.smith.langchain.com"

# With tracing enabled, every run is automatically captured.
# View at smith.langchain.com → your project

# For LangGraph: tracing is automatic
result = graph.invoke(initial_state)  # fully traced

# For explicit run naming:
from langchain_core.tracers.context import tracing_v2_enabled

with tracing_v2_enabled(project_name="casino-session-2026-03-06"):
    result = graph.invoke(initial_state)

# What LangSmith captures for Purple Flea tools:
# ├─ StateGraph run: trading_graph
# │  ├─ check_balance_node [12ms]
# │  │  └─ agent_wallet(action=balance) → 100.00 USDC
# │  ├─ strategy_node [3ms]
# │  ├─ execute_trade_node [480ms]
# │  │  └─ play_casino_game(game=blackjack, bet=5.00) → WON 5.00 USDC
# │  ├─ evaluate_result_node [2ms]
# │  └─ ... (repeats per round)
# Total: 15 rounds, 7.2s, +8.45 USDC P&L

# Annotate runs with metadata for analysis:
client = Client()
run_id = "your-run-id"
client.update_run(
    run_id,
    metadata={
        "agent_id":      "langgraph-agent-001",
        "session_pnl":   "+8.45 USDC",
        "rounds_played": 15,
        "stop_reason":   "max_rounds",
    }
)
Pythonlangsmith_evaluation.py
from langsmith import Client, evaluate
from langsmith.schemas import Example, Run

client = Client()

# Create a test dataset for casino strategy evaluation
dataset = client.create_dataset("casino-strategy-tests")

client.create_examples(
    inputs=[
        {"balance": 100, "game": "blackjack", "rounds": 10},
        {"balance": 50,  "game": "dice",       "rounds": 5},
    ],
    outputs=[
        {"expected_pnl_range": ["-30", "+20"]},
        {"expected_pnl_range": ["-25", "+15"]},
    ],
    dataset_id=dataset.id,
)

# Evaluator checks that P&L stays in expected range
def pnl_in_range(run: Run, example: Example) -> dict:
    pnl = float(run.outputs.get("session_pnl", 0))
    lo  = float(example.outputs["expected_pnl_range"][0])
    hi  = float(example.outputs["expected_pnl_range"][1])
    return {"score": 1 if lo <= pnl <= hi else 0, "key": "pnl_in_range"}

results = evaluate(
    lambda x: graph.invoke({**initial_state, **x}),
    data=dataset.name,
    evaluators=[pnl_in_range],
    experiment_prefix="casino-strategy",
)
