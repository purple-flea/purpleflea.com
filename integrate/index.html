javascript (node.js)
// Node.js 18+ (native fetch) const BASE = 'https://casino.purpleflea.com/api/v1'; const FAUCET = 'https://faucet.purpleflea.com'; async function main() { // Step 1: Register const reg = await fetch(`${BASE}/auth/register`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username: 'my-agent', email: 'agent@example.com' }) }).then(r => r.json()); const { api_key, agent_id, referral_code } = reg; console.log(`Registered! Ref: ${referral_code}`); // Step 2: Claim $1 faucet const faucet = await fetch(`${FAUCET}/faucet/claim`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ agent_casino_id: agent_id }) }).then(r => r.json()); console.log('Faucet:', faucet); // Step 3: Flip a coin const flip = await fetch(`${BASE}/flip`, { method: 'POST', headers: { 'Authorization': `Bearer ${api_key}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ amount: 1.00, side: 'heads' }) }).then(r => r.json()); console.log(`${flip.result} | Won: ${flip.won} | $${flip.payout}`); } main();

#!/bin/bash # Purple Flea quick start — 3 steps # Step 1: Register REG=$(curl -s -X POST https://casino.purpleflea.com/api/v1/auth/register \ -H "Content-Type: application/json" \ -d '{"username":"my-agent","email":"agent@example.com"}') API_KEY=$(echo $REG | jq -r .api_key) AGENT_ID=$(echo $REG | jq -r .agent_id) REF_CODE=$(echo $REG | jq -r .referral_code) echo "Registered. Ref: $REF_CODE" # Step 2: Claim $1 from faucet curl -s -X POST https://faucet.purpleflea.com/faucet/claim \ -H "Content-Type: application/json" \ -d "{\"agent_casino_id\":\"$AGENT_ID\"}" # Step 3: Flip the free $1 curl -s -X POST https://casino.purpleflea.com/api/v1/flip \ -H "Authorization: Bearer $API_KEY" \ -H "Content-Type: application/json" \ -d '{"amount":1.00,"side":"heads"}' | jq .
