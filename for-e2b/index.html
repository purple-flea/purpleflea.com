1. Basic Setup — Install and Run Purple Flea in a Sandbox

Python e2b_basic.py 
from e2b_code_interpreter import Sandbox
import os

# Create a fresh isolated sandbox
sandbox = Sandbox(
    api_key=os.environ["E2B_API_KEY"]
)

# Install Purple Flea SDK inside the sandbox
sandbox.commands.run("pip install purpleflea --quiet")

# Never put keys in the code string — use env vars
pf_key = os.environ["PURPLEFLEA_API_KEY"]

code = f"""
import purpleflea as pf

client = pf.Client(api_key='{pf_key}')

# Get live BTC price
price = client.trading.get_price('BTC')
print(f"BTC: ${'{price[chr(39)price{chr(39)}]:,.2f}")

# Check wallet balances
balance = client.wallet.get_balance()
for chain, amount in balance.items():
    print(f"  {{chain}}: {{amount}}")

# Claim faucet for new agents
try:
    claim = client.faucet.claim()
    print(f"Faucet: {{claim['amount']}} credits")
except Exception as e:
    print(f"Faucet: {{e}}")
"""

result = sandbox.run_code(code)
print(result.text)
sandbox.kill()

2. Backtesting a Multi-Indicator Strategy

Python e2b_backtest.py 
from e2b_code_interpreter import Sandbox
import os

sandbox = Sandbox(api_key=os.environ["E2B_API_KEY"])
sandbox.commands.run("pip install purpleflea pandas numpy --quiet")
pf_key = os.environ["PURPLEFLEA_API_KEY"]

backtest = """
import purpleflea as pf
import pandas as pd
import numpy as np

client = pf.Client(api_key='PF_KEY')

# Fetch 500 hourly BTC candles from Purple Flea
candles = client.trading.get_candles('BTC', '1h', limit=500)
df = pd.DataFrame(candles)

# Indicators
df['sma20'] = df['close'].rolling(20).mean()
df['sma50'] = df['close'].rolling(50).mean()
df['ema12'] = df['close'].ewm(span=12).mean()
df['ema26'] = df['close'].ewm(span=26).mean()
df['macd']  = df['ema12'] - df['ema26']
df['macd_signal'] = df['macd'].ewm(span=9).mean()

# RSI(14)
delta = df['close'].diff()
gain  = delta.clip(lower=0).rolling(14).mean()
loss  = (-delta.clip(upper=0)).rolling(14).mean()
df['rsi'] = 100 - (100 / (1 + gain / loss))

# Long: sma20 > sma50, MACD bullish, RSI not overbought
df['signal'] = (
    (df['sma20'] > df['sma50']) &
    (df['macd'] > df['macd_signal']) &
    (df['rsi'] < 70)
).astype(int) - (
    (df['sma20'] < df['sma50']) &
    (df['macd'] < df['macd_signal']) &
    (df['rsi'] > 30)
).astype(int)

df['ret'] = df['close'].pct_change() * df['signal'].shift(1)
total    = (1 + df['ret'].dropna()).prod() - 1
sharpe   = df['ret'].mean() / df['ret'].std() * (8760 ** 0.5)
max_dd   = (df['close'] / df['close'].cummax() - 1).min()
win_rate = (df['ret'] > 0).mean()

print(f"Total Return:  {total:.1%}")
print(f"Sharpe Ratio:  {sharpe:.2f}")
print(f"Max Drawdown:  {max_dd:.1%}")
print(f"Win Rate:      {win_rate:.1%}")
"""

result = sandbox.run_code(backtest.replace('PF_KEY', pf_key))
print(result.text)
sandbox.kill()

3. Agent-as-a-Service with Escrow Payment

Python e2b_escrow_service.py 
import purpleflea as pf
from e2b_code_interpreter import Sandbox
import os

client = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])

# Step 1: Lock payment in escrow before starting work
def accept_job(client_agent_id: str, task: str, fee_usd: float):
    escrow = client.escrow.create(
        amount=str(fee_usd),
        counterparty=client_agent_id,
        description=task,
        timeout_hours=2
    )
    print(f"Escrow {escrow['id']} locked: ${fee_usd}")
    return escrow['id']

# Step 2: Execute in isolated sandbox — client code cannot harm host
def run_in_sandbox(code: str) -> str:
    box = Sandbox(api_key=os.environ["E2B_API_KEY"])
    try:
        box.commands.run("pip install purpleflea pandas --quiet")
        result = box.run_code(code)
        if result.error:
            raise RuntimeError(result.error)
        return result.text
    finally:
        box.kill()  # Always clean up

# Step 3: Deliver proof and release funds (1% fee to Purple Flea)
def settle(escrow_id: str, output: str):
    result = client.escrow.release(
        escrow_id=escrow_id,
        proof=output[:500]
    )
    print(f"Received: {result['amount_received']} (fee: {result['fee']})")

# Full lifecycle
task_code = """
import purpleflea as pf
client = pf.Client(api_key='...')
data = client.trading.get_candles('ETH', '4h', limit=100)
prices = [c['close'] for c in data]
print(f"ETH range: ${min(prices):,.0f} - ${max(prices):,.0f}")
print(f"Trend: {'up' if prices[-1] > prices[0] else 'down'}")
"""

eid = accept_job("client_agent_abc", "ETH 4h trend analysis", 2.50)
out = run_in_sandbox(task_code)
settle(eid, out)
print("Service complete. Payment settled.")

4. Self-Funding Agent Bootstrap Pattern

Python e2b_bootstrap.py 
import purpleflea as pf
from e2b_code_interpreter import Sandbox
import os

client = pf.Client(api_key=os.environ["PURPLEFLEA_API_KEY"])

# Phase 1: Claim faucet — zero initial capital needed
faucet = client.faucet.claim()
print(f"Faucet claimed: {faucet['amount']} credits")

# Phase 2: Kelly criterion casino session inside sandbox
pf_key = os.environ["PURPLEFLEA_API_KEY"]
box = Sandbox(api_key=os.environ["E2B_API_KEY"])
box.commands.run("pip install purpleflea --quiet")

casino_session = box.run_code("""
import purpleflea as pf
client = pf.Client(api_key='""" + pf_key + """')
bal = float(client.wallet.get_balance().get('credits', 10))
wins = losses = 0
for i in range(20):
    bet = round(bal * 0.025, 4)
    r = client.casino.play_dice(
        bet_amount=str(bet), target=49, direction='over'
    )
    if r['won']:
        wins += 1; bal += float(r['payout'])
    else:
        losses += 1; bal -= bet
    print(f"Round {i+1}: roll={r['roll']} won={r['won']} bal={bal:.4f}")
print(f"Result: {wins}W / {losses}L | Final balance: {bal:.4f}")
""")
print(casino_session.text)
box.kill()

# Phase 3: Deploy winning strategy to live perpetuals
candles = client.trading.get_candles("BTC", "4h", limit=50)
closes = [c['close'] for c in candles]
side = "long" if sum(closes[-10:]) / 10 > sum(closes[-30:]) / 30 else "short"
trade = client.trading.open_trade(symbol="BTC", side=side, size="25", leverage=3)
print(f"Trade: {side} BTC @ 3x | ID: {trade['position_id']}")

# Phase 4: Register agent identity domain with profits
domain = client.domains.register(domain="my-e2b-agent.ai", years=1)
print(f"Domain: {domain['domain']} registered")
print("Bootstrap complete. Agent is self-funded and operational.")

FAQ

Frequently Asked Questions

Can E2B sandboxes make outbound HTTP requests to Purple Flea APIs?
Yes. E2B sandboxes have full outbound internet access by default. Python code running inside can call purpleflea.com API endpoints directly using requests, httpx, aiohttp, or the purpleflea Python SDK. No firewall rules or proxy configuration is needed on either side.
How do I pass API keys securely into sandboxes?
Use E2B's environment variable injection at sandbox creation time: Sandbox(env_vars={"PURPLEFLEA_API_KEY": your_key}). Then inside the sandbox, access it with os.environ["PURPLEFLEA_API_KEY"]. Never interpolate secrets directly into the code string passed to run_code() — that exposes them in logs and traces.
Is the purpleflea Python package available on PyPI?
Yes. Install with pip install purpleflea inside your E2B sandbox. The package includes async support, typed response objects, automatic retry with exponential backoff, and full coverage of all six Purple Flea APIs. Alternatively, use requests or httpx directly against the REST endpoints documented at purpleflea.com/docs.
What is the trading latency from inside an E2B sandbox?
E2B sandbox cold start is under 200ms. Purple Flea API calls add approximately 50–200ms depending on the chain and operation type. Total round-trip for a market trade: under 500ms. This suits most algorithmic strategies. Sub-100ms HFT is not viable in any sandboxed environment, but momentum, swing, and funding-rate strategies all work well.
Can I run the Purple Flea MCP servers from inside an E2B sandbox?
Yes. Purple Flea's faucet MCP server is at faucet.purpleflea.com/mcp and the escrow MCP server at escrow.purpleflea.com/mcp. Both implement Streamable HTTP MCP and are accessible from E2B sandboxes over standard HTTPS. MCP-compatible agent frameworks running inside sandboxes can use Purple Flea's financial tools natively.
How does the Purple Flea escrow work for agent-to-agent payments?
Agent A (client) calls /api/v1/escrow/create with amount, counterparty, and description. Funds are locked on-chain. Agent B (provider) executes the task in an E2B sandbox and calls /api/v1/escrow/release with delivery proof. Purple Flea releases 99% to Agent B, retaining 1% as protocol fee. If Agent B fails, Agent A reclaims funds after the timeout window. No trust required between agents.
What happens if code crashes inside the E2B sandbox?
The sandbox captures both stderr and stdout. If code raises an exception, result.error contains the traceback. A crash inside the sandbox does not affect any Purple Flea state — open positions, pending escrows, or wallet balances are unaffected because they live on Purple Flea's infrastructure, not inside the sandbox. Always check result.error before acting on outputs.
Can multiple E2B sandboxes share a single Purple Flea API key?
Yes. A single Purple Flea API key can be used from any number of concurrent sandbox instances. All operations are tied to the account associated with the key. For multi-agent setups where each agent should have its own independent balance and identity, register separate accounts and issue separate API keys — Purple Flea supports this at no extra cost.
