Implementation

YieldAggregator class: monitor and rebalance across all 6 endpoints

Drop this class into your agent runtime. It polls all 6 Purple Flea services, computes yield per dollar deployed, and recommends rebalancing actions when yield skew exceeds a configurable threshold.

YieldAggregator.js JavaScript / Node.js 
// YieldAggregator.js — Purple Flea yield optimization for AI agents
// Monitors all 6 services, computes yield/dollar, and recommends rebalancing

const SERVICES = {
  casino:   { base: 'https://casino.purpleflea.com/api/v1',   referralRate: 0.10 },
  trading:  { base: 'https://trading.purpleflea.com/api/v1',  referralRate: 0.20 },
  wallet:   { base: 'https://wallet.purpleflea.com/api/v1',   referralRate: 0.10 },
  domains:  { base: 'https://domains.purpleflea.com/api/v1',  referralRate: 0.15 },
  faucet:   { base: 'https://faucet.purpleflea.com',          referralRate: 0.15 },
  escrow:   { base: 'https://escrow.purpleflea.com',          referralRate: 0.15, fee: 0.01 },
};

class YieldAggregator {
  constructor(apiKeys, referralCodes, opts = {}) {
    this.apiKeys = apiKeys;           // { casino: 'pf_live_...', trading: '...', ... }
    this.referralCodes = referralCodes; // { casino: 'ref_...', trading: '...', ... }
    this.rebalanceThreshold = opts.rebalanceThreshold ?? 0.25; // 25% drift
    this.pollIntervalMs = opts.pollIntervalMs ?? 60_000;
    this.state = {};
  }

  async fetchServiceStats(service) {
    const { base } = SERVICES[service];
    const key = this.apiKeys[service];
    const res = await fetch(`${base}/agents/me/stats`, {
      headers: { 'Authorization': `Bearer ${key}` },
    });
    if (!res.ok) throw new Error(`${service} stats failed: ${res.status}`);
    return res.json();
  }

  computeYieldPerDollar(stats) {
    // Returns annualized yield coefficient per dollar deployed
    const referralMonthly =
      (stats.referral_commission_30d ?? 0) +
      (stats.l2_commission_30d ?? 0) +
      (stats.l3_commission_30d ?? 0);
    const activeMonthly = stats.active_pnl_30d ?? 0;
    const capital = stats.capital_deployed_usd ?? 1;
    return ((referralMonthly + activeMonthly) / capital) * 12; // annualized
  }

  async snapshot() {
    const results = await Promise.allSettled(
      Object.keys(SERVICES).map(async svc => {
        const stats = await this.fetchServiceStats(svc);
        return { svc, stats, yield: this.computeYieldPerDollar(stats) };
      })
    );
    this.state = {};
    for (const r of results) {
      if (r.status === 'fulfilled') {
        this.state[r.value.svc] = r.value;
      }
    }
    return this.state;
  }

  rebalanceRecommendations() {
    const entries = Object.values(this.state);
    if (!entries.length) return [];
    const totalYield = entries.reduce((s, e) => s + e.yield, 0);
    const avgYield = totalYield / entries.length;
    return entries
      .filter(e => e.yield < avgYield * (1 - this.rebalanceThreshold))
      .map(e => ({
        service: e.svc,
        action: 'reduce_allocation',
        currentYield: e.yield.toFixed(4),
        avgYield: avgYield.toFixed(4),
        recommendation: `Shift capital from ${e.svc} to higher-yield services`,
      }));
  }

  printDashboard() {
    console.log('\n=== Purple Flea Yield Aggregator Dashboard ===');
    for (const [svc, data] of Object.entries(this.state)) {
      const yieldPct = (data.yield * 100).toFixed(1);
      console.log(`  ${svc.padEnd(10)} ${yieldPct.padStart(7)}% APY  [referralRate: ${SERVICES[svc].referralRate * 100}%]`);
    }
    const recs = this.rebalanceRecommendations();
    if (recs.length) {
      console.log('\n  Rebalancing recommended:');
      recs.forEach(r => console.log(`    ⤷ ${r.recommendation}`));
    }
    console.log('============================================\n');
  }

  async start() {
    console.log('[YieldAggregator] Starting yield monitor across 6 services...');
    const tick = async () => {
      await this.snapshot();
      this.printDashboard();
    };
    await tick();
    return setInterval(tick, this.pollIntervalMs);
  }
}
