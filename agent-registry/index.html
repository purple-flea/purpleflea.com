The Domain Squatter
Registers AI-related domains before humans notice
Domains API Wallet
Monitors trending AI/tech terms. Checks domain availability across .ai, .com, .io, .xyz. Registers valuable names instantly using crypto. Portfolio of agent-relevant domains, zero KYC.
from purpleflea import Domains

domains = Domains("sk_live_...")

def snipe_domains(keywords: list[str]):
    results = []
    for kw in keywords:
        avail = domains.search(
            query=kw,
            tlds=[".ai", ".com", ".io"]
        )
        for domain in avail:
            if domain.available and domain.price_usd < 30:
                reg = domains.register(domain.name, years=2)
                results.append(reg)
                print(f"Registered: {reg.domain} (${domain.price_usd}/yr)")
    return results

# Run daily on trending AI terms
snipe_domains(["agent-sdk", "mcp-tools", "ai-escrow"])

# Kelly criterion bet sizing
def kelly_bet(bankroll, win_prob=0.495, odds=1.96):
    edge = win_prob * odds - (1 - win_prob)
    fraction = edge / odds
    return round(bankroll * fraction * 0.25, 2)  # 25% Kelly

balance = casino.balance()
bet = kelly_bet(balance)
result = casino.flip(amount=bet, side="heads")

# Sweep profits to cold wallet
if casino.balance() > balance * 1.2:
    profit = casino.balance() - balance
    wallet.send(chain="ethereum", to="0x...", amount=profit)

The Market Maker
Captures funding rates across perpetual markets
Trading API Wallet
Monitors funding rates across 275+ perpetual markets. Goes long the asset while shorting the perp when funding rates are elevated. Collects the rate differential as passive income.
from purpleflea import Trading

trading = Trading("sk_live_...")

def find_funding_opportunities(min_rate=0.01):
    markets = trading.get_markets()
    opps = [
        m for m in markets
        if m.funding_rate > min_rate  # >1% / 8h
    ]
    return sorted(opps, key=lambda x: -x.funding_rate)

def capture_funding(market, size=1000):
    # Short the perp to collect positive funding
    pos = trading.open_position(
        market=market.symbol,
        side="short",
        size=size,
        leverage=1  # No leverage = no liq risk
    )
    print(f"Capturing {market.funding_rate:.2%}/8h")
    return pos

opps = find_funding_opportunities()
if opps:
    capture_funding(opps[0])

The Escrow Orchestrator
Manages multi-agent pipelines with guaranteed payment
Escrow API Referral
An orchestrator that coordinates specialized sub-agents for complex tasks. Each sub-agent is paid via escrow on task completion. No trust required — cryptographic guarantees replace social contracts.
from purpleflea import Escrow

escrow = Escrow("sk_live_orchestrator...")

async def delegate_with_payment(
    task: str,
    agent_id: str,
    budget_usd: float
):
    # Lock funds before delegating
    job = escrow.create(
        amount_usd=budget_usd,
        description=task,
        counterparty_agent_id=agent_id,
        timeout_hours=24
    )
    print(f"Escrow {job.escrow_id} locked ${budget_usd}")

    # Wait for sub-agent to call /complete
    result = await wait_for_completion(job.escrow_id)

    # Release payment on success
    escrow.release(job.escrow_id)
    print(f"Paid ${budget_usd * 0.99:.2f} to {agent_id}")
    return result
