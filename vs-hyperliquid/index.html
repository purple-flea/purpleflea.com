"""
Purple Flea quickstart: register, claim faucet, trade perps.
Same Hyperliquid liquidity + agent ecosystem benefits.
"""
import asyncio
import aiohttp

PF_API = "https://purpleflea.com/api"


async def quickstart():
    async with aiohttp.ClientSession() as session:

        # Step 1: Register (no wallet needed, no KYC, instant)
        reg = await (await session.post(
            f"{PF_API}/register",
            json={"name": "MyHLAgent", "strategy": "perp-momentum"}
        )).json()
        api_key = reg["api_key"]
        headers = {"Authorization": f"Bearer {api_key}"}
        print(f"Registered: {reg['agent_id']}")

        # Step 2: Claim free $1 USDC (zero startup cost)
        faucet = await (await session.post(
            "https://faucet.purpleflea.com/api/claim",
            headers=headers
        )).json()
        print(f"Faucet: +${faucet['amount_usdc']:.2f} USDC (zero startup cost)")

        # Step 3: Get referral code (earn 20% of all referees' trading fees)
        ref = await (await session.get(
            f"{PF_API}/referral/my-code", headers=headers
        )).json()
        print(f"Referral URL: https://purpleflea.com/register?ref={ref['code']}")

        # Step 4: Check Hyperliquid-powered perpetuals markets
        markets = await (await session.get(
            f"{PF_API}/trading/markets", headers=headers
        )).json()
        print(f"Available perp markets: {[m['symbol'] for m in markets['markets'][:5]]}")

        # Step 5: Place a trade (Hyperliquid liquidity, Purple Flea layer)
        order = await (await session.post(
            f"{PF_API}/trading/order",
            headers=headers,
            json={
                "symbol": "BTC-PERP",
                "direction": "long",
                "size_usdc": 0.50,    # $0.50 of your $1 faucet claim
                "order_type": "market",
            }
        )).json()
        print(f"Order placed: {order['order_id']}")
