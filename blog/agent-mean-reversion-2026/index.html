Mean Reversion Trading for AI Agents 2026: Z-Score & Bollinger Band Signals

March 6, 2026 18 min read Purple Flea Team

Mean reversion is one of the oldest and most battle-tested trading strategies in financial markets. The core premise: prices that deviate significantly from their historical mean tend to revert back. For AI agents operating in crypto markets 24/7, mean reversion strategies are particularly attractive — there's no fatigue, no emotional bias, and an agent can monitor hundreds of pairs simultaneously.

This guide builds a complete mean reversion trading agent from the ground up: Z-score signals, Bollinger Bands, position sizing, and execution via Purple Flea's Trading API.

1. The Mean Reversion Hypothesis

In crypto, mean reversion works best in specific contexts:

  • Correlated pairs — ETH/BTC spread tends to revert when one spikes without the other
  • Funding rate extremes — perpetual futures with extreme funding eventually flip
  • Intraday overreactions — news spikes on low-cap assets often reverse within hours
  • Stablecoin depegs — USDC/USDT should always return to 1.0

When mean reversion fails: Trending markets. A 3-sigma move in BTC during a bull run is not a reversion opportunity — it's a continuation. Always check trend direction before fading.

2. Z-Score Signal Implementation

The Z-score measures how many standard deviations the current price is from its rolling mean. A Z-score above +2 suggests overbought conditions; below -2 suggests oversold.

import numpy as np
import requests
from collections import deque

class ZScoreSignal:
    """
    Rolling Z-score mean reversion signal.
    Lookback: number of periods for rolling mean/std calculation.
    """

    def __init__(self, lookback: int = 20, entry_z: float = 2.0, exit_z: float = 0.5):
        self.lookback  = lookback
        self.entry_z   = entry_z
        self.exit_z    = exit_z
        self.prices    = deque(maxlen=lookback)

    def update(self, price: float) -> dict:
        self.prices.append(price)
        if len(self.prices) < self.lookback:
            return {"signal": "insufficient_data", "z_score": None}

        arr    = np.array(self.prices)
        mean   = arr.mean()
        std    = arr.std()

        if std < 1e-8:
            return {"signal": "flat", "z_score": 0}

        z = (price - mean) / std

        if z >= self.entry_z:
            signal = "short"    # Price too high — expect reversion down
        elif z <= -self.entry_z:
            signal = "long"     # Price too low — expect reversion up
        elif abs(z) <= self.exit_z:
            signal = "exit"     # Back near mean — close position
        else:
            signal = "hold"     # In the middle — do nothing

        return {
            "signal":    signal,
            "z_score":   round(z, 3),
            "mean":      round(mean, 4),
            "std":       round(std, 4),
            "price":     price,
            "deviation": round((price - mean) / mean * 100, 3),
        }

# Example usage
signal_gen = ZScoreSignal(lookback=20, entry_z=2.0)

# Simulate price stream
prices = [3000, 3010, 2990, 3005, 2995, 3020, 3000, 2980,
          3010, 3000, 2990, 3000, 3015, 3005, 2995, 3000,
          3010, 3005, 3000, 3200]  # Spike!

for p in prices:
    result = signal_gen.update(p)
    if result["signal"] in ("long", "short"):
        print(f"SIGNAL: {result['signal'].upper()} at ${p} | Z={result['z_score']}")

3. Bollinger Bands Implementation

Bollinger Bands are a visualization of the Z-score signal: they draw the mean ± N standard deviations. Trading touches of the upper band (short) and lower band (long) is a classic mean reversion approach.

class BollingerBands:
    """
    Standard Bollinger Bands with %B and bandwidth indicators.
    %B = 0 means price at lower band; %B = 1 means price at upper band.
    """

    def __init__(self, period: int = 20, num_std: float = 2.0):
        self.period  = period
        self.num_std = num_std
        self.prices  = deque(maxlen=period)

    def update(self, price: float) -> dict:
        self.prices.append(price)
        if len(self.prices) < self.period:
            return {"ready": False}

        arr    = np.array(self.prices)
        mid    = arr.mean()
        std    = arr.std()
        upper  = mid + self.num_std * std
        lower  = mid - self.num_std * std

        pct_b     = (price - lower) / (upper - lower) if upper != lower else 0.5
        bandwidth = (upper - lower) / mid  # Squeeze indicator

        return {
            "ready":     True,
            "upper":     round(upper, 4),
            "mid":       round(mid, 4),
            "lower":     round(lower, 4),
            "pct_b":     round(pct_b, 4),
            "bandwidth": round(bandwidth, 4),
            "signal":    "long"  if pct_b < 0.05 else ("short" if pct_b > 0.95 else "hold"),
        }

    def is_squeeze(self, bandwidth_threshold: float = 0.02) -> bool:
        """Bollinger squeeze: low bandwidth often precedes a big move."""
        result = self.update(self.prices[-1] if self.prices else 0)
        return result.get("bandwidth", 1) < bandwidth_threshold

4. Combined Signal with Confirmation Filter

Using Z-score and Bollinger Bands together reduces false signals. Only trade when both agree, and additionally confirm that RSI is not in strong trend territory.

class MeanReversionStrategy:
    """
    Combines Z-score + Bollinger Bands + RSI filter.
    Only enters when all three confirm mean reversion.
    """

    def __init__(self, lookback: int = 20):
        self.zscore = ZScoreSignal(lookback=lookback, entry_z=2.0)
        self.bb     = BollingerBands(period=lookback, num_std=2.0)
        self.rsi_prices = deque(maxlen=15)

    def calc_rsi(self, period: int = 14) -> float:
        if len(self.rsi_prices) < period + 1:
            return 50

        prices_arr = np.array(self.rsi_prices)
        deltas = np.diff(prices_arr)
        gains  = deltas[deltas > 0]
        losses = abs(deltas[deltas < 0])
        avg_gain = gains.mean() if len(gains) > 0 else 0
        avg_loss = losses.mean() if len(losses) > 0 else 0.001
        rs = avg_gain / avg_loss
        return round(100 - (100 / (1 + rs)), 2)

    def get_signal(self, price: float) -> dict:
        self.rsi_prices.append(price)
        z_result  = self.zscore.update(price)
        bb_result = self.bb.update(price)
        rsi       = self.calc_rsi()

        z_sig  = z_result.get("signal")
        bb_sig = bb_result.get("signal", "hold")

        # Both must agree
        if z_sig == bb_sig == "long" and rsi < 35:
            confidence = min((abs(z_result["z_score"]) - 2.0) / 2.0, 1.0)
            return {"action": "long",  "confidence": confidence, "z": z_result["z_score"], "rsi": rsi}

        if z_sig == bb_sig == "short" and rsi > 65:
            confidence = min((z_result["z_score"] - 2.0) / 2.0, 1.0)
            return {"action": "short", "confidence": confidence, "z": z_result["z_score"], "rsi": rsi}

        if z_sig == "exit":
            return {"action": "exit", "confidence": 1.0, "z": z_result["z_score"], "rsi": rsi}

        return {"action": "hold", "z": z_result.get("z_score"), "rsi": rsi}

5. Execution via Purple Flea Trading API

import requests, time
from datetime import datetime

TRADING_API = "https://trading.purpleflea.com"
AGENT_ID    = "mean-rev-agent-001"
API_KEY     = "pf_live_your_key"

class MeanReversionExecutor:

    def __init__(self, symbol: str = "ETH-USDC", max_position_usd: float = 500):
        self.symbol          = symbol
        self.max_position    = max_position_usd
        self.current_side    = None   # "long" or "short"
        self.strategy        = MeanReversionStrategy(lookback=20)
        self.pnl             = 0.0
        self.entry_price     = None

    def place_order(self, side: str, size_usd: float) -> dict:
        r = requests.post(f"{TRADING_API}/order",
            headers={"X-Agent-Id": AGENT_ID, "Authorization": f"Bearer {API_KEY}"},
            json={
                "symbol":    self.symbol,
                "side":      side,
                "size_usd":  size_usd,
                "order_type":"market"
            }
        )
        return r.json()

    def run_bar(self, price: float):
        sig = self.strategy.get_signal(price)
        action = sig["action"]
        ts     = datetime.utcnow().strftime("%H:%M:%S")

        if action == "long" and self.current_side is None:
            size = self.max_position * sig["confidence"]
            order = self.place_order("buy", size)
            self.current_side = "long"
            self.entry_price  = price
            print(f"[{ts}] LONG ${size:.0f} @ ${price:.2f} | Z={sig['z']:.2f} RSI={sig['rsi']:.0f}")

        elif action == "short" and self.current_side is None:
            size = self.max_position * sig["confidence"]
            order = self.place_order("sell", size)
            self.current_side = "short"
            self.entry_price  = price
            print(f"[{ts}] SHORT ${size:.0f} @ ${price:.2f} | Z={sig['z']:.2f} RSI={sig['rsi']:.0f}")

        elif action == "exit" and self.current_side is not None:
            close_side = "sell" if self.current_side == "long" else "buy"
            self.place_order(close_side, self.max_position)
            trade_pnl = (price - self.entry_price) / self.entry_price
            if self.current_side == "short":
                trade_pnl = -trade_pnl
            self.pnl += trade_pnl * self.max_position
            print(f"[{ts}] EXIT @ ${price:.2f} | P&L: ${trade_pnl*self.max_position:.2f} | Total: ${self.pnl:.2f}")
            self.current_side = None
            self.entry_price  = None

# --- Main loop ---

    r = requests.get("https://trading.purpleflea.com/ticker",

while True:
