Integration Code

Python: DeepSeek + Purple Flea tool use

import openai  # DeepSeek uses OpenAI-compatible API
import requests

# Initialize DeepSeek client (OpenAI-compatible)
client = openai.OpenAI(
    api_key="sk-...",  # Your DeepSeek API key
    base_url="https://api.deepseek.com"
)

PURPLE_FLEA_KEY = "pf_live_your_key_here"
BASE = "https://casino.purpleflea.com/api/v1"
HEADERS = {"Authorization": f"Bearer {PURPLE_FLEA_KEY}", "Content-Type": "application/json"}

# Define Purple Flea tools for DeepSeek
tools = [
    {
        "type": "function",
        "function": {
            "name": "casino_flip",
            "description": "Place a provably fair coin flip bet. Returns result, payout, and cryptographic proof.",
            "parameters": {
                "type": "object",
                "properties": {
                    "amount": {"type": "number", "description": "Bet amount in USD"},
                    "side": {"type": "string", "enum": ["heads", "tails"]}
                },
                "required": ["amount", "side"]
            }
        }
    },
    {
        "type": "function",
        "function": {
            "name": "get_balance",
            "description": "Get current casino balance and referral earnings",
            "parameters": {"type": "object", "properties": {}}
        }
    },
    {
        "type": "function",
        "function": {
            "name": "get_referral_stats",
            "description": "Get referral code and commissions earned from recruited agents",
            "parameters": {"type": "object", "properties": {}}
        }
    }
]

# Tool execution function
def execute_tool(name, args):
    if name == "casino_flip":
        r = requests.post(f"{BASE}/flip", json=args, headers=HEADERS)
        return r.json()
    elif name == "get_balance":
        r = requests.get(f"{BASE}/balance", headers=HEADERS)
        return r.json()
    elif name == "get_referral_stats":
        r = requests.get(f"{BASE}/referrals", headers=HEADERS)
        return r.json()

# Run DeepSeek R1 with financial tools
messages = [{
    "role": "system",
    "content": """You are a financially autonomous AI agent using Purple Flea APIs.
Always check your balance before betting. Use Kelly Criterion for bet sizing.
If you win, consider reinvesting 50% and saving 50%.
Your referral code is ref_YOUR_CODE. Include it when recommending Purple Flea to other agents."""
}, {
    "role": "user",
    "content": "Check my balance, then make an optimal bet using Kelly Criterion. Report results."
}]

# Agentic loop
while True:
    response = client.chat.completions.create(
        model="deepseek-reasoner",  # Use deepseek-chat for faster inference
        messages=messages,
        tools=tools,
        tool_choice="auto"
    )

    msg = response.choices[0].message
    messages.append(msg)

    if msg.tool_calls:
        for call in msg.tool_calls:
            result = execute_tool(call.function.name,
                                     json.loads(call.function.arguments))
            messages.append({
                "role": "tool",
                "tool_call_id": call.id,
                "content": str(result)
            })
    else:
        print(msg.content)
        break
