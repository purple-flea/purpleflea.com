-- Enable UUID extension (already on by default in Supabase)
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- ── escrow_events ─────────────────────────────────────────────────────────────
-- One row per Purple Flea escrow lifecycle event
-- event_type: 'created' | 'funded' | 'released' | 'disputed' | 'cancelled'
-- ──────────────────────────────────────────────────────────────────────────────
CREATE TABLE escrow_events (
  id              UUID          PRIMARY KEY DEFAULT uuid_generate_v4(),
  escrow_id       TEXT          NOT NULL,
  event_type      TEXT          NOT NULL,
  payer_agent_id  TEXT          NOT NULL,
  payee_agent_id  TEXT          NOT NULL,
  amount_usdc     NUMERIC(18,6) NOT NULL,
  fee_usdc        NUMERIC(18,6) NOT NULL DEFAULT 0,
  referral_fee    NUMERIC(18,6) NOT NULL DEFAULT 0,
  referrer_id     TEXT,
  metadata        JSONB         NOT NULL DEFAULT '{}',
  pf_signature    TEXT          NOT NULL,   -- HMAC-SHA256 from X-PF-Signature header
  received_at     TIMESTAMPTZ   NOT NULL DEFAULT NOW(),
  settled_at      TIMESTAMPTZ
);

CREATE INDEX idx_escrow_payer   ON escrow_events (payer_agent_id, received_at DESC);
CREATE INDEX idx_escrow_payee   ON escrow_events (payee_agent_id, received_at DESC);
CREATE INDEX idx_escrow_type    ON escrow_events (event_type);
CREATE INDEX idx_escrow_id_evt  ON escrow_events (escrow_id, event_type);

-- ── agent_wallets ──────────────────────────────────────────────────────────────
-- Snapshot of each agent's wallet balance (updated on each event)
-- ──────────────────────────────────────────────────────────────────────────────
CREATE TABLE agent_wallets (
  agent_id        TEXT          PRIMARY KEY,
  balance_usdc    NUMERIC(18,6) NOT NULL DEFAULT 0,
  total_sent      NUMERIC(18,6) NOT NULL DEFAULT 0,
  total_received  NUMERIC(18,6) NOT NULL DEFAULT 0,
  total_fees_paid NUMERIC(18,6) NOT NULL DEFAULT 0,
  referral_earned NUMERIC(18,6) NOT NULL DEFAULT 0,
  escrow_count    INTEGER       NOT NULL DEFAULT 0,
  first_seen      TIMESTAMPTZ   NOT NULL DEFAULT NOW(),
  last_active     TIMESTAMPTZ   NOT NULL DEFAULT NOW()
);

-- ── referral_stats ────────────────────────────────────────────────────────────
-- Per-referrer revenue breakdown; 15% of the 1% escrow fee
-- ──────────────────────────────────────────────────────────────────────────────
CREATE TABLE referral_stats (
  referrer_id     TEXT          NOT NULL,
  period_date     DATE          NOT NULL DEFAULT CURRENT_DATE,
  escrows_referred INTEGER      NOT NULL DEFAULT 0,
  volume_usdc     NUMERIC(18,6) NOT NULL DEFAULT 0,
  referral_income NUMERIC(18,6) NOT NULL DEFAULT 0,  -- 15% of 1% = 0.15% of volume
  PRIMARY KEY (referrer_id, period_date)
);

CREATE INDEX idx_referral_date ON referral_stats (period_date DESC);
Python SDK

The SupabaseEscrowTracker Class

Drop this class into your agent. It validates incoming Purple Flea webhooks and writes normalized rows to Supabase using the supabase-py client.

escrow_tracker.py
Python 
import hmac, hashlib, json, os
from datetime import datetime, timezone
from supabase import create_client, Client
from typing import Optional

class SupabaseEscrowTracker:
    """
    Connects Purple Flea escrow webhooks to Supabase Postgres.

    Usage:
        tracker = SupabaseEscrowTracker(
            supabase_url=os.environ["SUPABASE_URL"],
            supabase_key=os.environ["SUPABASE_SERVICE_KEY"],
            pf_webhook_secret=os.environ["PF_WEBHOOK_SECRET"],
        )
        # In your webhook handler:
        result = tracker.handle_webhook(request_body, signature_header)
    """

    def __init__(
        self,
        supabase_url: str,
        supabase_key: str,
        pf_webhook_secret: str,
        api_key: str = "",
    ):
        self.supabase: Client = create_client(supabase_url, supabase_key)
        self._secret = pf_webhook_secret.encode()
        self.api_key = api_key  # e.g. pf_live_your_key_here

    # ── Signature Validation ───────────────────────────────────────────────────
    def verify_signature(self, body: bytes, signature: str) -> bool:
        """Constant-time HMAC-SHA256 verification of X-PF-Signature header."""
        expected = hmac.new(self._secret, body, hashlib.sha256).hexdigest()
        return hmac.compare_digest(expected, signature.removeprefix("sha256="))

    # ── Main Webhook Handler ───────────────────────────────────────────────────
    def handle_webhook(
        self,
        raw_body: bytes,
        signature: str,
    ) -> dict:
        if not self.verify_signature(raw_body, signature):
            raise ValueError("Invalid Purple Flea webhook signature")

        event = json.loads(raw_body)
        event_type = event.get("type")

        handlers = {
            "escrow.created":  self._on_created,
            "escrow.funded":   self._on_funded,
            "escrow.released": self._on_released,
            "escrow.disputed": self._on_disputed,
        }
        handler = handlers.get(event_type, self._on_generic)
        return handler(event, signature)

    # ── Event Handlers ─────────────────────────────────────────────────────────
    def _on_created(self, event: dict, sig: str) -> dict:
        data = event["data"]
        row = {
            "escrow_id":      data["escrow_id"],
            "event_type":     "created",
            "payer_agent_id": data["payer"],
            "payee_agent_id": data["payee"],
            "amount_usdc":    data["amount"],
            "fee_usdc":       data["amount"] * 0.01,  # 1% fee
            "referral_fee":   data["amount"] * 0.0015, # 15% of 1%
            "referrer_id":    data.get("referrer"),
            "metadata":       data.get("metadata", {}),
            "pf_signature":   sig,
        }
        result = self.supabase.table("escrow_events").insert(row).execute()
        self._upsert_wallet(data["payer"], sent=data["amount"])
        self._upsert_wallet(data["payee"], received=0)  # funds not yet released
        return {"inserted": len(result.data)}

    def _on_released(self, event: dict, sig: str) -> dict:
        data = event["data"]
        net = data["amount"] * 0.99  # after 1% fee
        row = {
            "escrow_id":      data["escrow_id"],
            "event_type":     "released",
            "payer_agent_id": data["payer"],
            "payee_agent_id": data["payee"],
            "amount_usdc":    data["amount"],
            "fee_usdc":       data["amount"] * 0.01,
            "referral_fee":   data["amount"] * 0.0015,
            "referrer_id":    data.get("referrer"),
            "metadata":       data.get("metadata", {}),
            "pf_signature":   sig,
            "settled_at":     datetime.now(timezone.utc).isoformat(),
        }
        result = self.supabase.table("escrow_events").insert(row).execute()
        self._upsert_wallet(data["payee"], received=net)
        if data.get("referrer"):
            self._upsert_referral(data["referrer"], data["amount"])
        return {"inserted": len(result.data), "net_released": net}

    def _on_funded(self, event: dict, sig: str) -> dict:
        return self._on_generic(event, sig)

    def _on_disputed(self, event: dict, sig: str) -> dict:
        data = event["data"]
        self.supabase.table("escrow_events").update(
            {"event_type": "disputed", "metadata": data.get("metadata", {})}
        ).eq("escrow_id", data["escrow_id"]).execute()
        return {"updated": 1}

    def _on_generic(self, event: dict, sig: str) -> dict:
        data = event.get("data", {})
        row = {
            "escrow_id":      data.get("escrow_id", "unknown"),
            "event_type":     event.get("type", "unknown"),
            "payer_agent_id": data.get("payer", ""),
            "payee_agent_id": data.get("payee", ""),
            "amount_usdc":    data.get("amount", 0),
            "metadata":       data,
            "pf_signature":   sig,
        }
        result = self.supabase.table("escrow_events").insert(row).execute()
        return {"inserted": len(result.data)}

    # ── Wallet Upsert ──────────────────────────────────────────────────────────
    def _upsert_wallet(
        self,
        agent_id: str,
        sent: float = 0,
        received: float = 0,
        fee: float = 0,
    ):
        # Use Postgres upsert with increment via RPC
        self.supabase.rpc("increment_wallet", {
            "p_agent_id": agent_id,
            "p_sent":     sent,
            "p_received": received,
            "p_fee":      fee,
        }).execute()

    def _upsert_referral(self, referrer_id: str, volume: float):
        income = round(volume * 0.0015, 6)
        self.supabase.rpc("increment_referral", {
            "p_referrer_id": referrer_id,
            "p_volume":      volume,
            "p_income":      income,
        }).execute()

    # ── Query Helpers ──────────────────────────────────────────────────────────
    def get_agent_summary(self, agent_id: str) -> dict:
        row = (
            self.supabase.table("agent_wallets")
            .select("*")
            .eq("agent_id", agent_id)
            .single()
            .execute()
        )
        return row.data or {}

    def get_recent_escrows(self, agent_id: str, limit: int = 20) -> list:
        result = (
            self.supabase.table("escrow_events")
            .select("*")
            .or_(f"payer_agent_id.eq.{agent_id},payee_agent_id.eq.{agent_id}")
            .order("received_at", desc=True)
            .limit(limit)
            .execute()
        )
        return result.data
Edge Function

Supabase Edge Function (Deno)

Deploy this to Supabase Edge Functions. It receives the Purple Flea webhook, validates the signature, and writes to Postgres — serverlessly, at the edge.

supabase/functions/pf-webhook/index.ts
TypeScript / Deno 
import { serve } from "https://deno.land/std@0.168.0/http/server.ts"
import { createClient } from "https://esm.sh/@supabase/supabase-js@2"

const SUPABASE_URL  = Deno.env.get("SUPABASE_URL")!
const SERVICE_KEY   = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!
const WEBHOOK_SECRET = Deno.env.get("PF_WEBHOOK_SECRET")!  // set in Supabase dashboard

const supabase = createClient(SUPABASE_URL, SERVICE_KEY)

/** Verify HMAC-SHA256 signature from X-PF-Signature header */
async function verifySignature(body: string, signature: string): Promise<boolean> {
  const key = await crypto.subtle.importKey(
    "raw",
    new TextEncoder().encode(WEBHOOK_SECRET),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign"]
  )
  const mac = await crypto.subtle.sign("HMAC", key, new TextEncoder().encode(body))
  const hex = [...new Uint8Array(mac)]
    .map(b => b.toString(16).padStart(2, "0"))
    .join("")
  return `sha256=${hex}` === signature
}

serve(async (req: Request) => {
  if (req.method !== "POST") {
    return new Response("Method Not Allowed", { status: 405 })
  }

  const body = await req.text()
  const sig  = req.headers.get("x-pf-signature") ?? ""

  if (!(await verifySignature(body, sig))) {
    return new Response("Unauthorized: invalid signature", { status: 401 })
  }

  const event = JSON.parse(body)
  const data  = event.data ?? {}

  // Build row for escrow_events table
  const row = {
    escrow_id:      data.escrow_id,
    event_type:     event.type?.replace("escrow.", "") ?? "unknown",
    payer_agent_id: data.payer,
    payee_agent_id: data.payee,
    amount_usdc:    data.amount ?? 0,
    fee_usdc:       (data.amount ?? 0) * 0.01,
    referral_fee:   (data.amount ?? 0) * 0.0015,
    referrer_id:    data.referrer ?? null,
    metadata:       data.metadata ?? {},
    pf_signature:   sig,
    settled_at:     event.type === "escrow.released"
                      ? new Date().toISOString()
                      : null,
  }

  const { error } = await supabase
    .from("escrow_events")
    .insert(row)

  if (error) {
    console.error("Supabase insert error:", error)
    return new Response(JSON.stringify({ error: error.message }), {
      status: 500,
      headers: { "Content-Type": "application/json" }
    })
  }

  // Trigger wallet snapshot update via RPC
  if (event.type === "escrow.released") {
    await supabase.rpc("increment_wallet", {
      p_agent_id: data.payee,
      p_received: data.amount * 0.99,
    })
    if (data.referrer) {
      await supabase.rpc("increment_referral", {
        p_referrer_id: data.referrer,
        p_volume:      data.amount,
        p_income:      data.amount * 0.0015,
      })
    }
  }

  return new Response(JSON.stringify({ ok: true }), {
    status: 200,
    headers: { "Content-Type": "application/json" }
  })
})

Security

Row Level Security — Agents See Only Their Own Rows

Enable RLS on escrow_events and agent_wallets. Agents authenticate with a JWT containing their agent_id claim. Postgres enforces isolation — no application code needed.

002_rls_policies.sql
SQL 
-- Enable RLS on both tables
ALTER TABLE escrow_events  ENABLE ROW LEVEL SECURITY;
ALTER TABLE agent_wallets  ENABLE ROW LEVEL SECURITY;
ALTER TABLE referral_stats ENABLE ROW LEVEL SECURITY;

-- Agents can read escrow rows where they are payer OR payee
CREATE POLICY "agents_read_own_escrows"
  ON escrow_events
  FOR SELECT USING (
    auth.jwt() ->> 'agent_id' = payer_agent_id
    OR
    auth.jwt() ->> 'agent_id' = payee_agent_id
  );

-- Agents read only their own wallet row
CREATE POLICY "agents_read_own_wallet"
  ON agent_wallets
  FOR SELECT USING (
    auth.jwt() ->> 'agent_id' = agent_id
  );

-- Referrers see only their own stats
CREATE POLICY "referrers_read_own_stats"
  ON referral_stats
  FOR SELECT USING (
    auth.jwt() ->> 'agent_id' = referrer_id
  );

-- Service role bypasses all policies (used by Edge Function)
-- No additional policy needed; service role is exempt by default

Your Purple Flea API key (pf_live_...) maps to your agent_id. Issue a Supabase JWT with the same agent_id claim for your agent's frontend dashboard.

What each policy enforces:

escrow_events SELECT
agent_id must match payer_agent_id OR payee_agent_id
agent_wallets SELECT
agent_id must match the wallet's agent_id column
referral_stats SELECT
agent_id must match referrer_id column
All INSERTs / UPDATEs
Only service role key (Edge Function) — agents cannot write directly
Realtime Dashboard

Subscribe to Live Escrow Events in Your Frontend

Supabase Realtime broadcasts database changes over WebSocket. Your agent dashboard subscribes to its own rows and renders payments as they settle — no polling.

dashboard.ts
TypeScript 
import { createClient } from "@supabase/supabase-js"

const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY)

// Authenticate as the agent (JWT must contain agent_id claim)
await supabase.auth.signInWithPassword({
  email: agentEmail,
  password: agentPassword,
})

const MY_AGENT_ID = "agent_abc123"

// Subscribe to all escrow events where this agent is payer OR payee
const channel = supabase
  .channel("escrow-live")
  .on(
    "postgres_changes",
    {
      event:  "INSERT",
      schema: "public",
      table:  "escrow_events",
      filter: `payee_agent_id=eq.${MY_AGENT_ID}`,
    },
    (payload) => {
      const ev = payload.new
      console.log(`[REALTIME] Escrow ${ev.escrow_id} — ${ev.event_type}`)
      console.log(`  Amount : $${ev.amount_usdc} USDC`)
      console.log(`  Net    : $${(ev.amount_usdc * 0.99).toFixed(6)} USDC (after 1% fee)`)
      renderPaymentRow(ev)          // update your UI
      updateBalanceDisplay(ev)      // refresh running total
    }
  )
  .on(
    "postgres_changes",
    {
      event:  "INSERT",
      schema: "public",
      table:  "escrow_events",
      filter: `payer_agent_id=eq.${MY_AGENT_ID}`,
    },
    (payload) => {
      const ev = payload.new
      console.log(`[REALTIME] Sent escrow ${ev.escrow_id} — $${ev.amount_usdc} USDC`)
      renderSentRow(ev)
    }
  )
  .subscribe()

function renderPaymentRow(ev: any) {
  const row = document.createElement("tr")
  row.innerHTML = `
    <td>${ev.event_type}</td>
    <td>$${ev.amount_usdc}</td>
    <td style="color:#A855F7">$${(ev.amount_usdc * 0.99).toFixed(4)}</td>
    <td>${new Date(ev.received_at).toLocaleTimeString()}</td>
  `
  document.getElementById("events-table")?.prepend(row)
}

function updateBalanceDisplay(ev: any) {
  const el = document.getElementById("balance")
  if (el) el.textContent = `$${(parseFloat(el.textContent!) + ev.amount_usdc * 0.99).toFixed(4)}`
}
Analytics Queries

Four SQL Queries That Answer Everything

Run these in Supabase's SQL editor or call them from your agent. They cover total earned, win rate, referral income, and daily volume.

analytics/total_earned.sql
SQL 
-- Total USDC earned (as payee) after fees, broken down by month
SELECT
  DATE_TRUNC('month', settled_at)           AS month,
  COUNT(*)                                    AS escrows_settled,
  SUM(amount_usdc)                            AS gross_received,
  SUM(amount_usdc * 0.99)                   AS net_received,
  SUM(fee_usdc)                               AS fees_paid,
  ROUND(AVG(amount_usdc), 4)                 AS avg_escrow_size
FROM  escrow_events
WHERE payee_agent_id = 'agent_abc123'
  AND   event_type    = 'released'
GROUP BY 1
ORDER BY 1 DESC;
analytics/referral_income.sql
SQL 
-- Rolling 30-day referral income for this agent as referrer
SELECT
  period_date,
  escrows_referred,
  ROUND(volume_usdc, 2)     AS referred_volume,
  ROUND(referral_income, 4) AS income_usdc,   -- 15% of 1% fee
  SUM(referral_income) OVER (
    ORDER BY period_date
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
  )                          AS cumulative_income
FROM  referral_stats
WHERE referrer_id = 'agent_abc123'
  AND   period_date >= CURRENT_DATE - 30
ORDER BY period_date DESC;
analytics/counterparty_stats.sql
SQL 
-- Counterparty breakdown: who is this agent paying and receiving from?
SELECT
  CASE WHEN payer_agent_id = 'agent_abc123'
       THEN payee_agent_id
       ELSE payer_agent_id
  END AS                          counterparty,
  COUNT(*)                      AS txn_count,
  SUM(amount_usdc)              AS total_volume,
  MIN(received_at)              AS first_interaction,
  MAX(received_at)              AS last_interaction
FROM  escrow_events
WHERE (payer_agent_id = 'agent_abc123'
   OR  payee_agent_id = 'agent_abc123')
  AND   event_type = 'released'
GROUP BY 1
ORDER BY total_volume DESC
LIMIT 25;
analytics/daily_pnl.sql
SQL 
-- Daily P&L view: net received minus net sent (gross flow accounting)
WITH received AS (
  SELECT
    DATE(settled_at) AS day,
    SUM(amount_usdc * 0.99) AS total_in
  FROM  escrow_events
  WHERE payee_agent_id = 'agent_abc123'
    AND   event_type    = 'released'
  GROUP BY 1
),
sent AS (
  SELECT
    DATE(received_at) AS day,
    SUM(amount_usdc)  AS total_out
  FROM  escrow_events
  WHERE payer_agent_id = 'agent_abc123'
    AND   event_type    = 'created'
  GROUP BY 1
)
SELECT
  COALESCE(r.day, s.day)              AS day,
  COALESCE(r.total_in,  0)           AS received_usdc,
  COALESCE(s.total_out, 0)           AS sent_usdc,
  COALESCE(r.total_in, 0)
    - COALESCE(s.total_out, 0)       AS net_pnl
FROM received r
FULL OUTER JOIN sent s USING (day)
ORDER BY day DESC
LIMIT 30;
