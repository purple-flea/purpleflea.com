Quickstart

EscrowQueryEngine:
Lock-Before-Query Pattern

Subclass LlamaIndex's BaseQueryEngine to wrap any existing query engine with automatic escrow. The wrapper locks USDC before retrieval, releases on success, and refunds on failure — completely transparent to the underlying index.

Pythonescrow_query_engine.py
import os, requests
from typing import Optional
from llama_index.core import VectorStoreIndex, SimpleDirectoryReader
from llama_index.core.query_engine import BaseQueryEngine
from llama_index.core.schema import QueryBundle
from llama_index.core.response.schema import RESPONSE_TYPE

PF_KEY   = os.environ["PURPLEFLEA_API_KEY"]
AGENT_ID = os.environ["PURPLEFLEA_AGENT_ID"]
CLIENT_ID = os.environ["CLIENT_AGENT_ID"]
ESCROW   = "https://escrow.purpleflea.com"
HEADERS  = {"Authorization": f"Bearer {PF_KEY}", "Content-Type": "application/json"}


class EscrowQueryEngine(BaseQueryEngine):
    """A LlamaIndex QueryEngine wrapper with per-query escrow billing."""

    def __init__(
        self,
        base_engine: BaseQueryEngine,
        price_per_query_usdc: float = 0.10,
        payer_agent_id: str = CLIENT_ID,
        payee_agent_id: str = AGENT_ID,
        auto_release: bool = True
    ):
        self._engine       = base_engine
        self._price        = price_per_query_usdc
        self._payer        = payer_agent_id
        self._payee        = payee_agent_id
        self._auto_release = auto_release
        self._query_log: list = []
        super().__init__()

    def _lock_escrow(self, task_desc: str) -> Optional[str]:
        """Lock USDC in escrow for one query. Returns escrow_id or None."""
        try:
            r = requests.post(f"{ESCROW}/api/escrow/create", headers=HEADERS, json={
                "payer_agent_id": self._payer,
                "payee_agent_id": self._payee,
                "amount_usdc":     self._price,
                "task_description": task_desc
            }, timeout=5)
            return r.json().get("escrow_id")
        except Exception as e:
            print(f"[EscrowQE] Lock failed: {e}")
            return None

    def _settle_escrow(self, escrow_id: str, approved: bool) -> None:
        action = "release" if approved else "dispute"
        try:
            requests.post(f"{ESCROW}/api/escrow/{escrow_id}/{action}",
                headers=HEADERS, json={"agent_id": self._payee}, timeout=5)
        except Exception as e:
            print(f"[EscrowQE] Settle failed: {e}")

    def _query(self, query_bundle: QueryBundle) -> RESPONSE_TYPE:
        query_str = query_bundle.query_str
        escrow_id = self._lock_escrow(f"rag-query: {query_str[:80]}")
        try:
            response = self._engine.query(query_bundle)
            if escrow_id and self._auto_release:
                self._settle_escrow(escrow_id, approved=True)
            self._query_log.append({
                "query": query_str, "escrow_id": escrow_id,
                "status": "paid", "amount": self._price
            })
            return response
        except Exception as e:
            if escrow_id:
                self._settle_escrow(escrow_id, approved=False)
            self._query_log.append({
                "query": query_str, "escrow_id": escrow_id,
                "status": "refunded", "error": str(e)
            })
            raise

    def get_billing_report(self) -> dict:
        """Return a billing summary for reconciliation."""
        paid    = [q for q in self._query_log if q["status"] == "paid"]
        refunds = [q for q in self._query_log if q["status"] == "refunded"]
        return {
            "total_queries":     len(self._query_log),
            "paid_queries":      len(paid),
            "refunded_queries":  len(refunds),
            "total_earned_usdc": len(paid) * self._price,
            "price_per_query":   self._price,
            "log":               self._query_log
        }


# ── Usage ─────────────────────────────────────────────────────

index     = VectorStoreIndex.from_documents(documents)
base_qe   = index.as_query_engine()

Pythonbilling_workflow.py
import os, requests, asyncio
from llama_index.core.workflow import (
    Workflow, StartEvent, StopEvent, step, Event
)
from llama_index.core import VectorStoreIndex, SimpleDirectoryReader

PF_KEY  = os.environ["PURPLEFLEA_API_KEY"]
ESCROW  = "https://escrow.purpleflea.com"
AGENT   = os.environ["PURPLEFLEA_AGENT_ID"]
HEADERS = {"Authorization": f"Bearer {PF_KEY}"}

class DepositEvent(Event):
    client_id:  str
    total_usdc: float
    queries:    list[str]
    price_each: float

class QueryEvent(Event):
    query:      str
    escrow_id:  str
    price:      float

class ResultEvent(Event):
    query:      str
    answer:     str
    escrow_id:  str
    paid:       bool


class PayPerQueryWorkflow(Workflow):
    def __init__(self, index: VectorStoreIndex, **kwargs):
        super().__init__(**kwargs)
        self._qe      = index.as_query_engine()
        self._results = []

    @step
    async def start(self, ev: StartEvent) -> DepositEvent:
        queries    = ev.get("queries", [])
        price_each = ev.get("price_each", 0.10)
        client_id  = ev.get("client_id")
        needed     = len(queries) * price_each
        print(f"[Workflow] Need {needed} USDC for {len(queries)} queries")
        return DepositEvent(client_id=client_id, total_usdc=needed,
                             queries=queries, price_each=price_each)

    @step
    async def open_query_escrows(self, ev: DepositEvent) -> list[QueryEvent]:
        events = []
        for q in ev.queries:
            r = requests.post(f"{ESCROW}/api/escrow/create", headers=HEADERS, json={
                "payer_agent_id": ev.client_id,
                "payee_agent_id": AGENT,
                "amount_usdc":     ev.price_each,
                "task_description": f"rag-query: {q[:60]}"
            }).json()
            events.append(QueryEvent(query=q, escrow_id=r["escrow_id"],
                                     price=ev.price_each))
        return events

    @step
    async def execute_query(self, ev: QueryEvent) -> ResultEvent:
        try:
            response = self._qe.query(ev.query)
            requests.post(f"{ESCROW}/api/escrow/{ev.escrow_id}/release",
                headers=HEADERS, json={"agent_id": AGENT})
            return ResultEvent(query=ev.query, answer=str(response),
                                escrow_id=ev.escrow_id, paid=True)
        except Exception as e:
            requests.post(f"{ESCROW}/api/escrow/{ev.escrow_id}/dispute",
                headers=HEADERS, json={"agent_id": AGENT})
            return ResultEvent(query=ev.query, answer=f"Error: {e}",
                                escrow_id=ev.escrow_id, paid=False)

    @step
    async def collect_results(self, ev: ResultEvent) -> Optional[StopEvent]:
        self._results.append(ev)
        if len(self._results) == self.total_expected:
            paid   = [r for r in self._results if r.paid]
            report = {
                "total":   len(self._results),
                "paid":    len(paid),
                "earned":  len(paid) * 0.10,
                "results": [{"q": r.query, "a": r.answer[:100]}
                             for r in self._results]
            }
            return StopEvent(result=report)
        return None


async def main():
    docs  = SimpleDirectoryReader("./financial_docs").load_data()
    index = VectorStoreIndex.from_documents(docs)
    wf    = PayPerQueryWorkflow(index=index, timeout=60)
    wf.total_expected = 3
    result = await wf.run(
        client_id  = "client-agent-007",
        price_each = 0.10,
        queries    = [
            "What is the current ratio for FY2025?",
            "Summarize cash flow from operations.",
            "What are the top three risk disclosures?"
        ]
    )
    print(result)

asyncio.run(main())

import os, requests
from llama_index.core import VectorStoreIndex, SimpleDirectoryReader
from llama_index.core.tools import QueryEngineTool, ToolMetadata
from llama_index.core.query_engine import SubQuestionQueryEngine
from llama_index.llms.openai import OpenAI

PF_KEY  = os.environ["PURPLEFLEA_API_KEY"]
AGENT   = os.environ["PURPLEFLEA_AGENT_ID"]
CLIENT  = os.environ["CLIENT_AGENT_ID"]
ESCROW  = "https://escrow.purpleflea.com"
HEADERS = {"Authorization": f"Bearer {PF_KEY}"}


def paid_query_engine_tool(
    index: VectorStoreIndex,
    name: str,
    description: str,
    price_usdc: float = 0.10
) -> QueryEngineTool:
    """Create a QueryEngineTool that charges escrow per invocation."""
    base_engine = index.as_query_engine()

    class BilledEngine:
        def query(self, query_str: str):
            r = requests.post(f"{ESCROW}/api/escrow/create", headers=HEADERS, json={
                "payer_agent_id": CLIENT, "payee_agent_id": AGENT,
                "amount_usdc": price_usdc,
                "task_description": f"{name}: {query_str[:60]}"
            }).json()
            eid = r.get("escrow_id")
            try:
                result = base_engine.query(query_str)
                if eid:
                    requests.post(f"{ESCROW}/api/escrow/{eid}/release",
                        headers=HEADERS, json={"agent_id": AGENT})
                return result
            except:
                if eid:
                    requests.post(f"{ESCROW}/api/escrow/{eid}/dispute",
                        headers=HEADERS, json={"agent_id": AGENT})
                raise

    return QueryEngineTool(
        query_engine=BilledEngine(),
        metadata=ToolMetadata(name=name, description=description)
    )


# Build two paid indexes at different price points
financials_index = VectorStoreIndex.from_documents(
    SimpleDirectoryReader("./financials").load_data())
risk_index       = VectorStoreIndex.from_documents(
    SimpleDirectoryReader("./risk_disclosures").load_data())

financials_tool = paid_query_engine_tool(
    index=financials_index,
    name="financial_data",
    description="Answers questions about revenue, margins, cash flow, and P&L.",
    price_usdc=0.25    # premium rate for financial data
)
risk_tool = paid_query_engine_tool(
    index=risk_index,

    description="Answers questions about risk factors and regulatory disclosures.",

)

llm = OpenAI(model="gpt-4o", api_key=os.environ["OPENAI_API_KEY"])

# Two escrow transactions: $0.25 (financials) + $0.10 (risk)
response = router_engine.query(
    "Compare revenue growth rate with the primary market risks identified."
)

Usage Tracking

IndexStore Billing
Reconciliation

Wrap your IndexStore with a UsageBillingStore that records every query operation alongside its escrow ID and payment status. Clients receive an auditable billing report linking each query to its escrow record.

Pythonusage_billing_store.py
import time, json, requests, os
from typing import Optional
from llama_index.core.storage.index_store import SimpleIndexStore

PF_KEY  = os.environ["PURPLEFLEA_API_KEY"]
ESCROW  = "https://escrow.purpleflea.com"
HEADERS = {"Authorization": f"Bearer {PF_KEY}"}


class UsageBillingStore:
    """Decorates any IndexStore with per-query usage and billing tracking."""

    def __init__(self, base_store: SimpleIndexStore, agent_id: str):
        self._store    = base_store
        self._agent_id = agent_id
        self._log: list[dict] = []

    def record_query(
        self,
        query: str,
        nodes_retrieved: int,
        latency_ms: float,
        escrow_id: Optional[str] = None,
        amount_usdc: float = 0.0,
        status: str = "paid"
    ) -> None:
        self._log.append({
            "timestamp":       time.time(),
            "query":           query,
            "nodes_retrieved": nodes_retrieved,
            "latency_ms":      latency_ms,
            "escrow_id":       escrow_id,
            "amount_usdc":     amount_usdc,
            "status":          status,
        })

    def get_client_report(self, client_id: str) -> dict:
        paid     = [e for e in self._log if e["status"] == "paid"]
        refunds  = [e for e in self._log if e["status"] == "refunded"]
        total_charged = sum(e["amount_usdc"] for e in paid)
        avg_lat  = sum(e["latency_ms"] for e in self._log) / max(len(self._log), 1)
        avg_nodes = (sum(e["nodes_retrieved"] for e in self._log)
                     / max(len(self._log), 1))
        return {
            "client_id":          client_id,
            "total_queries":      len(self._log),
            "paid_queries":       len(paid),
            "refunded_queries":   len(refunds),
            "total_charged_usdc": round(total_charged, 4),
            "avg_latency_ms":     round(avg_lat, 2),
            "avg_nodes_per_query": round(avg_nodes, 1),
            "escrow_ids":         [e["escrow_id"] for e in paid],
            "full_log":           self._log
        }

    def export_json(self, filepath: str) -> None:
        with open(filepath, "w") as f:
            json.dump(self._log, f, indent=2)

from llama_index.core.storage import StorageContext

store   = SimpleIndexStore()
tracker = UsageBillingStore(base_store=store, agent_id="rag-operator-001")
storage = StorageContext.from_defaults(index_store=store)
docs    = SimpleDirectoryReader("./docs").load_data()
index   = VectorStoreIndex.from_documents(docs, storage_context=storage)
engine  = index.as_query_engine(similarity_top_k=5)

def tracked_query(query: str, price: float = 0.10, client: str = "client-001"):
    r = requests.post(f"{ESCROW}/api/escrow/create", headers=HEADERS, json={
        "payer_agent_id": client, "payee_agent_id": "rag-operator-001",
        "amount_usdc": price, "task_description": query[:60]
    }).json()
    eid = r.get("escrow_id")
    t0  = time.time()
    resp = engine.query(query)
    lat  = (time.time() - t0) * 1000
    requests.post(f"{ESCROW}/api/escrow/{eid}/release",
        headers=HEADERS, json={"agent_id": "rag-operator-001"})
    tracker.record_query(query, nodes_retrieved=len(resp.source_nodes),
        latency_ms=lat, escrow_id=eid, amount_usdc=price, status="paid")
    return resp

tracked_query("What is the EBITDA for FY2025?")
tracked_query("Explain the debt covenant terms.")

Pythonreact_query_market.py
import os, requests
from llama_index.core.agent import ReActAgent
from llama_index.core.tools import FunctionTool
from llama_index.llms.openai import OpenAI

PF_KEY  = os.environ["PURPLEFLEA_API_KEY"]
ESCROW  = "https://escrow.purpleflea.com"
FAUCET  = "https://faucet.purpleflea.com"
ROUTER  = "router-lli-001"
HEADERS = {"Authorization": f"Bearer {PF_KEY}"}

SPECIALISTS = {
    "crypto_analyst":  {"agent_id": "spec-crypto-001", "price": 0.20},
    "macro_economist": {"agent_id": "spec-macro-001",  "price": 0.15},
    "risk_analyst":    {"agent_id": "spec-risk-001",   "price": 0.12},
}

def hire_specialist(specialist: str, query: str) -> str:
    """Route a query to a specialist RAG agent via escrow payment."""
    spec = SPECIALISTS.get(specialist)
    if not spec:
        return f"Unknown specialist. Options: {list(SPECIALISTS)}"
    r = requests.post(f"{ESCROW}/api/escrow/create", headers=HEADERS, json={
        "payer_agent_id": ROUTER,
        "payee_agent_id": spec["agent_id"],
        "amount_usdc":     spec["price"],
        "task_description": f"specialist-query: {query[:60]}"
    }).json()
    eid = r.get("escrow_id")
    answer = f"[{specialist} response to: {query}]"
    if eid:
        requests.post(f"{ESCROW}/api/escrow/{eid}/release",
            headers=HEADERS, json={"agent_id": ROUTER})
    return f"{specialist} answered (escrow {eid} settled, ${spec['price']}): {answer}"

def list_specialists() -> str:
    """List available query specialists and their per-query prices."""
    return "\n".join(f"  {k}: ${v['price']}/query" for k, v in SPECIALISTS.items())

def register_new_specialist(specialist_id: str, referral: str) -> str:
    """Register a new specialist agent at the faucet to give it initial USDC."""
    r = requests.post(f"{FAUCET}/api/claim", json={
        "agent_id": specialist_id, "referral_code": referral
    }).json()
    return f"Registered {specialist_id}: {r}"

llm = OpenAI(model="gpt-4o", api_key=os.environ["OPENAI_API_KEY"])
agent = ReActAgent.from_tools([
    FunctionTool.from_defaults(fn=hire_specialist,          name="hire_specialist"),
    FunctionTool.from_defaults(fn=list_specialists,         name="list_specialists"),
    FunctionTool.from_defaults(fn=register_new_specialist,  name="register_specialist"),
], llm=llm, verbose=True,
system_prompt="""You route financial queries to specialist RAG agents.
Always list_specialists first, then hire the best match via escrow.
If a needed specialist is missing, register a new one first.""")

response = agent.chat(
    "I need: (1) BTC correlation with gold in 2025, "
    "(2) How Fed policy affects crypto, (3) VaR-95 for a diversified crypto portfolio."
)
print(response)
